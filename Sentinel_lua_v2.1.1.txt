--[[
  SENTINEL LOCAL v2.1.1 / LUASNAPPER STATIC RECOVERY
  Bounded whitelist-system tester. Executor compatibility must be tested on-device.

  TOOLS (v2.1 — bounded runtime extraction)
    1. Target capture          selected execution + nested loads on its coroutine
    2. Runtime tools           compiled-function constants + environment observations
    3. Save & Stop             finalizes reports and writes Sentinel_LATEST.txt
    4. Force Stop              instant hook/timer cancellation; skips heavy analysis
    5. Constant inventory      bounded strings/numbers with secret-like values redacted
    6. Environment surface     executor API availability, device and key presence metadata
    7. Endpoint inventory      redacted URLs plus whitelist/auth term groups
    8. Source structure        function/branch/loop/table metrics
    9. Surface reconstruction  decimal/hex/Base64/string.char/concat/XOR-table passes
   10. CFF and VM scoring      dispatcher states and byte/bit-operation density
   11. Family detection        LuaSnapper / MoonSec / 00fuscator / StackFuscate / HexBlob-VM / …
   12. Evidence ledger         origin + severity + confidence for every finding
   13. Artifact manifest       lists the reports produced for each session
   14. Cooperative budgets     120 KB mobile scan window with regular scheduler yields
   15. Payload translation     HexBlob-VM: unwrap hex→XOR→LZ4 container, deserialize the
                               VM program, recover the constant pool (strings/numbers)
                               to deobfuscated_payload.txt — runs on Save, 90 s budget
   16. Build self-configuring  per-build randomized constant tags are read out of the
                               loader's own dispatch (candidates validated by parse), and
                               the string key is recovered by scoring — nothing hardcoded
   17. LuaSnapper full_deob    statically unwraps the final container and exports the
                               complete decoded VM IR (protos, constants, instructions)
                               to full_deob.txt without executing the protected source
   18. Stage hop capture       each nested load stage is written live to
                               extracts/stage_NN.txt; hops logged in stage_hops.txt
   19. Unknown ledger          unclassified runtime events append to unknown.txt
   20. Text-only artifacts     every session report is .txt (no .json / .lua outputs)

  Usage:
    loadstring(...)()                 -- auto-attaches; run target next
    Sentinel.Attach()                 -- optional/manual re-arm
    Sentinel.Analyze(source)          -- static-only; never executes source
    Sentinel.FullDeob(source)         -- static-only; returns full_deob.txt path
    -- or
    Sentinel.Run([=[ obfuscated ]=], { idle = 8 })
    Sentinel.Stop()                   -- or wait for 2-minute collection
    Sentinel.ForceStop()              -- instant: restore hooks, skip analysis
    Sentinel.SetCollection(120)
    Sentinel.SelfTest()               -- harmless nested-load fixture + pass/fail report
    Sentinel.Loadstring(source)()     -- explicit capture if automatic binding is unsupported
    Sentinel.Bind(callback)           -- wrap a task callback to preserve target attribution

  CAPTURE PROFILE (the shipped default)
    Full capture, scoped to the script under test.  Every hook past the
    loadstring wrapper sits on a function the whole place calls, so what is
    installed and what it records are two separate decisions:

      LITE_MODE = false      all sinks armed
      SCOPED_CAPTURE = true  in_scope() drops anything that is not running on
                             the target's own threads, so prints, GetService
                             calls and global writes made by the game do not
                             enter the report
      CAPTURE.remotes = true   FireServer/InvokeServer, via __namecall

    __namecall is a metamethod, so once installed it runs on every method call
    in the place and there is no way to install it for one script only.  What
    the scope buys is the rejection path: target_depth is zero except while the
    captured script is actually on a stack, so almost every one of those calls
    costs a single upvalue compare before returning to the real metamethod --
    no coroutine.running(), no table hash, no getnamecallmethod.  Only the
    target's own remote traffic is resolved and recorded, bounded by
    MAX_REMOTE_EVENTS.

    HTTP is the exception that is deliberately unscoped: game:HttpGet and
    HttpGetAsync are hooked directly (via hookfunction where available, never
    read out of __namecall) and the executor request family is wrapped, so a
    key check fired from a spawned task is still recorded.  These fire rarely
    enough that recording all of them is cheap.

    Scope follows the target off its own stack.  Threads it starts are bound
    -- task.spawn, task.defer, task.delay, coroutine.wrap, coroutine.create and
    the deprecated spawn and delay globals -- and so are callbacks it hands to
    signal:Connect, Once, ConnectParallel or the legacy connect.  Both bind at
    schedule time rather than run time, because that is the only moment the
    target is still identifiable, and both bind only when the caller is
    already in scope: a task or connection belonging to the game passes
    through untouched.  Signal interception needs a working hookfunction; when
    there is none the session says so in its evidence rather than quietly
    losing the callbacks.  Sentinel.Bind(callback) remains the escape hatch for
    anything handed to an API Sentinel does not wrap.
    tool_diagnostics.txt names the active scope, every hook that installed
    and how, and any that could not.

    LITE_MODE = true reverts to the loadstring wrapper alone, allocating none
    of the above and reserving no report files it cannot fill.

  Output: Sentinel/Script00N/
]]

local VERSION = "2.1.1-luasnapper"
local bit32 = bit32 or bit
if not bit32 then
  -- Arithmetic fallback for executors whose parser/runtime has no bit32.
  local function u32(n) return (tonumber(n) or 0) % 4294967296 end
  local function band(a, b)
    a, b = u32(a), u32(b)
    local out, place = 0, 1
    for _ = 1, 32 do
      if (a % 2) >= 1 and (b % 2) >= 1 then out = out + place end
      a, b, place = math.floor(a / 2), math.floor(b / 2), place * 2
    end
    return out
  end
  local function bxor(a, b)
    a, b = u32(a), u32(b)
    local out, place = 0, 1
    for _ = 1, 32 do
      if (a % 2) ~= (b % 2) then out = out + place end
      a, b, place = math.floor(a / 2), math.floor(b / 2), place * 2
    end
    return out
  end
  bit32 = { band = band, bxor = bxor }
end

local function measured(name, fn, ...)
  local started = os.clock()
  local marked = false
  if debug and type(debug.profilebegin) == "function" and type(debug.profileend) == "function" then
    marked = pcall(debug.profilebegin, "Sentinel/" .. name)
  end
  local result = table.pack(pcall(fn, ...))
  if marked then pcall(debug.profileend) end
  local ms = (os.clock() - started) * 1000
  local stat = S.timings[name] or { calls = 0, total_ms = 0, max_ms = 0 }
  stat.calls = stat.calls + 1; stat.total_ms = stat.total_ms + ms
  stat.max_ms = math.max(stat.max_ms, ms); S.timings[name] = stat
  return table.unpack(result, 1, result.n)
end

local work_queue, worker_scheduled, job_active = {}, false, false
local function drain_work()
  if job_active then return end
  local started = os.clock()
  while #work_queue > 0 and not S.force_stopped do
    local job = table.remove(work_queue, 1)
    job_active = true
    local ok = pcall(job)
    job_active = false
    if not ok then S.queue_dropped = S.queue_dropped + 1 end
    if os.clock() - started >= 0.0015 then break end
  end
end
local function enqueue_work(job)
  if #work_queue >= 8 or S.force_stopped then
    S.queue_dropped = S.queue_dropped + 1; return false
  end
  work_queue[#work_queue + 1] = job
  if not worker_scheduled then
    worker_scheduled = true
    task.defer(function()
      while #work_queue > 0 and not S.force_stopped do
        drain_work()
        if #work_queue > 0 then task.wait() end
      end
      worker_scheduled = false
    end)
  end
  return true
end

-- ── FS ──────────────────────────────────────────────────────────────
local function has_fs()
  return type(writefile) == "function" and type(makefolder) == "function"
end
local function isfolder_safe(p)
  if type(isfolder) ~= "function" then return false end
  local ok, r = pcall(isfolder, p)
  return ok and r
end
local function isfile_safe(p)
  if type(isfile) ~= "function" then return false end
  local ok, r = pcall(isfile, p)
  return ok and r
end
local function ensure_dir(p)
  if not has_fs() then
    S.fs_error = "writefile/makefolder unavailable"
    return false
  end
  if isfolder_safe(p) then return true end
  local ok, err = pcall(makefolder, p)
  if ok then return true end
  if isfolder_safe(p) then return true end
  -- Some Android executors expose makefolder/writefile but omit isfolder.
  -- In that case an "already exists" error is harmless; the following write
  -- is the authoritative filesystem check.
  if type(isfolder) ~= "function" then return true end
  S.fs_error = "makefolder " .. tostring(p) .. ": " .. tostring(err)
  return false
end
local function write(path, content)
  if not has_fs() then
    S.fs_error = "writefile/makefolder unavailable"
    return false, S.fs_error
  end
  content = tostring(content or "")
  local bytes = #content
  if S.write_bytes + bytes > MAX_LOCAL_WRITE_BYTES then
    if not S.write_limit_hit then
      S.write_limit_hit = true
      warn("[Sentinel] 5 MB session write limit reached; remaining output skipped")
    end
    return false
  end
  S.write_bytes = S.write_bytes + bytes
  local ok, err = measured("write", writefile, path, content)
  if not ok then
    S.write_bytes = math.max(0, S.write_bytes - bytes)
    S.fs_error = "writefile " .. tostring(path) .. ": " .. tostring(err)
    if not S.fs_warned then
      S.fs_warned = true
      warn("[Sentinel] local write failed: " .. tostring(S.fs_error))
    end
  end
  return ok, err
end
local function append(path, content)
  content = tostring(content or "")
  if type(appendfile) == "function" then
    local bytes = #content
    if S.write_bytes + bytes > MAX_LOCAL_WRITE_BYTES then
      if not S.write_limit_hit then
        S.write_limit_hit = true
        warn("[Sentinel] 5 MB session write limit reached; remaining output skipped")
      end
      return false
    end
    S.write_bytes = S.write_bytes + bytes
    local ok, err = measured("write", appendfile, path, content)
    if not ok then S.write_bytes = math.max(0, S.write_bytes - bytes) end
    return ok, err
  end
  local prev = ""
  if isfile_safe(path) and type(readfile) == "function" then
    local ok, data = pcall(readfile, path)
    if ok and type(data) == "string" then prev = data end
  end
  return write(path, prev .. content)
end

end

-- Cheap, non-cryptographic content fingerprint. This lets reports correlate
-- repeated payloads without copying their contents into runtime telemetry.
local function fingerprint(s)
  s = tostring(s or "")
  local total = #s
  if total > MAX_FINGERPRINT_BYTES then
    local half = math.floor(MAX_FINGERPRINT_BYTES / 2)
    s = s:sub(1, half) .. s:sub(total - half + 1) .. ":bytes=" .. tostring(total)
  end
  local h = 5381
  for i = 1, #s do
    h = bit32.band(h * 33 + string.byte(s, i), 0xFFFFFFFF)
  end
  return string.format("djb2-%08x", h)
end

local function record_runtime(kind, fields)
  if #S.runtime_events >= MAX_RUNTIME_EVENTS then return end
  fields = fields or {}
  fields.kind = tostring(kind or "event")
  fields.index = #S.runtime_events + 1
  fields.elapsed_s = string.format("%.3f", os.clock() - (S.t0 or os.clock()))
  table.insert(S.runtime_events, fields)
  local k = fields.kind
  local known = {
    loadstring = true, http = true, print = true, remote = true, service = true,
    stage_hop = true, global = true, env = true, const = true, task = true,
  }
  if not known[k] and S.path then
    local detail = {}
    for key, val in pairs(fields) do
      if key ~= "kind" and key ~= "index" then
        detail[#detail + 1] = tostring(key) .. "=" .. tostring(val):sub(1, 200)
      end
    end
    pcall(function()
      append(S.path .. "/unknown.txt", string.format("[%s] kind=%s  %s\n",
        now(), k, table.concat(detail, " | "):sub(1, 4000)))
    end)
  end
end

end

local function count_pattern(src, pat, cap)
  local n = 0
  for _ in tostring(src or ""):gmatch(pat) do
    n = n + 1
    analysis_checkpoint(1)
    if n >= (cap or 1000) then break end

-- ── provenance metrics (BIG plan §4) ────────────────────────────────
-- Bounded structural metrics for a byte string.  Every scan samples at most
-- MAX_FINGERPRINT_BYTES (head+tail, same window fingerprint uses) and yields
-- through analysis_checkpoint, so this stays frame-safe on a megabyte stage.

-- Head+tail sample used by the entropy/profile passes.
local function bounded_sample(s)
  s = tostring(s or "")
  local n = #s
  if n <= MAX_FINGERPRINT_BYTES then return s end
  local half = math.floor(MAX_FINGERPRINT_BYTES / 2)
  return s:sub(1, half) .. s:sub(n - half + 1)
end

-- Byte histogram over the sample: returns counts[0..255] and the sampled size.
local function byte_histogram(s)
  local sample = bounded_sample(s)
  local counts, sn = {}, #sample
  for i = 1, sn do
    local b = string.byte(sample, i)
    counts[b] = (counts[b] or 0) + 1
    if (i % 4096) == 0 then analysis_checkpoint(1) end
  end
  return counts, sn
end

-- Shannon entropy in bits/byte (0..8) from a histogram + total.
local function entropy_from_hist(counts, total)
  if not total or total <= 0 then return 0 end
  local ln2, ent = math.log(2), 0
  for _, c in pairs(counts) do
    if c > 0 then
      local p = c / total
      ent = ent - p * (math.log(p) / ln2)
    end
  end
  return ent
end

-- Shannon entropy of a string (bounded sample).
local function shannon_entropy(s)
  local counts, sn = byte_histogram(s)
  return entropy_from_hist(counts, sn), sn
end

-- Printable / null-byte profile over the sample.
local function byte_profile(s)
  local counts, sn = byte_histogram(s)
  local printable, nul, distinct = 0, 0, 0
  for b, c in pairs(counts) do
    distinct = distinct + 1
    if b == 0 then nul = nul + c
    elseif (b >= 32 and b < 127) or b == 9 or b == 10 or b == 13 then printable = printable + c end
  end
  return {
    sampled_bytes = sn,
    distinct_bytes = distinct,
    printable_ratio = sn > 0 and printable / sn or 0,
    null_ratio = sn > 0 and nul / sn or 0,
  }
end

-- Controlled-test markers present in a stage (BIG plan §6/§8).  Deliberately
-- narrow: only our own FAULMOR_TEST_<X>_<digits> markers, capped.
local function marker_scan(s)
  s = tostring(s or "")
  local found, seen, n = {}, {}, 0
  for m in s:gmatch("FAULMOR_TEST_[A-Z]_%d+") do
    if not seen[m] then
      seen[m] = true; n = n + 1; found[n] = m
      if n >= 8 then break end
    end
    analysis_checkpoint(1)
  end
  return found

-- Coarse SURFACE classification (not the Phase 7 VM classifier): what a blob
-- looks like from entropy + printability alone, with a confidence.
local function surface_classify(entropy, printable_ratio, bytes)
  if not bytes or bytes == 0 then return { class = "empty", confidence = 1 } end
  if entropy >= 7.5 and printable_ratio < 0.30 then
    return { class = "high_entropy_binary", confidence = math.min(0.95, (entropy - 7.5) / 0.5 + 0.6) }
  end
  if entropy >= 6.5 and printable_ratio < 0.55 then
    return { class = "encoded_or_compressed", confidence = 0.7 }
  end
  if printable_ratio >= 0.85 and entropy <= 5.0 then
    return { class = "text_or_source", confidence = 0.8 }
  end
  return { class = "mixed_or_unknown", confidence = 0.4 }
end

-- Container / compression framing by magic bytes (executor-independent -- no
-- API needed, just a byte sniff).  A hit means the blob is a packed container,
-- so its high entropy is expected and it should be handed to Requiem's
-- isolated worker to decompress, not decoded live (BIG plan §9/§16).  zstd is
-- called out because newer executors ship zstddecompress for exactly it.
local function detect_container(s)
  s = tostring(s or "")
  if #s < 4 then return nil end
  local b1, b2, b3, b4 = string.byte(s, 1, 4)
  if b1 == 0x28 and b2 == 0xB5 and b3 == 0x2F and b4 == 0xFD then return "zstd" end
  if b1 == 0x04 and b2 == 0x22 and b3 == 0x4D and b4 == 0x18 then return "lz4_frame" end
  if b1 == 0x1F and b2 == 0x8B then return "gzip" end
  if b1 == 0x50 and b2 == 0x4B and b3 == 0x03 and b4 == 0x04 then return "zip" end
  if b1 == 0x78 and (b2 == 0x01 or b2 == 0x9C or b2 == 0xDA) then return "zlib" end
  -- Luau bytecode chunks start with a small version byte (currently <= 6).
  -- Cheap, and getfunctionbytecode/dumpbytecode output lands here.
  if b1 and b1 >= 1 and b1 <= 6 then return "maybe_luau_bytecode" end
  return nil
end

local function detect_device()
  local exec = ""
  pcall(function()
    exec = tostring((identifyexecutor and identifyexecutor()) or (getexecutorname and getexecutorname()) or "")
  end)
  local low = string.lower(exec)
  local is_mobile_exec = false
  for _, pat in ipairs(MOBILE_EXEC) do
    if low:find(pat, 1, true) then is_mobile_exec = true break end
  end
  local touch, kb = false, true
  pcall(function()
    touch = UserInputService.TouchEnabled == true
    kb = UserInputService.KeyboardEnabled == true
  end)
  local cam = workspace.CurrentCamera
  local vw = (cam and cam.ViewportSize.X) or 1920
  local vh = (cam and cam.ViewportSize.Y) or 1080
  local mobile = is_mobile_exec or (touch and not kb) or (touch and (vw < 900 or vh < 500))
  local layout
  if mobile then
    local w = math.clamp(math.floor(vw * 0.92), 260, 360)
    layout = {
      mobile = true, executor = exec, width = w, height = 168,
      pos = UDim2.new(0.5, -math.floor(w / 2), 1, -(168 + 24)),
      titleSize = 16, phaseSize = 16, subSize = 12, statsSize = 12, sessionSize = 12,
      iconSize = 32, corner = 16, stroke = 2, pad = 14,
    }
  else
    layout = {
      mobile = false, executor = exec, width = 280, height = 148,
      pos = UDim2.new(1, -296, 0, 72),
      titleSize = 14, phaseSize = 15, subSize = 11, statsSize = 11, sessionSize = 11,
      iconSize = 28, corner = 14, stroke = 1.5, pad = 12,
    }
  end
  S.device = layout
  return layout
end

    end)

-- Snapshot only explicitly supplied values. Never invoke table metamethods.
local function snapshot_value(value)
  local seen, nodes, next_id = {}, 0, 0
  local cut = false
  local started = os.clock()
  local function text_value(s)
    local prefix = s:sub(1, 128)
    if prefix:find("[^ -~]") then
      return { type = "string", bytes = #s, encoding = "hex",
        preview = (prefix:gsub(".", function(c) return string.format("%02x", string.byte(c)) end)),
        truncated = #s > 128 }
    end
    return { type = "string", bytes = #s, preview = prefix, truncated = #s > 128 }
  end
  local function secret_key(k)
    if type(k) ~= "string" then return false end
    local low = k:sub(1, 128):lower()
    for _, word in ipairs({ "key", "token", "password", "secret", "cookie", "authorization" }) do
      if low:find(word, 1, true) then return true end
    end
    return false
  end
  local visit
  visit = function(v, depth)
    if nodes >= 256 or os.clock() - started >= 0.0015 then
      cut = true
      return { omitted = "work_budget" }
    end
    nodes = nodes + 1
    local t = type(v)
    if t == "string" then return text_value(v) end
    if t == "boolean" then return { type = t, value = v } end
    if t == "number" then
      if v ~= v or v == math.huge or v == -math.huge then return { type = t, value = tostring(v) } end
      return { type = t, value = v }
    end
    if t ~= "table" then return { type = t } end
    if seen[v] then return { type = t, ref = seen[v] } end
    if depth >= 4 then cut = true; return { type = t, omitted = "depth_limit" } end
    next_id = next_id + 1
    seen[v] = next_id
    local out = { type = t, id = next_id, entries = {} }
    local k = nil
    for i = 1, 65 do
      local item
      k, item = next(v, k)
      if k == nil then break end
      if i > 64 or nodes >= 254 or os.clock() - started >= 0.0015 then
        out.truncated = true; cut = true; break
      end
      local val
      if secret_key(k) then val = { type = type(item), redacted = true }
      else val = visit(item, depth + 1) end
      out.entries[#out.entries + 1] = { key = visit(k, depth + 1), value = val }
    end
    return out
  end
  local root = visit(value, 0)
  return { schema = "sentinel.value-snapshot.v1", root = root, nodes = nodes,
    truncated = cut, scope = "explicit_value_only", named_secrets_redacted = true,
    limits = { nodes = 256, depth = 4, entries_per_table = 64, string_preview_bytes = 128, traversal_ms = 1.5 } }
end

-- Scoped runtime extraction. No GC/prototype walk or game-wide event hooks.
local target_threads = setmetatable({}, { __mode = "k" })
-- Executors run scripts on a task-scheduler coroutine, so coroutine.running()
-- is normally a real thread -- but a host that calls in on the main thread
-- makes it nil, and everything below is keyed by it.  One shared stand-in
-- turns an outright crash into ordinary main-thread attribution.
local MAIN_THREAD = {}
local function current_thread() return coroutine.running() or MAIN_THREAD end
-- How many target frames are on a stack right now, across all threads.  Zero
-- for virtually every call the game makes, which is what lets in_scope reject
-- without touching coroutine.running().
local target_depth = 0

-- The gate that keeps a whitelist test from becoming a whole-game trace.
-- Every secondary hook asks this before doing any work, and with
-- CAPTURE.remotes on it is asked once per method call the place makes, so the
-- rejection path is what matters: target_depth is zero except while the
-- captured script is actually running, so almost every one of those calls
-- costs a single upvalue compare -- no coroutine.running(), no table hash.
-- Only once the target is on a stack do we pay to find out which thread this
-- is.  Sentinel.Bind is how a target's task callback gets into scope.
local function in_scope()
  if not SCOPED_CAPTURE then
    return S.running and not S.force_stopped and not S._finalizing
  end
  if target_depth == 0 then return false end
  if not S.running or S.force_stopped or S._finalizing then return false end
  local context = target_threads[current_thread()]
  if context ~= nil then return context.generation == S.generation end
  -- A bound thread cancelled while suspended never runs its exit path, so the
  -- counter can drift above the map.  Correctness never rested on it -- the
  -- lookup above is the real answer -- but resync so the cheap rejection above
  -- comes back instead of every game call paying for this one.
  if next(target_threads) == nil then target_depth = 0 end
  return false
end
local function reset_runtime_capture()
  S.generation = S.generation + 1
  target_threads = setmetatable({}, { __mode = "k" })
  target_depth = 0
  S.runtime_capture = { selected = false, bytes = 0, loads = {}, constants = {}, vm = {},
    environments = {}, dropped = 0, environment_dropped = 0,
    scope = "selected_function_and_same_coroutine_nested_loads",
    limitations = { "Threads the target starts and callbacks it connects to signals are bound automatically; a callback handed to an API Sentinel does not wrap still requires Sentinel.Bind(callback).",
      "Shared environment changes cannot be attributed exclusively to the target.",
      "Native executor debug calls cannot be preempted by Sentinel." } }
  S.runtime_envs = {}
end

local function runtime_api(name)
  if type(debug) == "table" and type(rawget(debug, name)) == "function" then return rawget(debug, name) end
  for _, env in ipairs(cached_envs) do
    local fn = rawget(env, name)
    if type(fn) == "function" then return fn end
  end
end

local function environment_sample(env)
  local fields, count, key = {}, 0, nil
  local started = os.clock()
  if type(env) ~= "table" then return { status = "unavailable", fields = {} } end
  while count < 64 and os.clock() - started < 0.0015 do
    local value
    key, value = next(env, key)
    if key == nil then return { status = "sampled", fields = fields, complete = true } end
    count = count + 1
    if type(key) == "string" and #key <= 128 then
      local t = type(value)
      local note = { type = t }
      if t == "string" then
        note.bytes = #value; note.fingerprint = fingerprint(value:sub(1, 128)); note.sampled_bytes = math.min(#value, 128)
      elseif t == "number" or t == "boolean" then note.value = tostring(value) end
      local low = key:lower()
      if low:find("key", 1, true) or low:find("token", 1, true) or low:find("secret", 1, true)
        or low:find("password", 1, true) or low:find("cookie", 1, true) then
        note = { type = t, redacted = true }
      end
      fields[key] = note
    end
  end
  return { status = "sampled", fields = fields, complete = false }
end

local function record_environment(id, stage, env, generation)
  if generation ~= S.generation or not S.running or S.force_stopped then return end
  local r = S.runtime_capture
  if #r.environments >= 128 then r.environment_dropped = r.environment_dropped + 1; return end
  local ok, sample = measured("environment", environment_sample, env)
  if ok then
    sample.load_id = id; sample.stage = stage
    sample.elapsed_s = os.clock() - S.t0
    for i = #r.environments, 1, -1 do
      local before = r.environments[i]
      if before.load_id == id and before.status == "sampled" then
        sample.changes = {}
        for key, value in pairs(sample.fields) do
          local old = before.fields[key]
          if not old or old.type ~= value.type or old.value ~= value.value
            or old.bytes ~= value.bytes or old.fingerprint ~= value.fingerprint then
            sample.changes[#sample.changes + 1] = { key = key, kind = old and "changed" or "newly_observed" }
          end
        end
        if sample.complete then
          for key in pairs(before.fields) do
            if not sample.fields[key] then sample.changes[#sample.changes + 1] = { key = key, kind = "removed" } end
          end
        end
        break
      end
    end
    r.environments[#r.environments + 1] = sample
  end
end

local function extract_constants(fn, id)
  local generation = S.generation
  local result = { load_id = id, status = "unsupported", entries = {}, inspected_indices = 0 }
  local one, all = runtime_api("getconstant"), runtime_api("getconstants")
  if one then
    result.api = "getconstant"; result.status = "partial"
    local slice = os.clock()
    for i = 1, 128 do
      if S.force_stopped or generation ~= S.generation then result.status = "cancelled"; break end
      local ok, value = measured("runtime_constants", one, fn, i)
      if not ok then result.ended_by_error = true; break end
      result.inspected_indices = i
      if value ~= nil then result.entries[#result.entries + 1] = { index = i, value = snapshot_value(value) } end
      if i % 16 == 0 or os.clock() - slice >= 0.0015 then task.wait(); slice = os.clock() end
    end
  elseif all then
    result.api = "getconstants"
    local ok, values = measured("runtime_constants", all, fn)
    if ok and type(values) == "table" then
      result.status = "partial"
      local key
      local slice = os.clock()
      for i = 1, 128 do
        if S.force_stopped or generation ~= S.generation then result.status = "cancelled"; break end
        local value
        key, value = next(values, key)
        if key == nil then result.status = "read"; break end
        result.inspected_indices = result.inspected_indices + 1
        if type(key) == "number" then
          result.entries[#result.entries + 1] = { index = key, value = snapshot_value(value) }
        end
        if i % 16 == 0 or os.clock() - slice >= 0.0015 then task.wait(); slice = os.clock() end
      end
    else result.status = "api_failed" end
  end
  result.count = #result.entries
  return result
end

-- ── VM instruction extraction (BIG plan §5/§7) ──────────────────────
-- A VM obfuscator compiles the real Lua into a custom instruction table plus a
-- constant pool, then ships an interpreter that walks that table.  After the
-- loader decodes its packed blob, the decoded program lives in the compiled
-- function's UPVALUES or CONSTANTS (custom-table VMs) or in its nested
-- PROTOTYPES (Luau-style VMs).  classify_instruction_table decides whether a
-- Lua table is that instruction stream and samples the real opcodes; extract_vm
-- walks the captured function's upvalues/constants/protos to find it.  All
-- bounded and read-only -- no target code runs.

-- Is this table a VM instruction stream?  Two shapes:
--   opcode_stream : a long array mostly of small integers (opcodes/operands)
--   record_stream : a long array of {op, a, b, c, …} instruction records
local function classify_instruction_table(t)
  local n, nums, small_ints, signed, records, rec_numeric = 0, 0, 0, 0, 0, 0
  local i = 1
  while t[i] ~= nil and n < MAX_VM_TABLE_SCAN do
    local v = t[i]
    n = n + 1
    local tv = type(v)
    if tv == "number" then
      nums = nums + 1
      if v == math.floor(v) then
        if v >= 0 and v <= 255 then small_ints = small_ints + 1 end
        if v < 0 then signed = signed + 1 end
      end
    elseif tv == "table" then
      if type(v[1]) == "number" then records = records + 1; rec_numeric = rec_numeric + 1 end
    end
    i = i + 1
    if (n % 256) == 0 then analysis_checkpoint(1) end
  end
  if n < 16 then return nil end
  local numeric_ratio = nums / n
  local record_ratio = records / n
  local small_ratio = nums > 0 and small_ints / nums or 0
  local is_opcode = numeric_ratio >= 0.7 and small_ratio >= 0.5
  local is_record = record_ratio >= 0.6
  if not (is_opcode or is_record) then return nil end
  -- Copy a bounded sample of the actual instructions and a leading-opcode
  -- histogram (opcode density is the cheap VM signal).
  local lim = math.min(n, MAX_VM_INSTR_SAMPLE)
  local sample, hist = {}, {}
  for k = 1, lim do
    local v = t[k]
    local op
    if type(v) == "number" then
      sample[k] = v; op = v
    elseif type(v) == "table" then
      local rec = {}
      for j = 1, math.min(6, #v) do rec[j] = (type(v[j]) == "number") and v[j] or nil end
      sample[k] = rec; op = rec[1]
    end
    if type(op) == "number" then hist[op] = (hist[op] or 0) + 1 end
    if (k % 128) == 0 then analysis_checkpoint(1) end
  end
  local distinct_ops = 0
  for _ in pairs(hist) do distinct_ops = distinct_ops + 1 end
  return {
    length = n, sampled = lim,
    shape = is_record and "record_stream" or "opcode_stream",
    numeric_ratio = numeric_ratio, small_int_ratio = small_ratio,
    record_ratio = record_ratio, signed_values = signed, distinct_opcodes = distinct_ops,
    sample = sample, opcode_histogram = hist,
    confidence = math.min(0.95, (is_record and record_ratio or (numeric_ratio + small_ratio) / 2)),
  }
end

-- Walk a captured function's upvalues, constants and nested prototypes to
-- recover the VM program.  Capability-gated (needs getupvalues/getprotos or
-- their singular forms); no-op and status="unsupported" otherwise.
local function extract_vm(fn, id)
  local generation = S.generation
  local result = { load_id = id, status = "ok", proto_count = 0, protos = {},
    instruction_streams = {}, strings = {} }
  local string_seen = {}
  local getupvals, getupval = runtime_api("getupvalues"), runtime_api("getupvalue")
  local getprotos, getproto = runtime_api("getprotos"), runtime_api("getproto")
  local getconsts = runtime_api("getconstants")
  if not (getupvals or getupval or getprotos or getproto) then
    result.status = "unsupported"; return result
  end
  local vm_deadline = os.clock() + MAX_VM_WALL_SECS
  local visited = 0
  local function read_upvalues(f)
    local ups = {}
    if getupvals then
      local ok, list = pcall(getupvals, f)
      if ok and type(list) == "table" then
        local k, v
        for _ = 1, MAX_VM_UPVALS do
          k, v = next(list, k); if k == nil then break end
          ups[#ups + 1] = v
        end
      end
    elseif getupval then
      for i = 1, MAX_VM_UPVALS do
        local ok, v = pcall(getupval, f, i)
        if not ok or v == nil then break end
        ups[#ups + 1] = v
      end
    end
    return ups
  end
  local function read_constants(f)
    local arr = {}
    if getconsts then
      local ok, list = pcall(getconsts, f)
      if ok and type(list) == "table" then
        for k, v in pairs(list) do if type(k) == "number" then arr[#arr + 1] = v end end
      end
    end
    return arr
  end
  local function scan_for_streams(values, where, proto_id)
    for _, v in ipairs(values) do
      if type(v) == "table" and #result.instruction_streams < MAX_VM_STREAMS then
        local ok, info = pcall(classify_instruction_table, v)
        if ok and info then
          info.source = where; info.proto_id = proto_id
          result.instruction_streams[#result.instruction_streams + 1] = info
        end
      end
      analysis_checkpoint(1)
    end
  end
  -- Harvest string constants/upvalues: the VM's decoded constant pool is where
  -- the real URLs, keys, function names and messages live.  Deduped and capped;
  -- decoding/classification happens at finalize (layered_decode is defined
  -- later), so here we only collect the raw bytes.
  local function collect_strings(values, where, proto_id)
    for _, v in ipairs(values) do
      if type(v) == "string" and #v > 0 and #result.strings < MAX_VM_STRINGS then
        local s = #v > MAX_VM_STRING_BYTES and v:sub(1, MAX_VM_STRING_BYTES) or v
        if not string_seen[s] then
          string_seen[s] = true
          result.strings[#result.strings + 1] = { value = s, bytes = #v,
            source = where, proto_id = proto_id, truncated = #v > MAX_VM_STRING_BYTES }
        end
      end
      analysis_checkpoint(1)
    end
  end
  local function child_protos(f)
    local kids = {}
    if getprotos then
      local ok, list = pcall(getprotos, f)
      if ok and type(list) == "table" then
        for i = 1, math.min(MAX_VM_PROTOS, #list) do
          if type(list[i]) == "function" then kids[#kids + 1] = list[i] end
        end
      end
    elseif getproto then
      for i = 1, MAX_VM_PROTOS do
        local ok, p = pcall(getproto, f, i, true)
        if not ok or p == nil then break end
        if type(p) == "function" then kids[#kids + 1] = p
        elseif type(p) == "table" then
          for _, pf in ipairs(p) do if type(pf) == "function" then kids[#kids + 1] = pf end end
        end
      end
    end
    return kids
  end
  local function walk(f, depth, proto_id)
    if visited >= MAX_VM_PROTOS or depth > MAX_VM_PROTO_DEPTH then return end
    if S.force_stopped or generation ~= S.generation then return end
    if os.clock() > vm_deadline then result.status = "time_budget"; return end
    visited = visited + 1; result.proto_count = visited
    local ups = read_upvalues(f)
    local consts = read_constants(f)
    scan_for_streams(ups, "upvalue", proto_id)
    scan_for_streams(consts, "constant", proto_id)
    collect_strings(consts, "constant", proto_id)
    collect_strings(ups, "upvalue", proto_id)
    result.protos[#result.protos + 1] = { id = proto_id, depth = depth,
      upvalue_count = #ups, constant_count = #consts }
    for idx, child in ipairs(child_protos(f)) do
      walk(child, depth + 1, proto_id .. "." .. idx)
      if task and task.wait then task.wait() end
    end
  end
  local ok = pcall(walk, fn, 0, "0")
  if not ok then result.status = "error" end
  result.instruction_stream_count = #result.instruction_streams
  result.string_count = #result.strings
  return result
end

local function runtime_function(fn, record, generation)
  local env, env_checked
  return function(...)
    if generation ~= S.generation or not S.running or S.force_stopped then return fn(...) end
    if not env_checked then
      env_checked = true
      if type(getfenv) == "function" then
        local ok, value = pcall(getfenv, fn)
        if ok and type(value) == "table" then
          env = value
          if S.runtime_envs and #S.runtime_envs < 8 then
            S.runtime_envs[#S.runtime_envs + 1] = { id = record.id, env = env }
          end
        end
      end
    end
    local thread = current_thread()
    local previous = target_threads[thread]
    target_threads[thread] = { id = record.id, generation = generation }
    target_depth = target_depth + 1
    record_environment(record.id, "before", env, generation)
    record.execution = "running"
    local result = table.pack(pcall(fn, ...))
    target_depth = target_depth - 1
    target_threads[thread] = previous
    if generation == S.generation then
      record.execution = result[1] and "returned" or "error"
      record_environment(record.id, "after", env, generation)
      if result[1] then
        local returns = {}
        for i = 2, math.min(result.n, 9) do returns[i - 1] = result[i] end
        record.returns = snapshot_value(returns)
        record.return_count = result.n - 1
      end
    end
    if not result[1] then error(result[2], 0) end
    return table.unpack(result, 2, result.n)
  end
end

-- Carry the target's attribution onto another coroutine.  Deliberately
-- lighter than runtime_function: no environment sampling and no return
-- snapshot, because this runs for every thread the target starts and all the
-- sinks need is to know whose work it is.
local function adopt_scope(fn, id, generation)
  return function(...)
    if generation ~= S.generation or not S.running or S.force_stopped then return fn(...) end
    local thread = current_thread()
    local previous = target_threads[thread]
    target_threads[thread] = { id = id, generation = generation }
    target_depth = target_depth + 1
    local result = table.pack(pcall(fn, ...))
    target_depth = target_depth - 1
    target_threads[thread] = previous
    if not result[1] then error(result[2], 0) end
    return table.unpack(result, 2, result.n)
  end
end

-- Bind now if the caller is the target, otherwise hand the function back
-- untouched.  Every scheduler wrapper funnels through here, so a task the
-- game schedules costs one in_scope() and nothing else.
local function bind_if_target(fn)
  if type(fn) ~= "function" or not in_scope() then return fn end
  local context = target_threads[current_thread()]
  S.counters.bound = S.counters.bound + 1
  return adopt_scope(fn, context and context.id or 1, S.generation)
end

local function note_unknown(kind, detail)
  if not S.path then return end
  ensure_dir(S.path)
  local line = string.format("[%s] kind=%s  %s\n", now(), tostring(kind or "unknown"),
    tostring(detail or ""):sub(1, 4000))
  append(S.path .. "/unknown.txt", line)
end

local function write_stage_file(item, reason)
  if not S.path or not item or type(item.source) ~= "string" then return end
  ensure_dir(S.path .. "/extracts")
  local rel = "extracts/stage_" .. string.format("%02d", item.id) .. ".txt"
  local header = table.concat({
    "-- SENTINEL STAGE CAPTURE --",
    "stage_id=" .. tostring(item.id),
    "parent=" .. tostring(item.parent or 0),
    "name=" .. tostring(item.name or ""),
    "bytes=" .. tostring(item.bytes or 0),
    "saved_bytes=" .. tostring(item.saved_bytes or 0),
    "truncated=" .. tostring(item.truncated),
    "hop_reason=" .. tostring(reason or "capture"),
    "elapsed_s=" .. string.format("%.3f", os.clock() - (S.t0 or os.clock())),
    "",
  }, "\n")
  write(S.path .. "/" .. rel, header .. item.source)
  item.path = rel
  item.written_live = true
  -- Stage hop: parent points at a prior load stage
  if item.parent and item.parent > 0 then
    append(S.path .. "/extracts/stage_hops.txt", string.format(
      "[%s] hop stage_%02d <- stage_%02d | %s | %dB | %s\n",
      now(), item.id, item.parent, tostring(item.name or ""), item.bytes or 0, tostring(reason or "nested_load")))
    record_runtime("stage_hop", {
      load_id = item.id, parent_id = item.parent, bytes = item.bytes,
      name = item.name, reason = reason or "nested_load",
    })
  end
  return rel
end

local function reserve_runtime_load(source, name, parent)
  local r = S.runtime_capture
  if #r.loads >= MAX_LOADS or r.bytes >= MAX_CAPTURE_BYTES then
    r.dropped = r.dropped + 1
    note_unknown("stage_dropped", string.format("name=%s parent=%s bytes=%s (cap hit)",
      tostring(name), tostring(parent), tostring(type(source) == "string" and #source or 0)))
    return
  end
  local remaining = MAX_CAPTURE_BYTES - r.bytes
  local body = source:sub(1, math.min(MAX_SOURCE_BYTES, remaining))
  local item = { id = #r.loads + 1, parent = parent or 0, name = name,
    bytes = #source, saved_bytes = #body, truncated = #body < #source,
    execution = "not_executed", source = body }
  r.loads[#r.loads + 1] = item; r.bytes = r.bytes + #body; r.selected = true
  S.target_captured = true; S.counters.load = #r.loads
  S.capture_started = S.capture_started or os.clock()
  S.last_activity = os.clock()
  -- Live stage file as soon as a hop/stage is observed
  local hop = (item.parent and item.parent > 0) and "stage_hop" or "initial_stage"
  pcall(write_stage_file, item, hop)
  return item
end

local function save_runtime_capture()
  local r = S.runtime_capture
  if not r or not S.path then return end
  -- Nothing captured on the target's threads → don't leave empty runtime files.
  if #r.loads == 0 and #r.constants == 0 and #r.environments == 0 then return end
  local manifest = {}
  for _, item in ipairs(r.loads) do
    local path = "extracts/stage_" .. string.format("%02d", item.id) .. ".txt"
    local saved, err = write(S.path .. "/" .. path, item.source)
    manifest[#manifest + 1] = { id = item.id, parent = item.parent, name = item.name,
      bytes = item.bytes, saved_bytes = item.saved_bytes, truncated = item.truncated,
      path = path, saved = saved, save_error = err, execution = item.execution,
      processing = item.processing,
      return_count = item.return_count, returns = item.returns }
  end
  if #manifest > 0 then
    write(S.path .. "/runtime_loads.txt", safe_json({ loads = manifest, dropped = r.dropped, scope = r.scope,
      limitations = r.limitations }))
  end
  if #r.constants > 0 then
    write(S.path .. "/runtime_constants.txt", safe_json(r.constants))
  end
  if #r.environments > 0 then
    write(S.path .. "/environment_changes.txt", safe_json({ observations = r.environments,
      dropped = r.environment_dropped, attribution = "sampled_function_environment_may_be_shared" }))
  end
  write(S.path .. "/env_log.txt", safe_json({ startup = S.env, observations = r.environments }))
end

-- Read-only tool diagnostics: API presence is not a runtime compatibility test.
local function feature_state(enabled)
  if LITE_MODE then return "disabled_lite" end
  return enabled and "enabled" or "disabled_profile"
end

-- Six-state observation vocabulary (BIG plan §12).  Whether a sink was armed,
-- whether its hook actually installed, and how many events it saw are three
-- separate facts; the old reports collapsed them so that "0" could mean "no
-- activity" or "never instrumented".  These never conflate again.
local OBS_NOT_MONITORED = "NOT_MONITORED"          -- sink not armed (lite, or profile off)
local OBS_NOT_OBSERVED  = "NOT_OBSERVED"           -- armed, but no target has run through it yet
local OBS_ZERO_EVENTS   = "MONITORED_ZERO_EVENTS"  -- armed, target ran, genuinely nothing
local OBS_OBSERVED      = "OBSERVED"
local OBS_PARTIAL       = "PARTIALLY_OBSERVED"      -- observed, but a cap/limit truncated it
local OBS_INCONCLUSIVE  = "INCONCLUSIVE"            -- capability absent / hook install failed

-- True if at least one of the named hook labels installed via a real
-- mechanism (anything but "absent"/"unavailable"/never-set).
local function hook_installed(...)
  for i = 1, select("#", ...) do
    local st = S.hooks[select(i, ...)]
    if st and st ~= "absent" and st ~= "unavailable" then return true, st end
  end
  return false, nil
end

-- Map (armed?, events, capped?, hook labels…) to one observation state.
local function observe(enabled, events, capped, ...)
  if LITE_MODE or not enabled then return { state = OBS_NOT_MONITORED } end
  local installed, mechanism = hook_installed(...)
  if select("#", ...) > 0 and not installed then
    return { state = OBS_INCONCLUSIVE, reason = "instrumentation_unavailable" }
  end
  events = tonumber(events)
  if events and events > 0 then
    return { state = capped and OBS_PARTIAL or OBS_OBSERVED, events = events, mechanism = mechanism }
  end
  if S.target_captured then return { state = OBS_ZERO_EVENTS, events = 0, mechanism = mechanism } end
  return { state = OBS_NOT_OBSERVED, events = 0, mechanism = mechanism }
end

-- Per-sink observation report in the six-state vocabulary.
local function observation_report()
  local c = S.counters
  return {
    schema = "sentinel.observations.v1",
    vocabulary = { OBS_NOT_MONITORED, OBS_NOT_OBSERVED, OBS_ZERO_EVENTS,
      OBS_OBSERVED, OBS_PARTIAL, OBS_INCONCLUSIVE },
    scope = SCOPED_CAPTURE and "target_threads_only" or "global",
    loads   = observe(true,             c.load,   (S.runtime_capture and #S.runtime_capture.loads or 0) >= MAX_LOADS, "loadstring", "load"),
    prints  = observe(CAPTURE.prints,   c.print,  false,                                     "print", "warn"),
    http    = observe(CAPTURE.http,     c.http,   c.http >= MAX_HTTP_EVENTS,                  "HttpGet", "request"),
    remotes = observe(CAPTURE.remotes,  c.remote, c.remote >= MAX_REMOTE_EVENTS,             "namecall"),
    services= observe(CAPTURE.services, c.svc,    false,                                     "GetService"),
    global_writes = (function()
      -- Global writes are streamed to extracts, not tallied in a counter, so
      -- report the install state honestly rather than inventing an event count.
      if LITE_MODE or not CAPTURE.globals then return { state = OBS_NOT_MONITORED } end
      local installed, mechanism = hook_installed("global_writes")
      if not installed then return { state = OBS_INCONCLUSIVE, reason = "instrumentation_unavailable" } end
      return { state = OBS_ZERO_EVENTS, mechanism = mechanism,
        note = "global writes are captured to extracts/, not counted here" }
    end)(),
    task_binding       = observe(CAPTURE.tasks,       c.bound, false),
    connection_binding = observe(CAPTURE.connections, c.bound, false),
  }
end

-- Capability matrix (BIG plan §3): machine-readable yes/no for the runtime
-- features Sentinel can lean on.  API presence only -- deliberately not a
-- runtime compatibility guarantee; the note says so.
local function capability_present(name)
  if type(debug) == "table" and type(rawget(debug, name)) == "function" then return true end
  for _, env in ipairs(cached_envs) do
    if type(rawget(env, name)) == "function" then return true end
  end
  return false
end
-- Resolve a namespaced UNC function (e.g. crypt.hash, cache.replace) from any
-- reachable environment; returns the function or nil.
local function lib_resolve(libname, fnname)
  for _, env in ipairs(cached_envs) do
    local lib = rawget(env, libname)
    if type(lib) == "table" then
      local fn = rawget(lib, fnname)
      if type(fn) == "function" then return fn end
    end
  end
  return nil
end
local function lib_present(libname, fnname) return lib_resolve(libname, fnname) ~= nil end
local function capability_matrix()
  local function yn(v) return v and "yes" or "no" end
  local debug_info    = type(debug) == "table" and type(rawget(debug, "info")) == "function"
  local debug_getinfo = type(debug) == "table" and type(rawget(debug, "getinfo")) == "function"
  local debug_getstack= type(debug) == "table" and type(rawget(debug, "getstack")) == "function"
  return {
    schema = "sentinel.capability-matrix.v1", version = VERSION,
    note = "API presence only; not a runtime compatibility guarantee.",
    least_invasive_first = { "static_analyze", "constant_inspection",
      "environment_sampling", "targeted_function_hook", "metamethod_hook" },
    capabilities = {
      ["debug.info"]           = yn(debug_info or debug_getinfo),
      ["buffer"]               = yn(type(buffer) == "table" and type(rawget(buffer, "create")) == "function"),
      ["string.unpack"]        = yn(type(string) == "table" and type(rawget(string, "unpack")) == "function"),
      ["bit32"]                = yn(type(bit32) == "table" and type(bit32.bxor) == "function"),
      ["function_hook"]        = yn(capability_present("hookfunction")),
      ["metamethod_hook"]      = yn(capability_present("hookmetamethod")),
      ["constant_inspection"]  = yn(capability_present("getconstant") or capability_present("getconstants")),
      ["upvalue_inspection"]   = yn(capability_present("getupvalue") or capability_present("getupvalues")),
      ["prototype_inspection"] = yn(capability_present("getproto") or capability_present("getprotos")),
      ["stack_inspection"]     = yn(debug_getinfo or debug_getstack),
      ["http_observation"]     = yn(capability_present("request") or capability_present("http_request")),
      ["task_coroutine_tracking"] = yn(type(task) == "table" or type(coroutine) == "table"),
      -- Compiled-bytecode dump of a function (Luau). Newer executors expose
      -- getfunctionbytecode / dumpbytecode alongside the older dumpstring;
      -- getscriptbytecode dumps a game Script instead of a function.
      ["function_bytecode_dump"] = yn(capability_present("getfunctionbytecode")
        or capability_present("dumpbytecode") or capability_present("dumpstring")),
      ["script_bytecode_dump"]   = yn(capability_present("getscriptbytecode")),
      -- zstd (de)compression, for zstd-framed payloads. Sentinel only DETECTS
      -- container framing live; decompression of untrusted blobs is Requiem's
      -- isolated-worker job (BIG plan §9/§16), never run in-game.
      ["zstd"]                   = yn(capability_present("zstddecompress") or capability_present("zstdcompress")),
      -- Closure typing: is a candidate decoder written in Luau (introspectable),
      -- native C (opaque), or the executor's own? Decides analysis strategy.
      ["closure_type"]           = yn(capability_present("islclosure") and capability_present("iscclosure")),
      ["executor_closure_check"] = yn(capability_present("isexecutorclosure")),
      ["calling_script"]         = yn(capability_present("getcallingscript")),
      ["script_closure"]         = yn(capability_present("getscriptclosure")),
      -- Native crypto/encoding: stronger provenance hashes and native decoders
      -- than Sentinel's built-in djb2 / base64 / hand-rolled LZ4.
      ["native_hash"]            = yn(lib_present("crypt", "hash")),
      ["native_base64"]          = yn(lib_present("crypt", "base64decode") or capability_present("base64decode")),
      ["native_aes"]             = yn(lib_present("crypt", "decrypt")),
      ["native_lz4"]             = yn(capability_present("lz4decompress")),
      -- Script/module surface for targeted (never continuous) enumeration.
      ["script_hash"]            = yn(capability_present("getscripthash")),
      ["loaded_modules"]         = yn(capability_present("getloadedmodules")),
      ["script_env"]             = yn(capability_present("getsenv")),
      ["thread_identity"]        = yn(capability_present("getthreadidentity")),
      -- Signal/callback observation: reach connections/callbacks that were
      -- bound before Sentinel attached (a current capture blind spot).
      ["signal_connections"]     = yn(capability_present("getconnections")),
      ["callback_value"]         = yn(capability_present("getcallbackvalue")),
      ["namecall_method"]        = yn(capability_present("getnamecallmethod")),
    },
  }
end

local function tool_diagnostics()
  local issues = {}
  if not has_fs() then issues[#issues + 1] = "File APIs unavailable; reports remain in memory." end
  if S.fs_error then issues[#issues + 1] = tostring(S.fs_error) end
  if S.write_limit_hit then issues[#issues + 1] = "Write budget reached; output is partial." end
  if S.hook_mode == "binding" then
    issues[#issues + 1] = "Environment binding may miss separate executor scripts; use Analyze(source) for static input."
  end
  if not S.target_captured then issues[#issues + 1] = "No target source accepted yet." end
  if not LITE_MODE and SCOPED_CAPTURE then
    if CAPTURE.tasks and CAPTURE.connections then
      issues[#issues + 1] = "Runtime sinks record the target's own work only. Threads it starts and callbacks it connects are bound automatically; a callback handed to an executor or engine API Sentinel does not wrap still needs Sentinel.Bind."
    elseif CAPTURE.tasks then
      issues[#issues + 1] = "Runtime sinks record the target's own threads only. Threads it starts are bound automatically; a callback the engine invokes later (a signal connection) still needs Sentinel.Bind."
    else
      issues[#issues + 1] = "Runtime sinks record the target's own threads only; wrap task callbacks with Sentinel.Bind to widen the scope."
    end
  end
  if not LITE_MODE and S.hooks.HttpGet == "unavailable" then
    issues[#issues + 1] = "game:HttpGet could not be intercepted; HTTP capture is limited to the executor request family."
  end
  return {
    schema = "sentinel.tool-diagnostics.v1", version = VERSION,
    mode = LITE_MODE and "lite" or "full",
    capture_scope = SCOPED_CAPTURE and "target_threads_only" or "global",
    hooks_installed = S.hooks,
    executor = S.env and S.env.executor or "unknown",
    compatibility = "runtime_unverified", hook_mode = S.hook_mode,
    restore_status = S.restore_status, timings = S.timings,
    queue = { pending = #work_queue, dropped = S.queue_dropped, capacity = 8 },
    tools = {
      constant_inventory = "bounded_static_literals",
      environment_log = "api_types_and_redacted_key_metadata",
      endpoint_inventory = "static_references_not_observed_requests",
      source_structure = "heuristic_counts",
      surface_reconstruction = "partial_not_original_source_recovery",
      payload_translation = "hexblob_vm_container_unwrap_plus_constant_pool_recovery",
      runtime_prints = feature_state(CAPTURE.prints),
      runtime_http = feature_state(CAPTURE.http),
      runtime_remotes = feature_state(CAPTURE.remotes),
      runtime_services = feature_state(CAPTURE.services),
      runtime_global_writes = feature_state(CAPTURE.globals),
      runtime_task_binding = feature_state(CAPTURE.tasks),
      runtime_connection_binding = feature_state(CAPTURE.connections),
      live_status_snapshots = LITE_MODE and "disabled" or "enabled",
      runtime_constants = (runtime_api("getconstant") or runtime_api("getconstants"))
        and "compiled_function_read" or "unsupported",
      environment_changes = "before_after_and_periodic_bounded_samples",
      explicit_snapshots = "bounded_values_with_cycle_references",
    },
    capture = { accepted = S.target_captured, largest_captured_stage_bytes = #S.largest_captured_stage,
      validation = S.analysis_status.runtime_validation },
    observations = observation_report(),
    capability_matrix = capability_matrix(),
    budgets = { written_bytes = S.write_bytes, write_limit = MAX_LOCAL_WRITE_BYTES,
      write_limit_hit = S.write_limit_hit, scan_bytes = effective_static_scan_bytes() },
    issues = issues,
  }
end

-- ── detect ──────────────────────────────────────────────────────────
local function detect_obfuscator(src)
  src = tostring(src or "")
  local families, features, scores, family_seen, feature_seen = {}, {}, {}, {}, {}
  local function hit(fam, points, feat, detail)
    scores[fam] = math.min(100, (scores[fam] or 0) + points)
    if not family_seen[fam] then family_seen[fam] = true; table.insert(families, fam) end
    if feat and not feature_seen[feat] then feature_seen[feat] = true; table.insert(features, feat) end
    add_evidence("family." .. tostring(feat or fam), points >= 70 and "high" or "medium",
      "StaticAnalysis", detail or (fam .. " indicator: " .. tostring(feat)), math.min(0.99, points / 100))
  end
  if src:find("MoonSec V3", 1, true) or src:find("protected with MoonSec", 1, true) then hit("MoonSec V3", 98, "moonsec_banner") end
  if src:find("__oo_payload", 1, true) then hit("00fuscator", 92, "oo_payload") end
  if src:find("__oo_", 1, true) then hit("00fuscator", 18, "oo_prefix") end
  if src:find("local%-only integrity wrapper") then hit("00fuscator", 70, "oo_integrity") end
  if src:find("StackFuscate", 1, true) or src:find("stackfuscate", 1, true) then hit("StackFuscate/FaulMor", 90, "sf_banner") end
  if src:find("FaulMor", 1, true) or src:find("faulmor", 1, true) then hit("StackFuscate/FaulMor", 88, "faulmor") end
  if src:find("MoonVeil", 1, true) or src:find("moonveil", 1, true) then hit("MoonVeil", 85, "moonveil") end
  if src:find("LUREX", 1, true) or src:find("Lurex", 1, true) or src:find("lurex", 1, true) then hit("Lurex", 85, "lurex") end
  if src:find("Luraph", 1, true) or src:find("LURAPH", 1, true) then hit("Luraph", 82, "luraph") end
  if src:find("IronBrew", 1, true) or src:find("IB_DECODE", 1, true) then hit("IronBrew", 82, "ironbrew") end
  if src:find("Prometheus", 1, true) then hit("Prometheus", 82, "prometheus") end
  if src:find("Goofy", 1, true) or src:find("goofyscator", 1, true) then hit("Goofy", 72, "goofy") end
  if src:find("POLSEC", 1, true) or src:find("PolSec", 1, true) then hit("PolSec/MoonVeil", 78, "polsec") end
  if src:find("bit32%.bxor%(bit32%.bxor") and src:find("while%s+true") then
    hit("VM-Packed", 35, "nested_xor_dispatch", "nested XOR plus persistent dispatcher loop")
  end
  if src:find("getfenv") and src:find("setfenv") and src:find("string%.byte") and #src > 50000 then
    hit("VM-Packed", 30, "classic_vm", "environment access plus byte-oriented decoder in a large source")
  end
  -- Hex-blob VM container (LZ4 + repeating-key XOR). Name-independent
  -- structural signals: a string bootstrap off getmetatable(""), a very long
  -- hex string assembled by `var = var .. "…"` accumulation, and a final
  -- `return(F(blob, k1, k2))(...)` handing the decoded blob plus two
  -- blob-derived keys to an interpreter.
  do
    local boot = src:find('getmetatable%s*%(%s*""%s*%)') ~= nil
    -- Count long hex string literals regardless of how the build glues them
    -- together (`..` accumulation, table-insert + concat, etc.), so a variant
    -- assembly style still fingerprints.
    local hexrun = 0
    for lit in src:gmatch('"([0-9A-Fa-f][0-9A-Fa-f]+)"') do
      if #lit >= 16 and (#lit % 2) == 0 then
        hexrun = hexrun + 1
        if hexrun > 64 then break end
      end
    end
    local tail_call = src:find("return%s*%(%s*[%w_]+%s*%(%s*[%w_]+%s*,%s*[%w_]+%s*,%s*[%w_]+%s*%)%s*%)%s*%(") ~= nil
    if hexrun > 32 then
      local pts = 52
      if boot then pts = pts + 20 end
      if tail_call then pts = pts + 18 end
      hit("HexBlob-VM", math.min(96, pts), "hex_blob_lz4_xor",
        "long concatenated hex blob (" .. tostring(hexrun) .. "+ chunks)" ..
        (boot and " + getmetatable-string bootstrap" or "") ..
        (tail_call and " + blob-derived-key tail call" or ""))
    end
  end
  -- LuaSnapper is a structural subfamily of the HexBlob VM loader.  Its
  -- identifiers, constant tags and keys are randomized, so attribution is
  -- based on independent implementation features rather than a banner or a
  -- particular local name.  Keep HexBlob-VM in `families`: it describes the
  -- transport/capability and is what routes the sample to the static decoder.
  do
    local escaped = 0
    for _ in src:gmatch("\\%d%d%d") do
      escaped = escaped + 1
      if escaped > 240 then break end
    end
    local bootstrap = src:find('getmetatable%s*%(%s*""%s*%)') ~= nil
      and src:find("for%s+[%w_]+%s*=%s*0%s*,%s*7%s+do") ~= nil
      and src:find(":byte%s*%(") ~= nil
      and escaped > 80
    local u32_reader = src:find(":byte%s*%([^)]*,%s*[^)]*%+%s*3%s*%)") ~= nil
      and src:find("%*%s*256") ~= nil
      and src:find("0[xX]10000") ~= nil
    local f64_reader = src:find(":byte%s*%([^)]*,%s*[^)]*%+%s*7%s*%)") ~= nil
      and src:find("0[xX]7[fF][fF]") ~= nil
      and src:find("2%^%s*48") ~= nil
      and src:find("2%^%s*40") ~= nil
    local vm_dispatch = src:find("while%s+true%s+do") ~= nil
      and src:find("0[xX]3[fF]") ~= nil
      and src:find("0[xX]1[fF][fF]") ~= nil
      and src:find("262143", 1, true) ~= nil
    local key_derivation = src:find(":sub%s*%(") ~= nil
      and (src:find("%*%s*0[xX]%x+%s*%+") ~= nil
        or src:find("%*%s*%d+%s*%+") ~= nil)
      and src:find("%%%s*4294967296") ~= nil
    local anti_analysis = src:find("IsStudio%s*%(%s*%)") ~= nil
      and src:find("while%s+true%s+do") ~= nil
    if bootstrap and u32_reader and f64_reader and vm_dispatch then
      local pts = 24 + 29 + 25
      if key_derivation then pts = pts + 12 end
      if anti_analysis then pts = pts + 8 end
      hit("LuaSnapper", math.min(98, pts), "luasnapper_structural",
        string.format("structural loader match: bootstrap/u32/f64/vm%s%s; %d+ decimal escapes",
          key_derivation and "/derived-keys" or "",
          anti_analysis and "/anti-analysis" or "", escaped))
    end
  end
  local esc_n = 0
  for _ in src:gmatch("\\%d%d%d") do esc_n = esc_n + 1 if esc_n > 200 then break end end
  if esc_n > 80 then hit("VM-Packed", 18, "dense_decimal_escapes", tostring(esc_n) .. "+ decimal escapes") end
  local bxor_n = 0
  for _ in src:gmatch("bxor") do bxor_n = bxor_n + 1 if bxor_n > 80 then break end end
  if bxor_n > 20 then hit("VM-Packed", 16, "xor_heavy", tostring(bxor_n) .. "+ XOR references") end
  local best_family, best_score = "Unknown", 0
  for fam, score in pairs(scores) do
    if score > best_score then best_family, best_score = fam, score end
  end
  return {
    family = best_family,
    confidence = math.min(0.99, best_score / 100),
    features = features,
    families = families,
    family_scores = scores,
    container_family = family_seen["HexBlob-VM"] and "HexBlob-VM" or nil,
  }
end

local function is_printable(s)
  if not s or #s < 2 then return false end
  local ok = 0
  for i = 1, #s do
    local c = string.byte(s, i)
    if c == 9 or c == 10 or c == 13 or (c >= 32 and c < 127) then ok = ok + 1 end
  end
  return (ok / #s) > 0.85
end

-- tiny base64 (alphabet)
local B64 = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
local B64MAP = {}
for i = 1, #B64 do B64MAP[B64:sub(i, i)] = i - 1 end
B64MAP["="] = 0
local function b64decode(data)
  data = tostring(data or ""):gsub("%s", "")
  if #data < 16 or #data % 4 ~= 0 then return nil end
  local out = {}
  for i = 1, #data, 4 do
    local a = B64MAP[data:sub(i, i)]
    local b = B64MAP[data:sub(i + 1, i + 1)]
    local c = B64MAP[data:sub(i + 2, i + 2)]
    local d = B64MAP[data:sub(i + 3, i + 3)]
    if not (a and b and c and d) then return nil end
    local n = a * 262144 + b * 4096 + c * 64 + d
    table.insert(out, string.char(bit32.band(math.floor(n / 65536), 255)))
    if data:sub(i + 2, i + 2) ~= "=" then table.insert(out, string.char(bit32.band(math.floor(n / 256), 255))) end
    if data:sub(i + 3, i + 3) ~= "=" then table.insert(out, string.char(bit32.band(n, 255))) end
  end
  local s = table.concat(out)
  if is_printable(s) then return s end
  return nil
end

-- ── static analysis (original techniques) ───────────────────────────
local function unfold_concats(src)
  local found = {}
  -- "a".."b" / 'a'..'b' chains of 2–8 literals
  for a, b in src:gmatch([=["([^"]+)"%s*%.%.%s*"([^"]+)"]=]) do
    if #a + #b < 400 then table.insert(found, a .. b) end
    if #found >= 40 then break end
  end
  return found
end

local function peel_cff(src)
  -- Control-flow flatten: `if VAR==123 then` / `elseif VAR==`
  local states = {}
  for n in src:gmatch("if%s+[%w_]+%s*==%s*(%d+)%s+then") do
    table.insert(states, tonumber(n))
    if #states >= 80 then break end
  end
  for n in src:gmatch("elseif%s+[%w_]+%s*==%s*(%d+)%s+then") do
    table.insert(states, tonumber(n))
    if #states >= 160 then break end
  end
  return states
end

local function vm_density(src)
  local score = 0
  local function count(pat, w, cap)
    local n = 0
    for _ in src:gmatch(pat) do n = n + 1 if n >= (cap or 50) then break end end
    score = score + n * w
  end
  count("bit32%.bxor", 2, 40)
  count("getfenv", 3, 20)
  count("setfenv", 3, 20)
  count("string%.byte", 1, 30)
  count("string%.char", 1, 30)
  count("loadstring", 4, 20)
  count("HttpGet", 4, 20)
  count("while%s+true", 2, 10)
  return score
end

local function static_analyze(src)
  src = tostring(src or "")
  if #src < 8 then return S.static end
  set_phase("tools", "Bounded static analysis — strings / VM / capabilities…")
  local source_bytes = #src
  local scan_limit = effective_static_scan_bytes()
  local literal_limit = effective_literal_limit()
  if #src > scan_limit then
    src = src:sub(1, scan_limit)
    add_evidence("analysis.truncated", "info", "StaticAnalysis",
      "scanned first " .. tostring(scan_limit) .. " of " .. tostring(source_bytes) .. " bytes", 1)
  end
  local strings, urls, keys = {}, {}, {}
  local constants = { strings = {}, numbers = {}, summary = {} }
  local function classify_literal(literal)
    local lower = literal:lower()
    if lower:find("http://", 1, true) or lower:find("https://", 1, true) then return "endpoint" end
    for _, term in ipairs({ "token", "script_key", "apikey", "api_key", "secret", "password", "cookie", "bearer", "hwid" }) do
      if lower:find(term, 1, true) then return "secret_like" end
    end
    if #literal >= 24 and #literal % 4 == 0 and not literal:find("[^A-Za-z0-9+/=]") then return "encoded" end
    if #literal > 120 then return "long_text" end
    return "text"
  end
  local function inventory_literal(literal, quote_style)
    if #constants.strings >= 96 then return end
    local class = classify_literal(literal)
    local preview = "<redacted>"
    if class == "endpoint" then
      preview = redact_url(literal)
    elseif class == "text" then
      preview = literal:gsub("[%c]", " "):sub(1, 80)
    end
    table.insert(constants.strings, {
      index = #constants.strings + 1,
      class = class,
      quote = quote_style,
      bytes = #literal,
      fingerprint = fingerprint(literal),
      preview = preview,
      value_redacted = class == "secret_like" or class == "encoded" or class == "long_text",
    })
  end
  local indicators = {
    decimal_escapes = 0, hex_escapes = 0, base64_literals = 0,
    base85_like_literals = 0, quoted_literals_scanned = 0,
  }

  local n_char = 0
  for chunk in src:gmatch("string%.char%(([%d%s,]+)%)") do
    analysis_checkpoint(4)
    local bytes = {}
    for n in chunk:gmatch("%d+") do
      local v = tonumber(n)
      if v and v <= 255 then table.insert(bytes, string.char(v)) end
    end
    local s = table.concat(bytes)
    if is_printable(s) and #s >= 3 then
      table.insert(strings, { method = "string.char", plain = s })
      n_char = n_char + 1
      if n_char >= 60 then break end
    end
  end

  -- Lua patterns do not support regex groups or {n} quantifiers. Scan quoted
  -- literals first, then validate candidates with bounded Lua-native tests.
  local function inspect_literal(literal)
    if indicators.quoted_literals_scanned >= literal_limit then return end
    indicators.quoted_literals_scanned = indicators.quoted_literals_scanned + 1
    inventory_literal(literal, "quoted")

    local decimal_count = count_pattern(literal, "\\%d%d?%d?", 512)
    indicators.decimal_escapes = indicators.decimal_escapes + decimal_count
    if decimal_count >= 4 then
      local decoded = literal:gsub("\\(%d%d?%d?)", function(d)
        local n = tonumber(d)
        return n and n <= 255 and string.char(n) or ""
      end)
      if is_printable(decoded) and #decoded >= 3 then
        table.insert(strings, { method = "decimal_escape", plain = decoded:sub(1, 240) })
      end
    end

    local hex_count = count_pattern(literal, "\\x%x%x", 512)
    indicators.hex_escapes = indicators.hex_escapes + hex_count
    if hex_count >= 4 then
      local decoded = literal:gsub("\\x(%x%x)", function(h)
        return string.char(tonumber(h, 16) or 0)
      end)
      if is_printable(decoded) and #decoded >= 3 then
        table.insert(strings, { method = "hex_escape", plain = decoded:sub(1, 240) })
      end
    end

    if #literal >= 24 and #literal <= MAX_DECODE_INPUT
      and #literal % 4 == 0 and not literal:find("[^A-Za-z0-9+/=]") then
      indicators.base64_literals = indicators.base64_literals + 1
      if indicators.base64_literals <= 20 then
        local decoded = b64decode(literal)
        if decoded and #decoded >= 4 then
          table.insert(strings, { method = "base64", plain = decoded:sub(1, 240) })
        end
      end
    elseif #literal >= 40 and #literal <= MAX_DECODE_INPUT
      and is_printable(literal) then
      indicators.base85_like_literals = indicators.base85_like_literals + 1
    end
  end
  for literal in src:gmatch('"([^"\r\n]-)"') do
    inspect_literal(literal)
    if indicators.quoted_literals_scanned >= literal_limit then break end
  end
  if indicators.quoted_literals_scanned < literal_limit then
    for literal in src:gmatch("'([^'\r\n]-)'") do
      inspect_literal(literal)
      if indicators.quoted_literals_scanned >= literal_limit then break end
    end
  end

  local number_seen = {}
  local function inventory_number(token, base)
    if #constants.numbers >= 64 or number_seen[token] then return end
    number_seen[token] = true
    table.insert(constants.numbers, {
      token = token,
      base = base,
      value = base == 16 and tonumber(token:sub(3), 16) or tonumber(token),
    })
  end
  for token in src:gmatch("%f[%w]0[xX]%x+%f[^%w_]") do
    inventory_number(token, 16)
    analysis_checkpoint(1)
    if #constants.numbers >= 64 then break end
  end
  if #constants.numbers < 64 then
    for token in src:gmatch("%f[%d]%d+%.?%d*%f[^%w_]") do
      inventory_number(token, 10)
      analysis_checkpoint(1)
      if #constants.numbers >= 64 then break end
    end
  end

  local url_seen = {}
  for u in src:gmatch("https?://[%w%-%._~:/%?#%[%]@!$&'()*+,;=%]+") do
    u = redact_url(u)
    if #u < 320 and not url_seen[u] then url_seen[u] = true; table.insert(urls, u) end
    if #urls >= 40 then break end
  end
  for u in src:gmatch("HttpGet%(%s*[\"']([^\"']+)[\"']") do
    u = redact_url(u)
    if not url_seen[u] then url_seen[u] = true; table.insert(urls, u) end
    if #urls >= 40 then break end
  end

  for k, v in src:gmatch("([%w_]*[Kk]ey[%w_]*)%s*=%s*[\"']([^\"']+)[\"']") do
    table.insert(keys, { name = k, bytes = #v, fingerprint = fingerprint(v) })
    if #keys >= 30 then break end
  end

  local tables_n = 0
  for tbl in src:gmatch("%{([%d%s,]+)%}") do
    analysis_checkpoint(4)
    local nums = {}
    for n in tbl:gmatch("%d+") do
      analysis_checkpoint(1)
      local v = tonumber(n)
      if v and v <= 255 then table.insert(nums, v) end
    end
    if #nums >= 8 and #nums <= 256 then
      tables_n = tables_n + 1
      for _, xor_k in ipairs({ nums[1], 0x5A, 0xFF, 0x41, 185, 82 }) do
        local bytes = {}
        for _, n in ipairs(nums) do
          table.insert(bytes, string.char(bit32.band(bit32.bxor(n, xor_k), 255)))
        end
        local s = table.concat(bytes)
        if is_printable(s) and #s >= 6 then
          table.insert(strings, { method = "xor_table_" .. tostring(xor_k), plain = s:sub(1, 200) })
          break
        end
      end
    end
    if tables_n >= 30 then break end
  end

  local concats = unfold_concats(src)
  for _, s in ipairs(concats) do
    table.insert(strings, { method = "concat", plain = s })
  end
  local cff = peel_cff(src)
  local density = vm_density(src)

  local capability_specs = {
    { "dynamic_code", "loadstring", "high", 0.95 },
    { "environment_mutation", "setfenv", "high", 0.90 },
    { "gc_enumeration", "getgc", "high", 0.90 },
    { "function_hooking", "hookfunction", "high", 0.95 },
    { "metamethod_hooking", "hookmetamethod", "high", 0.95 },
    { "raw_metatable", "getrawmetatable", "medium", 0.85 },
    { "filesystem_write", "writefile", "medium", 0.90 },
    { "filesystem_read", "readfile", "medium", 0.85 },
    { "http_request", "request", "medium", 0.75 },
    { "http_get", "HttpGet", "medium", 0.85 },
    { "debug_constants", "getconstants", "high", 0.90 },
    { "debug_prototypes", "getprotos", "high", 0.90 },
    { "executor_detection", "identifyexecutor", "medium", 0.80 },
    { "caller_detection", "checkcaller", "medium", 0.80 },
  }
  local capabilities, risk_score = {}, 0
  for _, spec in ipairs(capability_specs) do
    local count = count_pattern(src, spec[2], 50)
    if count > 0 then
      table.insert(capabilities, { id = spec[1], count = count, severity = spec[3] })
      risk_score = risk_score + (spec[3] == "high" and 8 or 4)
      add_evidence("capability." .. spec[1], spec[3], "StaticAnalysis",
        tostring(count) .. " reference(s) to " .. spec[2], spec[4])
    end
  end

  local lower = src:lower()
  local auth_terms = {}
  for _, term in ipairs({ "whitelist", "hwid", "license", "verify", "session", "/retrieve", "/store" }) do
    local count = count_pattern(lower, term, 40)
    if count > 0 then table.insert(auth_terms, { term = term, count = count }) end
  end
  if #auth_terms > 0 then
    add_evidence("surface.whitelist_auth", "info", "StaticAnalysis",
      tostring(#auth_terms) .. " whitelist/auth term groups found", 0.75)
  end
  if #cff >= 6 and src:find("while%s+true") then
    add_evidence("vm.dispatcher", "medium", "StaticAnalysis",
      tostring(#cff) .. " numeric states plus persistent loop", 0.82)
  end
  if indicators.decimal_escapes >= 40 then
    add_evidence("encoding.decimal_escapes", "medium", "StaticAnalysis",
      tostring(indicators.decimal_escapes) .. " decimal escapes", 0.90)
  end
  if indicators.hex_escapes >= 40 then
    add_evidence("encoding.hex_escapes", "medium", "StaticAnalysis",
      tostring(indicators.hex_escapes) .. " hex escapes", 0.90)
  end
  if indicators.base64_literals > 0 then
    add_evidence("encoding.base64", "info", "StaticAnalysis",
      tostring(indicators.base64_literals) .. " Base64-like literal(s)", 0.80)
  end

  local newline_count = count_pattern(src, "\n", 10001)
  local source_structure = {
    lines = math.min(newline_count, 10000) + 1,
    lines_capped = newline_count >= 10001,
    functions = count_pattern(src, "%f[%a]function%f[^%w_]", 1000),
    local_declarations = count_pattern(src, "%f[%a]local%f[^%w_]", 2000),
    conditional_branches = count_pattern(src, "%f[%a]if%f[^%w_]", 2000),
    loops = count_pattern(src, "%f[%a]while%f[^%w_]", 1000)
      + count_pattern(src, "%f[%a]for%f[^%w_]", 1000)
      + count_pattern(src, "%f[%a]repeat%f[^%w_]", 1000),
    table_constructors = count_pattern(src, "%{", 4000),
  }
  constants.summary = {
    string_constants = #constants.strings,
    numeric_constants = #constants.numbers,
    values_bounded = true,
    sensitive_values_redacted = true,
  }

  local seen, uniq = {}, {}
  for _, it in ipairs(strings) do
    if not seen[it.plain] then seen[it.plain] = true; table.insert(uniq, it) end
  end
  strings = uniq
  S.static = {
    strings = strings, urls = urls, keys = keys, tables = tables_n,
    concats = concats, cff = #cff, density = density,
    capabilities = capabilities, auth_terms = auth_terms, indicators = indicators,
    constant_inventory = constants, source_structure = source_structure,
    risk_score = math.min(100, risk_score), source_bytes = source_bytes,
    scanned_bytes = #src, truncated = source_bytes > #src,
  }

  if S.path then
    local lines = { "-- static strings --", "" }
    for i, it in ipairs(strings) do
      if i > 120 then break end
      table.insert(lines, string.format("[%s] %s", it.method, it.plain:gsub("\n", "\\n"):sub(1, 240)))
    end
    write(S.path .. "/extracts/strings.txt", table.concat(lines, "\n"))
    local rep = {
      "-- static report --",
      "strings=" .. #strings .. " urls=" .. #urls .. " key_names=" .. #keys,
      "cff_states=" .. #cff .. " density=" .. tostring(density) .. " xor_tables=" .. tables_n,
      "risk_score=" .. tostring(S.static.risk_score) .. " scanned=" .. tostring(#src) .. "/" .. tostring(source_bytes),
      "decimal_escapes=" .. tostring(indicators.decimal_escapes) .. " hex_escapes=" .. tostring(indicators.hex_escapes),
      "base64_literals=" .. tostring(indicators.base64_literals) .. " base85_like=" .. tostring(indicators.base85_like_literals),
      "",
      "=== CFF STATES (first 40) ===",
      table.concat(cff, ", "):sub(1, 500),
      "",
      "=== URL GRAPH ===",
    }
    for _, u in ipairs(urls) do table.insert(rep, u) end
    table.insert(rep, "")
    table.insert(rep, "=== KEY NAMES (values redacted) ===")
    for _, k in ipairs(keys) do
      table.insert(rep, string.format("%s bytes=%d fingerprint=%s", k.name, k.bytes, k.fingerprint))
    end
    table.insert(rep, "")
    table.insert(rep, "=== CAPABILITIES ===")
    for _, cap in ipairs(capabilities) do
      table.insert(rep, string.format("%s count=%d severity=%s", cap.id, cap.count, cap.severity))
    end
    write(S.path .. "/extracts/static_report.txt", table.concat(rep, "\n"))
    write(S.path .. "/constant_inventory.txt", safe_json(constants))
    write(S.path .. "/endpoint_inventory.txt", safe_json({
      urls = urls,
      auth_terms = auth_terms,
      values_redacted = true,
      source = "captured_target_only",
    }))
    write(S.path .. "/source_structure.txt", safe_json(source_structure))
    if #cff > 0 then
      local cffl = { "-- CFF state ids --" }
      for _, n in ipairs(cff) do table.insert(cffl, tostring(n)) end
      write(S.path .. "/extracts/cff_states.txt", table.concat(cffl, "\n"))
    end
  end
  return S.static
end

-- ── deob ────────────────────────────────────────────────────────────
local function deob_00fuscator(src)
  local notes, out = {}, {}
  table.insert(out, "-- Sentinel Local partial analysis surface")
  table.insert(out, "-- status: partial_surface_only; behavioral equivalence not tested")
  table.insert(out, "-- family: 00fuscator")
  local payload = src
  local payload_match = src:match("local%s+__oo_payload%s*=%s*%[=%[([%s%S]-)%]=%]")
    or src:match("__oo_payload%s*=%s*%[=%[([%s%S]-)%]=%]")
  if payload_match then
    payload = payload_match
    table.insert(notes, "payload_bytes=" .. tostring(#payload))
    local h = 5381
    for i = 1, #payload do
      h = bit32.band(h * 33 + string.byte(payload, i), 0xFFFFFFFF)
    end
    table.insert(notes, "payload_checksum=" .. tostring(h))
  else
    table.insert(notes, "payload_delimiters_missing")
  end
  local recovered = {}
  local scan_limit = effective_static_scan_bytes()
  local scan_payload = #payload > scan_limit and payload:sub(1, scan_limit) or payload
  for chunk in scan_payload:gmatch("string%.char%(([%d%s,]+)%)") do
    analysis_checkpoint(4)
    local bytes = {}
    for n in chunk:gmatch("%d+") do
      local v = tonumber(n)
      if v and v >= 32 and v < 127 then table.insert(bytes, string.char(v)) end
    end
    local s = table.concat(bytes)
    if #s >= 4 then table.insert(recovered, s) end
    if #recovered >= 80 then break end
  end
  local cff = peel_cff(payload)
  table.insert(out, "-- notes: " .. table.concat(notes, ", "))
  table.insert(out, "-- cff_states: " .. #cff)
  table.insert(out, "")
  table.insert(out, "-- ===== recovered string.char =====")
  for _, s in ipairs(recovered) do table.insert(out, "-- " .. s) end
  table.insert(out, "")
  table.insert(out, "-- ===== payload body =====")
  table.insert(out, clip(payload, 100000))
  return table.concat(out, "\n")
end

-- ── hex-blob VM container unwrap (LZ4 + repeating-key XOR) ──────────
-- This family assembles a long hex string by `var = var .. "…"` accumulation
-- and decodes it through a fixed container:
--
--   hex → bytes → [u32 decoded_len][u8 keylen][keylen key][xor-body]
--       → body XORed with the repeating key → LZ4-style decompression
--       → serialized VM program
--
-- The XOR key is embedded in the blob header and the interpreter's two
-- integrity/VM keys are pure functions of the blob text, so the CONTAINER is
-- fully recoverable with no execution. The VM program it yields is a further,
-- separately-encrypted layer; this pass unwraps the container and characterises
-- that program, it does not devirtualize it.

-- Assemble the dominant hex blob: the longest chain of hex string literals
-- joined only by concatenation glue (`..`, names, `=`, `;`, whitespace).
local function hexvm_gather(src, cap)
  local best, best_len, cur, cur_len, prev_end = "", 0, {}, 0, nil
  local pos, n = 1, #src
  while pos <= n do
    local s, e, hex = src:find('"([0-9A-Fa-f]+)"', pos)
    if not s then break end
    if #hex >= 8 then
      local glue_ok = false
      if prev_end then
        local gap = src:sub(prev_end + 1, s - 1)
        if #gap < 80 and gap:match("^[%s%.%w_=;]*$") then glue_ok = true end
      end
      if glue_ok then
        cur[#cur + 1] = hex; cur_len = cur_len + #hex
      else
        if cur_len > best_len then best, best_len = table.concat(cur), cur_len end
        cur = { hex }; cur_len = #hex
      end
      prev_end = e
    end
    pos = e + 1
    if cap and cur_len >= cap then break end
    if (pos % 65536) < 2 then analysis_checkpoint(1) end
  end
  if cur_len > best_len then best, best_len = table.concat(cur), cur_len end
  -- Fallback: some builds break the accumulation with glue this rule rejects
  -- (different statement shapes, extra junk between chunks). The blob still
  -- dominates the file's hex literals, so concatenating them all in source
  -- order reconstructs it; the container header check validates the result.
  if best_len < 1024 then
    local all, all_len = {}, 0
    local p = 1
    while p <= n do
      local s, e, hex = src:find('"([0-9A-Fa-f]+)"', p)
      if not s then break end
      if #hex >= 16 and (#hex % 2) == 0 then
        all[#all + 1] = hex; all_len = all_len + #hex
      end
      p = e + 1
      if cap and all_len >= cap then break end
      if (p % 65536) < 2 then analysis_checkpoint(1) end
    end
    if all_len > best_len then best = table.concat(all) end
  end
  return best
end

local function hexvm_derived_keys(src, hex)
  -- The loader derives one or more integrity/VM keys as
  --   acc = (acc * MULT + nibble) % MOD   over hex chars at fixed positions,
  --   key = (acc + ADD) % MOD
  -- Positions and constants are per-build, so recover them from the loader
  -- text and evaluate on the assembled blob. The source is typically a single
  -- physical line, so bound the lookahead to a fixed window, and collect every
  -- such construction (the interpreter usually takes two).
  local keys = {}
  local pos = 1
  while true do
    local s, e, list = src:find("{%s*([%dxXa-fA-F][%dxXa-fA-F%s,]-)}", pos)
    if not s then break end
    pos = e + 1
    local window = src:sub(e + 1, e + 320)
    local mult = window:match("^[^\n]-%*%s*(%w+)%s*%+")
    local mod  = window:match("^[^\n]-%%%s*(%w+)%s*end")
    local add  = window:match("end%s*[;\r\n]?%s*local%s+[%w_]+%s*=%s*%([%w_]+%s*%+%s*(%w+)%s*%)%s*%%")
    if window:find(":sub%s*%(", 1, false) and mult and mod then
      local MULT, MOD, ADD = tonumber(mult), tonumber(mod), tonumber(add or "0")
      if MULT and MOD and MOD > 0 then
        local ps = {}
        for tok in list:gmatch("[%dxXa-fA-F]+") do
          local v = tonumber(tok)
          if v then ps[#ps + 1] = math.floor(v) end
        end
        if #ps >= 4 then
          local acc = 0
          for _, p in ipairs(ps) do
            acc = (acc * MULT + (tonumber(hex:sub(p, p), 16) or 0)) % MOD
          end
          keys[#keys + 1] = { positions = #ps, mult = MULT, mod = MOD, add = ADD,
                              value = (acc + ADD) % MOD }
        end
      end
    end
  end
  return keys
end

-- Deserialize the recovered VM program into its proto tree and recover the
-- constant pool — the actual payload translation. The serialized format
-- (recovered from the loader's IIIOIl1O deserializer) is, per proto:
--   u8 u8 u8 u8                       -- numparams / vararg / maxstack / upvalues
--   u32 ninstr, ninstr × u32          -- instruction words
--   u32 nconst, nconst × const:
--       four per-build tag values, recovered from the loader dispatch:
--       nil; u8 boolean; f64 (IEEE754 LE); u32 len + XORed string bytes
--   u32 nproto, nproto × (u32 index, proto)        -- children (recursive)
-- key1 is the blob-derived byte key; each string byte is XORed with it.
-- Bounded by a wall-clock deadline and node/const caps so a hostile or
-- oversized program can never exceed the tool's time budget.
local function hexvm_readf64(prog, p)
  local b1, b2, b3, b4 = prog[p], prog[p + 1], prog[p + 2], prog[p + 3]
  local b5, b6, b7, b8 = prog[p + 4], prog[p + 5], prog[p + 6], prog[p + 7]
  local sign = 1
  if b8 >= 0x80 then sign = -1; b8 = b8 - 0x80 end
  local exp = b8 * 16 + math.floor(b7 / 16)
  local frac = (b7 % 16) * 2 ^ 48 + b6 * 2 ^ 40 + b5 * 2 ^ 32
             + b4 * 2 ^ 24 + b3 * 2 ^ 16 + b2 * 2 ^ 8 + b1
  local v
  if exp == 0 then
    v = (frac == 0) and 0 or sign * frac * 2 ^ (-1074)
  elseif exp == 0x7FF then
    v = (frac == 0) and (sign * math.huge) or (0 / 0)
  else
    v = sign * (1 + frac / 2 ^ 52) * 2 ^ (exp - 1023)
  end
  return v, p + 8
end

-- Recover the deserializer's constant tag bytes from the loader source.
-- The obfuscator randomizes these per build, so they must never be hardcoded.
-- The dispatch is a chain of `==` comparisons on one variable; each branch is
-- classified by what it does, not by its tag value or its position:
--   no payload read        -> nil
--   compares the byte ~= 0 -> boolean
--   call taking (buf,pos)  -> number (f64)
--   call taking (buf,pos,k)-> string (the extra arg is the string XOR key)
local function hexvm_const_tags(src)
  local found_all = {}
  local pos = 1
  while true do
    local s, e, var, t0 = src:find("if%s+([%w_]+)%s*==%s*(%w+)%s+then", pos)
    if not s then break end
    pos = e + 1
    local window = src:sub(e + 1, e + 1800)
    -- collect this chain's branches: the anchor plus following elseifs
    local br = { { tag = tonumber(t0), at = 1, from = 1 } }
    local p = 1
    while #br < 6 do
      local a, b, tv = window:find("elseif%s+" .. var .. "%s*==%s*(%w+)%s+then", p)
      if not a then break end
      br[#br].stop = a - 1
      br[#br + 1] = { tag = tonumber(tv), at = a, from = b + 1 }
      p = b + 1
    end
    if br[#br].stop == nil then
      br[#br].stop = math.min(#window, br[#br].from + 400)
      br[#br].last = true
    end
    if #br >= 4 then
      local found = {}
      for _, it in ipairs(br) do
        if it.tag then
          local body = window:sub(it.from, it.stop)
          -- the final branch has no `elseif` to bound it: cut at the `end`
          -- that closes the chain, or its body bleeds into following code
          -- and a nil branch gets misread as a call-bearing one.
          if it.last then body = body:match("^(.-)%f[%w]end%f[%W]") or body end
          local call = body:match("=%s*[%w_]+(%b())")
          local commas = 0
          if call then for _ in call:gmatch(",") do commas = commas + 1 end end
          if body:find("~=%s*0") then
            found.bool_tag = found.bool_tag or it.tag
          elseif call and commas >= 2 then
            found.str_tag = found.str_tag or it.tag
          elseif call and commas == 1 then
            found.num_tag = found.num_tag or it.tag
          elseif not call then
            found.nil_tag = found.nil_tag or it.tag
          end
        end
      end
      if found.str_tag and found.num_tag and found.bool_tag and found.nil_tag then
        -- Return every candidate chain rather than committing to the first:
        -- junk code can contain a lookalike dispatch, so the caller validates
        -- each by whether the program actually parses to completion.
        found_all[#found_all + 1] = found
        if #found_all >= 6 then break end
      end
    end
  end
  return found_all
end

-- Deserialize the VM program. Two passes: the first walks the structure and
-- records where each string lives (without decoding it), the second decodes
-- them once the XOR key is settled. The key is recovered by scoring every
-- candidate 0..255 against the printable-ASCII ratio of the collected string
-- bytes, so a wrong or unextractable derived key can no longer break the
-- translation. `hint_key` (from the loader's derived-key formula) only wins
-- ties.
local function hexvm_deserialize(prog, pi, tags, hint_key, instr_key, deadline)
  local ranges, numbers, protos = {}, {}, {}
  local nproto, ninstr_total = 0, 0
  local complete, aborted_reason = true, nil
  local MAX_PROTOS, MAX_STRINGS, MAX_DEPTH = 6000, 12000, 400
  local MAX_INSTRUCTIONS, MAX_CONSTANTS = 200000, 100000
  local checks = 0
  local T_NIL, T_BOOL, T_NUM, T_STR =
    tags.nil_tag, tags.bool_tag, tags.num_tag, tags.str_tag

  local function u32(p)
    if not p or p < 1 or (p + 3) > pi then error("truncated_u32") end
    return prog[p] + prog[p + 1] * 256 + prog[p + 2] * 65536 + prog[p + 3] * 16777216, p + 4
  end

  local function proto(p, depth)
    if p == nil or aborted_reason then return p, nil end
    nproto = nproto + 1
    if nproto > MAX_PROTOS then aborted_reason = "proto_cap"; return p, nil end
    if depth > MAX_DEPTH then aborted_reason = "depth_cap"; return p, nil end
    checks = checks + 1
    if (checks % 256) == 0 then
      analysis_checkpoint(2)
      if deadline and os.clock() > deadline then
        aborted_reason = "deadline"; complete = false; return p, nil
      end
    end
    if (p + 7) > pi then error("truncated_proto_header") end
    local record = {
      id = nproto, depth = depth,
      header = { prog[p], prog[p + 1], prog[p + 2], prog[p + 3] },
      instructions = {}, constants = {}, children = {},
    }
    protos[nproto] = record
    p = p + 4                                   -- proto header bytes
    local ni; ni, p = u32(p)                    -- instruction words
    if ni > MAX_INSTRUCTIONS or (p + ni * 4 - 1) > pi then error("invalid_instruction_count:" .. tostring(ni)) end
    ninstr_total = ninstr_total + ni
    for pc = 1, ni do
      local encoded; encoded, p = u32(p)
      local decoded = bit32.bxor(encoded, instr_key or 0)
      local bx = math.floor(decoded / 16384) % 262144
      record.instructions[pc] = {
        encoded = encoded, raw = decoded,
        op = decoded % 64,
        A = math.floor(decoded / 64) % 256,
        B = math.floor(decoded / 16384) % 512,
        C = math.floor(decoded / 8388608) % 512,
        Bx = bx, sBx = bx - 131071,
      }
      if (pc % 2048) == 0 then
        analysis_checkpoint(1)
        if deadline and os.clock() > deadline then
          aborted_reason = "deadline"; complete = false; return p, record.id
        end
      end
    end
    local nc; nc, p = u32(p)                    -- constant pool
    if nc > MAX_CONSTANTS then error("constant_cap:" .. tostring(nc)) end
    for ci = 1, nc do
      if p > pi then error("truncated_constant_tag") end
      local tag = prog[p]; p = p + 1
      local constant = { tag = tag }
      if tag == T_NIL then
        constant.kind = "nil"
      elseif tag == T_BOOL then
        if p > pi then error("truncated_boolean") end
        constant.kind, constant.value = "boolean", prog[p] ~= 0
        p = p + 1
      elseif tag == T_NUM then
        if (p + 7) > pi then error("truncated_number") end
        constant.kind, constant.value = "number", hexvm_readf64(prog, p)
        if #numbers < 4000 then numbers[#numbers + 1] = constant.value end
        p = p + 8
      elseif tag == T_STR then
        local n; n, p = u32(p)
        if n > MAX_VM_STRING_BYTES or (p + n - 1) > pi then
          error("invalid_string_length:" .. tostring(n))
        end
        constant.kind, constant.bytes = "string", n
        if n > 0 then
          if #ranges < MAX_STRINGS then
            ranges[#ranges + 1] = { off = p, len = n, constant = constant }
          end
          p = p + n
        else
          constant.value = ""
        end
      else
        aborted_reason = "bad_const_tag:" .. tostring(tag); complete = false; return p, record.id
      end
      record.constants[ci] = constant
    end
    local np; np, p = u32(p)                    -- child protos
    if np > MAX_PROTOS then error("child_proto_cap:" .. tostring(np)) end
    for child = 1, np do
      local child_index; child_index, p = u32(p)
      local child_id
      p, child_id = proto(p, depth + 1)
      record.children[child] = { index = child_index, proto = child_id }
      if aborted_reason then return p, record.id end
    end
    return p, record.id
  end

  local ok, endp, root_id = pcall(proto, 1, 0)
  if not ok then
    complete = false
    aborted_reason = aborted_reason or ("parse_error:" .. tostring(endp):sub(1, 60))
    endp = nil
  end

  -- Settle the string XOR key. Plain printability is not enough on its own:
  -- a small key (e.g. 3) maps lowercase to lowercase, so the true key and a
  -- wrong one score alike. Weight the characters that actually dominate real
  -- code/text — space above all — and add a dictionary bonus for tokens these
  -- payloads almost always contain. The loader-derived key only breaks ties.
  local key, key_source = hint_key or 0, hint_key and "derived" or "none"
  do
    local sample, sn = {}, 0
    for _, r in ipairs(ranges) do
      for i = 0, r.len - 1 do
        sn = sn + 1; sample[sn] = prog[r.off + i]
        if sn >= 4096 then break end
      end
      if sn >= 4096 then break end
    end
    local DICT = { "game", "string", "table", "Instance", "GetService",
                   "Player", "http", "Enum", "Parent", "Name", "function",
                   "local", "math", "Value", "Text", "Color" }
    if sn > 0 then
      local best_k, best_score = nil, -1e9
      for k = 0, 255 do
        local score, chars = 0, {}
        for i = 1, sn do
          local c = bit32.bxor(sample[i], k)
          chars[i] = string.char(c)
          if c == 32 then score = score + 6
          elseif c >= 97 and c <= 122 then score = score + 3
          elseif c >= 65 and c <= 90 then score = score + 2
          elseif c >= 48 and c <= 57 then score = score + 2
          elseif c == 46 or c == 47 or c == 58 or c == 95 or c == 45
              or c == 40 or c == 41 or c == 44 then score = score + 1
          elseif c == 9 or c == 10 or c == 13 then score = score + 1
          elseif c < 32 or c >= 127 then score = score - 8
          end
        end
        local text = table.concat(chars)
        for _, w in ipairs(DICT) do
          if text:find(w, 1, true) then score = score + 400 end
        end
        if score > best_score or (score == best_score and hint_key and k == hint_key) then
          best_k, best_score = k, score
        end
        if (k % 32) == 0 then analysis_checkpoint(2) end
      end
      if best_k then
        if hint_key and best_k == hint_key then
          key, key_source = best_k, "derived+scored"
        else
          key, key_source = best_k, "scored"
        end
      end
    end
  end

  -- second pass: materialise the strings with the settled key
  local strings = {}
  for i, r in ipairs(ranges) do
    local chars = {}
    for j = 0, r.len - 1 do
      chars[j + 1] = string.char(bit32.bxor(prog[r.off + j], key))
    end
    strings[i] = table.concat(chars)
    if r.constant then r.constant.value = strings[i] end
    if (i % 512) == 0 then analysis_checkpoint(1) end
  end

  return {
    strings = strings,
    numbers = numbers,
    protos = protos,
    root_proto = root_id,
    proto_count = nproto,
    instr_count = ninstr_total,
    consumed = endp,
    total = pi,
    key = key,
    key_source = key_source,
    complete = complete and (endp == pi + 1 or endp == pi),
    aborted_reason = aborted_reason,
    instruction_key = instr_key or 0,
  }
end

-- Render a valid Lua data module containing every decoded prototype, constant
-- and instruction field. This is the strongest exact static recovery possible
-- after compilation: source comments, formatting and original local names no
-- longer exist in the protected program.
local function render_luasnapper_full_deob(deser, recovered)
  if not deser or not deser.protos then return "" end
  local opcodes = {
    [0] = "SETGLOBAL", [1] = "NOT", [2] = "JMP", [3] = "LE",
    [4] = "GETUPVAL", [5] = "SETTABLE", [6] = "SELF", [7] = "BXOR",
    [8] = "LOADK", [9] = "CALL_FIXED2", [10] = "CONCAT", [11] = "CALL",
    [12] = "SETUPVAL", [13] = "DIV", [14] = "TAILCALL", [15] = "GETTABLE",
    [16] = "LEN", [17] = "SUB", [18] = "MUL", [19] = "TESTSET",
    [20] = "SETLIST", [21] = "EQ", [22] = "GETGLOBAL", [23] = "ADD",
    [24] = "CLOSE", [25] = "FORLOOP", [26] = "LOADNIL", [27] = "RETURN",
    [28] = "UNM", [29] = "FORPREP", [30] = "LOADBOOL", [31] = "MOD",
    [32] = "NEWTABLE", [33] = "VARARG", [34] = "TEST", [35] = "LT",
    [36] = "MOVE", [37] = "CLOSURE",
  }
  local function quote(s)
    s = tostring(s or "")
    return '"' .. s:gsub('[%z\1-\31\127-\255\\"]', function(c)
      return string.format("\\%03d", string.byte(c))
    end) .. '"'
  end
  local function number_literal(v)
    if v ~= v then return "(0/0)" end
    if v == math.huge then return "math.huge" end
    if v == -math.huge then return "-math.huge" end
    return tostring(v)
  end
  local function instruction_text(i)
    local function rk(x)
      return x >= 256 and ("K[" .. tostring(x - 255) .. "]") or ("R[" .. tostring(x) .. "]")
    end
    local A, B, C = tostring(i.A), tostring(i.B), tostring(i.C)
    local op = i.op
    if op == 0 then return "ENV[K[" .. tostring(i.Bx + 1) .. "]] = R[" .. A .. "]"
    elseif op == 1 then return "R[" .. A .. "] = not R[" .. B .. "]"
    elseif op == 2 then return "pc = pc + " .. tostring(i.sBx)
    elseif op == 3 then return "LE " .. rk(i.B) .. ", " .. rk(i.C) .. " (conditional skip)"
    elseif op == 4 then return "R[" .. A .. "] = UP[" .. B .. "]"
    elseif op == 5 then return "R[" .. A .. "][" .. rk(i.B) .. "] = " .. rk(i.C)
    elseif op == 6 then return "R[" .. A .. "+1] = R[" .. B .. "]; R[" .. A .. "] = R[" .. B .. "][" .. rk(i.C) .. "]"
    elseif op == 7 then return "R[" .. A .. "] = bxor(" .. rk(i.B) .. ", " .. rk(i.C) .. ")"
    elseif op == 8 then return "R[" .. A .. "] = K[" .. tostring(i.Bx + 1) .. "]"
    elseif op == 9 then return "fixed two-argument call at R[" .. A .. "]"
    elseif op == 10 then return "R[" .. A .. "] = concat(R[" .. B .. ".." .. C .. "])"
    elseif op == 11 then return "CALL R[" .. A .. "] args=" .. B .. " results=" .. C
    elseif op == 12 then return "UP[" .. B .. "] = R[" .. A .. "]"
    elseif op == 13 then return "R[" .. A .. "] = " .. rk(i.B) .. " / " .. rk(i.C)
    elseif op == 14 then return "TAILCALL R[" .. A .. "] args=" .. B
    elseif op == 15 then return "R[" .. A .. "] = R[" .. B .. "][" .. rk(i.C) .. "]"
    elseif op == 16 then return "R[" .. A .. "] = #R[" .. B .. "]"
    elseif op == 17 then return "R[" .. A .. "] = " .. rk(i.B) .. " - " .. rk(i.C)
    elseif op == 18 then return "R[" .. A .. "] = " .. rk(i.B) .. " * " .. rk(i.C)
    elseif op == 19 then return "TESTSET R[" .. A .. "], R[" .. B .. "], " .. C
    elseif op == 20 then return "SETLIST R[" .. A .. "] count=" .. B .. " block=" .. C
    elseif op == 21 then return "EQ " .. rk(i.B) .. ", " .. rk(i.C) .. " (conditional skip)"
    elseif op == 22 then return "R[" .. A .. "] = ENV[K[" .. tostring(i.Bx + 1) .. "]]"
    elseif op == 23 then return "R[" .. A .. "] = " .. rk(i.B) .. " + " .. rk(i.C)
    elseif op == 24 then return "CLOSE upvalues from R[" .. A .. "]"
    elseif op == 25 then return "FORLOOP R[" .. A .. "] pc+=" .. tostring(i.sBx)
    elseif op == 26 then return "R[" .. A .. ".." .. B .. "] = nil"
    elseif op == 27 then return "RETURN R[" .. A .. "] count=" .. B
    elseif op == 28 then return "R[" .. A .. "] = -R[" .. B .. "]"
    elseif op == 29 then return "FORPREP R[" .. A .. "] pc+=" .. tostring(i.sBx)
    elseif op == 30 then return "R[" .. A .. "] = " .. tostring(i.B ~= 0) .. (i.C ~= 0 and "; pc++" or "")
    elseif op == 31 then return "R[" .. A .. "] = " .. rk(i.B) .. " % " .. rk(i.C)
    elseif op == 32 then return "R[" .. A .. "] = {}"
    elseif op == 33 then return "VARARG -> R[" .. A .. "] count=" .. B
    elseif op == 34 then return "TEST R[" .. A .. "], " .. C
    elseif op == 35 then return "LT " .. rk(i.B) .. ", " .. rk(i.C) .. " (conditional skip)"
    elseif op == 36 then return "R[" .. A .. "] = R[" .. B .. "]"
    elseif op == 37 then return "R[" .. A .. "] = CLOSURE(proto " .. tostring(i.Bx + 1) .. ")"
    end
    return "unclassified VM opcode " .. tostring(op)
  end
  local out = {
    "-- Sentinel LuaSnapper full_deob",
    "-- static recovery only; the protected source was never executed",
    "-- status: " .. (deser.complete and "decoded_vm_ir_complete" or ("decoded_vm_ir_partial:" .. tostring(deser.aborted_reason))),
    "-- exact original comments, whitespace and local names were removed by compilation",
    "local FULL_DEOB = {",
    "  schema = \"sentinel.luasnapper-ir.v1\",",
    "  family = \"LuaSnapper\",",
    "  recovery = \"decoded VM intermediate representation\",",
    "  complete = " .. tostring(deser.complete) .. ",",
    "  root_proto = " .. tostring(deser.root_proto or 1) .. ",",
    "  proto_count = " .. tostring(deser.proto_count or 0) .. ",",
    "  instruction_count = " .. tostring(deser.instr_count or 0) .. ",",
    "  string_xor_key = " .. tostring(deser.key or 0) .. ",",
    "  instruction_xor_key = " .. tostring(deser.instruction_key or 0) .. ",",
    "  container = {",
    "    blob_bytes = " .. tostring(recovered and recovered.blob_bytes or 0) .. ",",
    "    declared_length = " .. tostring(recovered and recovered.declared_len or 0) .. ",",
    "    decompressed_bytes = " .. tostring(recovered and recovered.decompressed or 0) .. ",",
    "    xor_key = " .. quote(recovered and recovered.xor_key or "") .. ",",
    "  },",
    "  protos = {",
  }
  local instruction_budget = (S.device and S.device.mobile) and 16000 or 50000
  local emitted_instructions = 0
  for _, proto in ipairs(deser.protos) do
    local h = proto.header or {}
    out[#out + 1] = string.format(
      "    [%d] = { depth=%d, numparams=%d, vararg=%d, maxstack=%d, upvalues=%d,",
      proto.id, proto.depth or 0, h[1] or 0, h[2] or 0, h[3] or 0, h[4] or 0)
    out[#out + 1] = "      constants = {"
    for ci, c in ipairs(proto.constants or {}) do
      local value
      if c.kind == "nil" then value = "{ kind=\"nil\" }"
      elseif c.kind == "boolean" then value = "{ kind=\"boolean\", value=" .. tostring(c.value) .. " }"
      elseif c.kind == "number" then value = "{ kind=\"number\", value=" .. number_literal(c.value) .. " }"
      elseif c.kind == "string" then value = "{ kind=\"string\", value=" .. quote(c.value or "") .. " }"
      else value = "{ kind=\"unknown\", tag=" .. tostring(c.tag) .. " }" end
      out[#out + 1] = string.format("        [%d] = %s,", ci, value)
    end
    out[#out + 1] = "      },"
    out[#out + 1] = "      instructions = {"
    for pc, ins in ipairs(proto.instructions or {}) do
      if emitted_instructions >= instruction_budget then
        out[#out + 1] = "        -- instruction output budget reached; remaining decoded instructions omitted"
        break
      end
      emitted_instructions = emitted_instructions + 1
      local name = opcodes[ins.op] or ("VM_OP_" .. tostring(ins.op))
      out[#out + 1] = string.format(
        "        { pc=%d, op=%d, name=%q, text=%s, A=%d, B=%d, C=%d, Bx=%d, sBx=%d, raw=%u, encoded=%u },",
        pc, ins.op, name, quote(instruction_text(ins)),
        ins.A, ins.B, ins.C, ins.Bx, ins.sBx, ins.raw, ins.encoded)
    end
    out[#out + 1] = "      },"
    out[#out + 1] = "      children = {"
    for _, child in ipairs(proto.children or {}) do
      out[#out + 1] = string.format("        { index=%d, proto=%s },",
        child.index or 0, tostring(child.proto or "nil"))
    end
    out[#out + 1] = "      },"
    out[#out + 1] = "    },"
    if (#out % 512) == 0 then analysis_checkpoint(1) end
  end
  out[#out + 1] = "  },"
  out[#out + 1] = "}"
  out[#out + 1] = "return FULL_DEOB"
  return table.concat(out, "\n")
end
local function deob_hexvm(src, deadline)
  local out, notes = {}, {}
  local decoded_family = S.detect and S.detect.family_scores
    and S.detect.family_scores.LuaSnapper and "LuaSnapper" or "HexBlob-VM"
  table.insert(out, "-- Sentinel Local blob-container recovery")
  table.insert(out, "-- family: " .. decoded_family .. " (HexBlob-VM/LZ4 transport)")
  table.insert(out, "-- status: container_recovered; vm_program_pending (not devirtualized)")

  local cap = (S.device and S.device.mobile) and 400000 or 3200000
  local hex = hexvm_gather(src, cap)
  if #hex < 64 then
    table.insert(out, "-- blob not located (hex chain too short)")
    return table.concat(out, "\n"), nil
  end
  table.insert(notes, "blob_hex_chars=" .. #hex)

  -- hex → bytes
  local blen = #hex - (#hex % 2)
  local t, ti = {}, 0
  for i = 1, blen, 2 do
    ti = ti + 1
    t[ti] = tonumber(hex:sub(i, i + 1), 16) or 0
    if (ti % 32768) == 0 then analysis_checkpoint(2) end
  end
  if ti < 6 then
    table.insert(out, "-- blob too small for a container header")
    return table.concat(out, "\n"), nil
  end

  -- header
  local declared = t[1] + t[2] * 256 + t[3] * 65536 + t[4] * 16777216
  local keylen = t[5]
  if keylen < 1 or keylen > 64 or (6 + keylen) > ti then
    table.insert(out, "-- header rejected (keylen=" .. tostring(keylen) .. ")")
    table.insert(out, "-- notes: " .. table.concat(notes, ", "))
    return table.concat(out, "\n"), nil
  end
  local max_program = (S.device and S.device.mobile) and 1000000 or MAX_SOURCE_BYTES
  if declared < 1 or declared > max_program then
    table.insert(out, "-- header rejected (declared output " .. tostring(declared)
      .. " exceeds " .. tostring(max_program) .. "-byte budget)")
    return table.concat(out, "\n"), {
      blob_hex_chars = #hex, blob_bytes = ti, declared_len = declared,
      keylen = keylen, decode_ok = false, length_match = false,
      translation_status = "declared_length_budget",
    }
  end
  local key, key_hex = {}, {}
  for i = 1, keylen do key[i] = t[5 + i]; key_hex[i] = string.format("%02x", t[5 + i]) end
  local body_start = 6 + keylen

  -- repeating-key XOR (in place)
  for i = body_start, ti do
    t[i] = bit32.bxor(t[i], key[((i - body_start) % keylen) + 1])
    if (i % 65536) == 0 then analysis_checkpoint(2) end
  end

  -- LZ4-style decompression
  local prog, pi = {}, 0
  local p = body_start
  local ok_decode, decode_reason = true, "ok"
  while p <= ti do
    local tok = t[p]; p = p + 1
    local litlen = math.floor(tok / 16)
    if litlen == 15 then
      local e
      repeat
        if p > ti then ok_decode = false; decode_reason = "truncated_literal_length"; break end
        e = t[p]; p = p + 1; litlen = litlen + e
      until e ~= 255
      if not ok_decode then break end
    end
    if (p + litlen - 1) > ti then
      ok_decode = false; decode_reason = "truncated_literals"; break
    end
    for _ = 1, litlen do
      pi = pi + 1; prog[pi] = t[p]; p = p + 1
      if pi > declared then ok_decode = false; decode_reason = "declared_length_overrun"; break end
    end
    if not ok_decode then break end
    if p > ti then break end
    if (p + 1) > ti then ok_decode = false; decode_reason = "truncated_offset"; break end
    local off = t[p] + t[p + 1] * 256; p = p + 2
    local mlen = tok % 16 + 4
    if mlen == 19 then
      local e
      repeat
        if p > ti then ok_decode = false; decode_reason = "truncated_match_length"; break end
        e = t[p]; p = p + 1; mlen = mlen + e
      until e ~= 255
      if not ok_decode then break end
    end
    local start = pi - off
    if off <= 0 or start < 0 then
      ok_decode = false; decode_reason = "invalid_offset:" .. tostring(off); break
    end
    for j = 1, mlen do
      if pi >= declared then ok_decode = false; decode_reason = "declared_length_overrun"; break end
      pi = pi + 1; prog[pi] = prog[start + j]
      if prog[pi] == nil then ok_decode = false; decode_reason = "invalid_backreference"; break end
    end
    if not ok_decode then break end
    if (pi % 65536) < 2 then analysis_checkpoint(2) end
    if deadline and os.clock() > deadline then
      ok_decode = false; decode_reason = "deadline"; break
    end
  end

  table.insert(notes, "declared_len=" .. declared)
  table.insert(notes, "keylen=" .. keylen)
  table.insert(notes, "xor_key=" .. table.concat(key_hex))
  table.insert(notes, "decompressed=" .. pi)
  table.insert(notes, "length_match=" .. tostring(pi == declared))
  table.insert(notes, "decode_ok=" .. tostring(ok_decode))
  table.insert(notes, "decode_status=" .. tostring(decode_reason))

  -- program characterisation (byte histogram + printable fragments + djb2)
  local hist, printable_runs = {}, {}
  for i = 0, 255 do hist[i] = 0 end
  local h = 5381
  local run, run_bytes = {}, 0
  local frag_count = 0
  local frag_cap = (S.device and S.device.mobile) and 24 or 80
  for i = 1, pi do
    local b = prog[i]
    hist[b] = hist[b] + 1
    h = (h * 33 + b) % 4294967296
    if b >= 32 and b < 127 then
      run[#run + 1] = string.char(b); run_bytes = run_bytes + 1
    else
      if run_bytes >= 5 and frag_count < frag_cap then
        frag_count = frag_count + 1
        printable_runs[frag_count] = table.concat(run)
      end
      run = {}; run_bytes = 0
    end
  end
  if run_bytes >= 5 and frag_count < frag_cap then
    printable_runs[#printable_runs + 1] = table.concat(run)
  end

  -- top opcode-ish byte values (leading byte density is a cheap VM signal)
  local top = {}
  for i = 0, 255 do top[#top + 1] = { b = i, n = hist[i] } end
  table.sort(top, function(a, b) return a.n > b.n end)

  -- Program-level entropy and printable ratio, computed for free from the
  -- full histogram just built (no re-scan).  This is the decoded VM program's
  -- own profile -- the far end of the string -> bytes -> program lineage.
  local prog_entropy = entropy_from_hist(hist, pi)
  local prog_printable = 0
  for b = 0, 255 do
    if (b >= 32 and b < 127) or b == 9 or b == 10 or b == 13 then prog_printable = prog_printable + hist[b] end
  end
  local prog_printable_ratio = pi > 0 and prog_printable / pi or 0
  local top_bytes = {}
  for i = 1, math.min(12, #top) do top_bytes[i] = { byte = top[i].b, count = top[i].n } end

  -- blob-derived integrity/VM keys (best-effort; usually two)
  local dks = hexvm_derived_keys(src, hex)

  table.insert(out, "-- notes: " .. table.concat(notes, ", "))
  table.insert(out, "-- program_djb2=" .. h)
  if #dks > 0 then
    for i, dk in ipairs(dks) do
      table.insert(out, string.format(
        "-- derived_key[%d]: value=%d (0x%X)  positions=%d  mult=%d  mod=%d  add=%d",
        i, dk.value, dk.value, dk.positions, dk.mult, dk.mod, dk.add))
    end
  else
    table.insert(out, "-- derived_key: present in loader; formula not auto-extracted this build")
  end
  table.insert(out, "-- top_byte_values (opcode-density hint):")
  do
    local line = {}
    for i = 1, math.min(12, #top) do
      line[#line + 1] = string.format("0x%02X:%d", top[i].b, top[i].n)
    end
    table.insert(out, "--   " .. table.concat(line, "  "))
  end

  -- ── payload translation: deserialize the VM program and recover the
  --    constant pool (the actual strings/numbers the payload runs on).
  --    Constant tags and the string key are randomized per build, so both are
  --    detected from this sample rather than assumed.
  local candidates = hexvm_const_tags(src)
  local hint_key = dks[1] and (dks[1].value % 256) or nil
  local instruction_key = dks[2] and dks[2].value or 0
  local deser, tags = nil, nil
  if ok_decode then
    for i, cand in ipairs(candidates) do
      local try = hexvm_deserialize(prog, pi, cand, hint_key, instruction_key, deadline)
      if try and try.complete then
        deser, tags = try, cand
        break
      end
      -- keep the best partial in case none validates
      if try and (not deser or #try.strings > #deser.strings) then
        deser, tags = try, cand
      end
      if try and try.aborted_reason == "deadline" then break end
      if deadline and os.clock() > deadline then break end
    end
  end
  if tags then
    table.insert(out, string.format(
      "-- const_tags (auto-detected, %d candidate%s): nil=0x%02X bool=0x%02X number=0x%02X string=0x%02X",
      #candidates, (#candidates == 1) and "" or "s",
      tags.nil_tag, tags.bool_tag, tags.num_tag, tags.str_tag))
  else
    table.insert(out, "-- const_tags: dispatch not located in loader; translation skipped")
  end

  table.insert(out, "")
  if deser and #deser.strings > 0 then
    -- dedupe strings, longest first (URLs/keys surface at the top)
    local seen, uniq = {}, {}
    for _, s in ipairs(deser.strings) do
      if s ~= "" and not seen[s] then seen[s] = true; uniq[#uniq + 1] = s end
    end
    table.sort(uniq, function(a, b) return #a > #b end)

    table.insert(out, string.format(
      "-- ===== PAYLOAD TRANSLATION (VM constant pool) =====", 0))
    table.insert(out, string.format(
      "-- protos=%d  instructions=%d  string_consts=%d (unique %d)  number_consts=%d  status=%s",
      deser.proto_count, deser.instr_count, #deser.strings, #uniq, #deser.numbers,
      deser.complete and "complete" or ("partial:" .. tostring(deser.aborted_reason))))
    table.insert(out, string.format("-- string XOR key = %d (0x%02X) via %s",
      deser.key, deser.key, tostring(deser.key_source)))
    table.insert(out, "")
    table.insert(out, "local RECOVERED_STRINGS = {")
    local budget = (S.device and S.device.mobile) and 1200 or 6000
    local bytes_out = 0
    for i, s in ipairs(uniq) do
      if i > budget or bytes_out > 400000 then
        table.insert(out, string.format("  -- … %d more strings omitted (output cap)", #uniq - i + 1))
        break
      end
      local safe = s:gsub("[%z\1-\31\127-\255\\\"]", function(c) return string.format("\\%d", c:byte()) end)
      if #safe > 300 then safe = safe:sub(1, 300) .. "\\z…" end
      table.insert(out, string.format("  [%d] = \"%s\",", i, safe))
      bytes_out = bytes_out + #safe + 12
    end
    table.insert(out, "}")
    if #deser.numbers > 0 then
      table.insert(out, "")
      table.insert(out, "-- number constants (first 60):")
      local nl = {}
      for i = 1, math.min(60, #deser.numbers) do
        nl[#nl + 1] = tostring(deser.numbers[i])
      end
      table.insert(out, "--   " .. table.concat(nl, ", "))
    end
    table.insert(out, "")
    table.insert(out, "-- The container and the VM constant pool are fully recovered above")
    table.insert(out, "-- (no execution). Instruction opcodes are read but not yet lifted to")
    table.insert(out, "-- source; constant recovery is the payload's readable surface.")
  else
    -- deserialization did not apply (wrong key / different build): fall back
    -- to the raw printable-fragment surface so output is never empty.
    table.insert(out, "-- ===== VM program: printable fragments (" .. #printable_runs .. ") =====")
    if deser and deser.aborted_reason then
      table.insert(out, "-- (deserialization stopped: " .. tostring(deser.aborted_reason) .. ")")
    end
    for _, s in ipairs(printable_runs) do
      table.insert(out, "-- " .. s:gsub("\n", " "):sub(1, 160))
    end
  end

  local recovered = {
    blob_hex_chars = #hex,
    blob_bytes = ti,
    declared_len = declared,
    keylen = keylen,
    xor_key = table.concat(key_hex),
    decompressed = pi,
    length_match = (pi == declared),
    decode_ok = ok_decode,
    decode_status = decode_reason,
    program_djb2 = h,
    program_entropy = prog_entropy,
    program_printable_ratio = prog_printable_ratio,
    top_bytes = top_bytes,
    printable_fragments = #printable_runs,
    derived_keys = (function()
      local v = {}
      for _, dk in ipairs(dks) do v[#v + 1] = dk.value end
      return v
    end)(),
    proto_count = deser and deser.proto_count or nil,
    instr_count = deser and deser.instr_count or nil,
    string_consts = deser and #deser.strings or nil,
    number_consts = deser and #deser.numbers or nil,
    translation_complete = deser and deser.complete or false,
    translation_status = deser and (deser.complete and "complete"
      or ("partial:" .. tostring(deser.aborted_reason)))
      or (tags and "not_attempted" or "tags_not_located"),
    const_tags = tags and string.format("nil=%d,bool=%d,num=%d,str=%d",
      tags.nil_tag, tags.bool_tag, tags.num_tag, tags.str_tag) or nil,
    string_key = deser and deser.key or nil,
    string_key_source = deser and deser.key_source or nil,
    instruction_key = deser and deser.instruction_key or instruction_key,
  }
  S.full_deob = render_luasnapper_full_deob(deser, recovered)
  recovered.full_deob_bytes = #S.full_deob
  return table.concat(out, "\n"), recovered
end

local function deob_generic(src, detect)
  local st = S.static or {}
  local head = {
    "-- Sentinel Local partial analysis surface",
    "-- status: partial_surface_only; runtime validation unverified",
    "-- this is not a claim of complete VM devirtualization",
    "-- family: " .. tostring(detect.family),
    "-- confidence: " .. tostring(detect.confidence),
    "-- density: " .. tostring(st.density or 0),
    "-- cff_states: " .. tostring(st.cff or 0),
    "-- recovered_strings: " .. tostring(st.strings and #st.strings or 0),
    "-- notes: generic_pass + surface reconstruction",
    "",
    "-- ===== reconstructed surface =====",
  }
  if st.urls then
    table.insert(head, "-- URLs:")
    for i, u in ipairs(st.urls) do
      if i > 20 then break end
      table.insert(head, "--   " .. u)
    end
  end
  if st.strings then
    table.insert(head, "-- Strings:")
    for i, it in ipairs(st.strings) do
      if i > 30 then break end
      table.insert(head, "--   [" .. it.method .. "] " .. it.plain:gsub("\n", " "):sub(1, 120))
    end
  end
  table.insert(head, "")
  table.insert(head, clip(src, 120000))
  return table.concat(head, "\n")
end

-- ── layered deobfuscator (BIG plan §7/§9) ───────────────────────────
-- Peel one obfuscation layer at a time: detect the outermost encoding
-- (decimal escapes / base64 / hex / a native container / repeating- or
-- single-byte XOR / Luau bytecode), decode just that one layer, then repeat on
-- the output until it reads as source, stops changing, or can't be peeled.
-- Every decoder is bounded and reversible; nothing is executed.
local LAYER_MAX = 12
local LAYER_SAMPLE = 4096
local LAYER_MAX_BYTES = 3 * 1024 * 1024
local LUA_WORDS = { "function", "local ", "end", "then", "return", "for ", "while ", "if " }

local function looks_like_lua(s)
  if not is_printable(s) then return false end
  local hits = 0
  for _, w in ipairs(LUA_WORDS) do if s:find(w, 1, true) then hits = hits + 1 end end
  return hits >= 3
end
local function text_score(s)
  local n = math.min(#s, LAYER_SAMPLE)
  if n == 0 then return 0 end
  local printable = 0
  for i = 1, n do
    local c = string.byte(s, i)
    if c == 9 or c == 10 or c == 13 or (c >= 32 and c < 127) then printable = printable + 1 end
  end
  local score = printable / n
  for _, w in ipairs(LUA_WORDS) do if s:find(w, 1, true) then score = score + 0.15 end end
  return score
end
local function all_hex(s)
  if #s < 16 or (#s % 2) ~= 0 then return false end
  return s:find("[^0-9A-Fa-f]") == nil
end
local function hex_decode(s)
  local out, oi = {}, 0
  for i = 1, #s - 1, 2 do
    oi = oi + 1; out[oi] = string.char(tonumber(s:sub(i, i + 1), 16) or 0)
    if (oi % 8192) == 0 then analysis_checkpoint(1) end
  end
  return table.concat(out)
end
local function mostly_base64(s)
  local packed = s:gsub("%s", "")
  if #packed < 24 or (#packed % 4) ~= 0 then return false end
  local bad, n = 0, math.min(#packed, LAYER_SAMPLE)
  for i = 1, n do
    local c = packed:sub(i, i)
    if not (B64MAP[c] or c == "=") then bad = bad + 1 end
  end
  return (bad / n) < 0.02, packed
end
local function count_decimal_escapes(s)
  local n = 0
  for _ in s:gmatch("\\%d%d?%d?") do n = n + 1; if n > 64 then break end end
  return n
end
local function unfold_decimal_escapes(s)
  return (s:gsub("\\(%d%d?%d?)", function(d)
    local v = tonumber(d); if v and v <= 255 then return string.char(v) end
  end))
end
local function xor_apply(s, keybytes)
  local kn = #keybytes
  local out, oi = {}, 0
  for i = 1, #s do
    oi = oi + 1
    out[oi] = string.char(bit32.bxor(string.byte(s, i), keybytes[((i - 1) % kn) + 1]))
    if (oi % 8192) == 0 then analysis_checkpoint(1) end
  end
  return table.concat(out)
end
local function xor_single_best(s)
  local sample = s:sub(1, LAYER_SAMPLE)
  local best_key, best_score = nil, text_score(s)
  for k = 1, 255 do
    local buf, bi = {}, 0
    for i = 1, #sample do bi = bi + 1; buf[bi] = string.char(bit32.bxor(string.byte(sample, i), k)) end
    local sc = text_score(table.concat(buf))
    if sc > best_score then best_score = sc; best_key = k end
    if (k % 32) == 0 then analysis_checkpoint(1) end
  end
  return best_key
end
-- repeating-key XOR via known-plaintext crib: key = cipher[i] xor crib[i].
local function xor_repeating_crib(s)
  for _, crib in ipairs({ "local function", "return function", "function", "local ", "--[[" }) do
    local keylen = math.min(#crib, 16)
    local key = {}
    for i = 1, keylen do key[i] = bit32.bxor(string.byte(s, i) or 0, string.byte(crib, i) or 0) end
    if text_score(xor_apply(s:sub(1, LAYER_SAMPLE), key)) > 0.9 then return key end
    analysis_checkpoint(1)
  end
  return nil
end
-- One peel step: {name, method, out, confidence[, deferred]} or nil.
local function peel_once(s)
  if count_decimal_escapes(s) > 16 then
    local out = unfold_decimal_escapes(s)
    if out ~= s then return { name = "decimal_escape", method = "string.char", out = out, confidence = 0.8 } end
  end
  local isb64, packed = mostly_base64(s)
  if isb64 then
    local out = b64decode(packed)
    if out and #out > 0 then return { name = "base64", method = "b64decode", out = out, confidence = 0.85 } end
  end
  if all_hex(s) then
    local out = hex_decode(s)
    if #out > 0 then return { name = "hex", method = "hex_decode", out = out, confidence = 0.9 } end
  end
  local container = detect_container(s)
  if container == "lz4_frame" then
    local f = runtime_api("lz4decompress")
    if f then local ok, out = pcall(f, s); if ok and type(out) == "string" and #out > 0 then
      return { name = "lz4", method = "lz4decompress(native)", out = out, confidence = 0.9 } end end
    return { name = "lz4", method = "deferred_no_native", confidence = 0.5, deferred = true }
  elseif container == "zstd" then
    local f = runtime_api("zstddecompress")
    if f then local ok, out = pcall(f, s); if ok and type(out) == "string" and #out > 0 then
      return { name = "zstd", method = "zstddecompress(native)", out = out, confidence = 0.9 } end end
    return { name = "zstd", method = "deferred_no_native", confidence = 0.5, deferred = true }
  elseif container == "gzip" or container == "zlib" then
    return { name = container, method = "deferred_needs_inflate", confidence = 0.5, deferred = true }
  elseif container == "maybe_luau_bytecode" then
    return { name = "luau_bytecode", method = "disassemble_via_vm_extraction", confidence = 0.4, deferred = true }
  end
  -- Repeating-key XOR via crib: a crib-derived key always reproduces the crib
  -- in the first bytes, so printability alone gives false "decodes" that turn a
  -- readable string into garbage.  Attempt only when the buffer is not already
  -- Lua source, and accept ONLY when the whole decode reads as Lua (real XOR'd
  -- source yields many keywords; a false crib-decode yields just the crib's two
  -- plus garbage).
  if not looks_like_lua(s) then
    local rk = xor_repeating_crib(s)
    if rk then
      local out = xor_apply(s, rk)
      if looks_like_lua(out) then
        local hexk = {}; for i = 1, #rk do hexk[i] = string.format("%02x", rk[i]) end
        return { name = "xor_repeating", method = "crib_key=" .. table.concat(hexk), out = out, confidence = 0.85 }
      end
    end
  end
  -- Single-byte XOR: only on non-printable high-entropy binary; accept when it
  -- turns into printable text that scores better than the ciphertext.
  if not is_printable(s) and shannon_entropy(s) >= 6.0 then
    local k = xor_single_best(s)
    if k then
      local out = xor_apply(s, { k })
      if is_printable(out) and text_score(out) > text_score(s) then
        return { name = "xor_single", method = string.format("key=0x%02x", k), out = out, confidence = 0.7 }
      end
    end
  end
  return nil
end
local function layered_decode(data, deadline)
  data = tostring(data or "")
  if #data == 0 or #data > LAYER_MAX_BYTES then
    return data, { { name = "skipped", reason = #data == 0 and "empty" or "too_large" } }
  end
  local layers, cur = {}, data
  for iter = 1, LAYER_MAX do
    analysis_checkpoint(2)
    if deadline and os.clock() > deadline then layers[#layers + 1] = { name = "stopped", reason = "deadline" }; break end
    if looks_like_lua(cur) then layers[#layers + 1] = { name = "source_reached", bytes = #cur }; break end
    local step = peel_once(cur)
    if not step then break end
    local rec = { layer = iter, name = step.name, method = step.method, confidence = step.confidence,
      input_bytes = #cur, input_entropy = shannon_entropy(cur) }
    if step.deferred or not step.out or #step.out == 0 or step.out == cur then
      rec.output_bytes = step.out and #step.out or 0
      rec.deferred = step.deferred or false
      rec.stopped = step.deferred and "no_native_decoder" or "no_progress"
      layers[#layers + 1] = rec; break
    end
    rec.output_bytes = #step.out
    rec.output_entropy = shannon_entropy(step.out)
    layers[#layers + 1] = rec
    cur = step.out
  end
  return cur, layers
end

-- Classify a recovered constant-pool string.
local function classify_recovered_string(s)
  local low = s:lower()
  if low:find("https://", 1, true) or low:find("http://", 1, true)
    or low:find("discord.com/api/webhooks", 1, true) then return "url" end
  for _, w in ipairs({ "token", "secret", "password", "cookie", "hwid", "apikey",
    "api_key", "script_key", "webhook", "authorization", "bearer" }) do
    if low:find(w, 1, true) then return "secret_like" end
  end
  if #s >= 16 and (#s % 4) == 0 and s:find("^[A-Za-z0-9+/=]+$") then return "base64_candidate" end
  if #s >= 16 and (#s % 2) == 0 and s:find("^[0-9A-Fa-f]+$") then return "hex_candidate" end
  if s:find("^[%a_][%w_%.:]*$") then return "identifier" end
  if is_printable(s) then return "text" end
  return "binary"
end

-- Turn the harvested VM constant-pool strings into a deduped, classified,
-- decoded list.  base64/hex constants are peeled with the layered decoder so a
-- URL or key hidden one more layer down still surfaces.  secret-like values are
-- redacted (length + fingerprint only).
local function recover_vm_strings(vm_list, deadline)
  local out, seen = {}, {}
  for _, vm in ipairs(vm_list or {}) do
    for _, item in ipairs(vm.strings or {}) do
      if #out >= MAX_VM_STRINGS then break end
      local s = item.value
      if type(s) == "string" and #s >= 3 and not seen[s] then
        seen[s] = true
        local class = classify_recovered_string(s)
        local rec = { stage = vm.load_id, source = item.source, class = class, bytes = item.bytes }
        if class == "secret_like" then
          rec.preview = "<redacted:" .. #s .. "B fp=" .. fingerprint(s) .. ">"
        else
          rec.preview = s:sub(1, 200)
          if (class == "base64_candidate" or class == "hex_candidate") and deadline and os.clock() < deadline then
            local peeled, layers = layered_decode(s, deadline)
            if peeled ~= s and is_printable(peeled) then
              rec.decoded_class = classify_recovered_string(peeled)
              rec.decoded_preview = rec.decoded_class == "secret_like"
                and ("<redacted:" .. #peeled .. "B>") or peeled:sub(1, 200)
              local names = {}
              for _, l in ipairs(layers) do if l.layer and not l.deferred then names[#names + 1] = l.name end end
              rec.decode_chain = names
            end
          end
        end
        out[#out + 1] = rec
      end
    end
  end
  return out
end

local function run_deob(src)
  set_phase("deob", "Mining payload…")
  src = tostring(src or "")
  local scan_limit = effective_static_scan_bytes()
  local analysis_src = #src > scan_limit and src:sub(1, scan_limit) or src
  local detect = detect_obfuscator(analysis_src)
  S.detect = detect
  local body
  local has_hexblob = detect.container_family == "HexBlob-VM"
  if has_hexblob then
    -- Container recovery + payload translation needs the whole blob, not the
    -- truncated scan window (LZ4 back-references reach across the full stream).
    -- Bounded to a 90-second wall-clock budget per the tool's deob limit.
    local recovered
    local deadline = os.clock() + DEOB_BUDGET_SECS
    body, recovered = deob_hexvm(src, deadline)
    S.hexvm = recovered
    if recovered and recovered.decode_ok and recovered.length_match then
      if recovered.string_consts and recovered.string_consts > 0 then
        S.analysis_status.deobfuscation = recovered.translation_complete
          and "payload_translated_constants" or "payload_translated_partial"
        add_evidence("deob.hexvm", "high", "StaticAnalysis",
          string.format("payload translated: %d protos, %d string / %d number constants recovered (xor_key=%s, status=%s)",
            recovered.proto_count or 0, recovered.string_consts or 0,
            recovered.number_consts or 0, recovered.xor_key,
            tostring(recovered.translation_status)),
          0.95)
      else
        S.analysis_status.deobfuscation = "container_recovered_vm_pending"
        add_evidence("deob.hexvm", "high", "StaticAnalysis",
          string.format("blob container unwrapped: %d-byte VM program (xor_key=%s%s)",
            recovered.decompressed, recovered.xor_key,
            (recovered.derived_keys and #recovered.derived_keys > 0)
              and (", derived_keys=" .. table.concat(recovered.derived_keys, "/")) or ""),
          0.9)
      end
      if (recovered.full_deob_bytes or 0) > 0 then
        S.analysis_status.deobfuscation = recovered.translation_complete
          and "decoded_vm_ir_complete" or "decoded_vm_ir_partial"
        add_evidence("deob.luasnapper.full_ir", "high", "StaticAnalysis",
          string.format("full decoded VM IR exported: %d protos, %d instructions, %d bytes",
            recovered.proto_count or 0, recovered.instr_count or 0,
            recovered.full_deob_bytes or 0), 0.98)
      end
    end
  elseif detect.family == "00fuscator" then
    body = deob_00fuscator(analysis_src)
  else
    body = deob_generic(analysis_src, detect)
  end
  S.deob_out = body
  -- Layered peel: detect + decode each encoding layer (base64 / hex / decimal
  -- escapes / native container / XOR), independent of family, producing a
  -- transform chain and the best-peeled buffer.  Guarded so a decode failure
  -- never aborts the finalize/report path.
  local peel_ok, peeled, layers = pcall(layered_decode, src, os.clock() + 12)
  if not peel_ok then peeled, layers = src, {} end
  S.layers = layers
  S.layered_output = peeled
  local decoded_names = {}
  for _, l in ipairs(layers or {}) do
    if l.layer and not l.deferred then decoded_names[#decoded_names + 1] = l.name end
  end
  if #decoded_names > 0 then
    add_evidence("deob.layers", "medium", "StaticAnalysis",
      "layered peel: " .. table.concat(decoded_names, " -> ")
        .. (looks_like_lua(peeled) and " -> source" or ""), 0.8)
  end
  return detect, body
end

-- ── data provenance engine (BIG plan §3/§4) ─────────────────────────
-- Build an object-lineage graph over the stages Sentinel already captured.
-- Each captured load is an object linked to its parent; every object records
-- at least its captured state (type/bytes/entropy/hash/profile/markers/class).
-- When the HexBlob-VM path decoded the largest stage, its transformation chain
-- (source string -> XOR/LZ4 blob bytes -> decompressed VM program) is attached
-- as extra states, so the lineage ends at the decoded program with the
-- structural counts already recovered.  Runs at finalize only; bounded by the
-- same sampling window as fingerprint().
-- Native collision-resistant hash (sha256 via crypt.hash) over the same
-- bounded sample fingerprint() uses, for cross-run comparison (BIG plan §8).
-- Returns nil when the executor exposes no crypt.hash, so djb2 stays the
-- always-present identifier and this is purely additive.
local function strong_hash(s)
  local h = lib_resolve("crypt", "hash")
  if not h then return nil end
  local ok, digest = pcall(h, bounded_sample(s), "sha256")
  if ok and type(digest) == "string" and #digest > 0 then return "sha256:" .. digest end
  return nil
end

local function state_from_string(label, s)
  local entropy = shannon_entropy(s)
  local profile = byte_profile(s)
  return {
    label = label,
    type = "string",
    bytes = #s,
    entropy = entropy,
    hash = fingerprint(s),
    sha256 = strong_hash(s),
    profile = profile,
    markers = marker_scan(s),
    surface_class = surface_classify(entropy, profile.printable_ratio, #s),
    container = detect_container(s),
  }
end

-- Best-effort compiled-bytecode dump of a captured stage.  loadstring/load
-- COMPILES the source without running it; the bytecode dumper then serialises
-- that chunk.  No target code executes -- this is the same safety line
-- Sentinel.Analyze() holds.  Returns the bytecode string and the API used, or
-- nil when neither compilation nor a dumper is available.  Bounded by the
-- caller before anything is stored.
local function dump_stage_bytecode(source)
  local compile = runtime_api("loadstring") or runtime_api("load")
  if type(compile) ~= "function" then return nil end
  local ok, fn = pcall(compile, source, "@sentinel_bytecode_probe")
  if not ok or type(fn) ~= "function" then return nil end
  for _, name in ipairs({ "getfunctionbytecode", "dumpbytecode", "dumpstring" }) do
    local dumper = runtime_api(name)
    if type(dumper) == "function" then
      local dok, bc = pcall(dumper, fn)
      if dok and type(bc) == "string" and #bc > 0 then return bc, name end
    end
  end
  return nil
end

local function build_provenance()
  local r = S.runtime_capture
  local objects, edges = {}, {}
  local largest = S.largest_captured_stage or ""
  if r and r.loads then
    for _, item in ipairs(r.loads) do
      analysis_checkpoint(1)
      local obj = {
        id = item.id,
        parent = item.parent or 0,
        name = item.name,
        execution = item.execution,
        origin = "captured_load",
        captured_bytes = item.bytes,
        truncated = item.truncated,
        states = { state_from_string("captured_source", item.source or "") },
      }
      -- Compiled-bytecode probe of the largest stage.  Uses the executor's
      -- bytecode dumper (getfunctionbytecode / dumpbytecode / dumpstring) on a
      -- compile-only chunk -- no target code runs.  Records the bytecode's
      -- size/entropy/hash/container so the lineage can carry the compiled form,
      -- not just the source; skipped silently when no dumper is present.
      if item.source and item.source == largest and #largest > 0 then
        local bc, dumper = dump_stage_bytecode(item.source)
        if bc then
          local bc_entropy = shannon_entropy(bc)
          local bc_profile = byte_profile(bc)
          obj.states[#obj.states + 1] = {
            label = "compiled_bytecode", type = "bytes", bytes = #bc,
            dumper = dumper, entropy = bc_entropy, hash = fingerprint(bc),
            sha256 = strong_hash(bc),
            profile = bc_profile, container = detect_container(bc),
            surface_class = surface_classify(bc_entropy, bc_profile.printable_ratio, #bc),
            note = "compile-only dump of the captured source; no target code executed",
          }
        end
      end
      -- Attach the HexBlob-VM transformation chain to the stage whose captured
      -- source is the largest -- that is the blob the container recovery ran on.
      if S.hexvm and S.hexvm.decode_ok and item.source and item.source == largest then
        local hv = S.hexvm
        obj.states[#obj.states + 1] = {
          label = "xor_lz4_blob", type = "bytes", bytes = hv.blob_bytes,
          transform = "hex_decode + repeating_key_xor + lz4",
          xor_key = hv.xor_key, blob_hex_chars = hv.blob_hex_chars,
        }
        obj.states[#obj.states + 1] = {
          label = "vm_program", type = "bytes", bytes = hv.decompressed,
          entropy = hv.program_entropy,
          hash = "djb2-" .. string.format("%08x", hv.program_djb2 or 0),
          length_match = hv.length_match,
          surface_class = surface_classify(hv.program_entropy or 0, hv.program_printable_ratio or 0, hv.decompressed or 0),
          top_bytes = hv.top_bytes,
          structure = {
            proto_count = hv.proto_count, instr_count = hv.instr_count,
            string_consts = hv.string_consts, number_consts = hv.number_consts,
            translation_status = hv.translation_status,
          },
          classification = (hv.proto_count or hv.instr_count) and "probable_vm_program"
            or (hv.decompressed and hv.decompressed > 0 and "decoded_structured_data" or "undecoded"),
        }
      end
      objects[#objects + 1] = obj
      if (item.parent or 0) ~= 0 then
        edges[#edges + 1] = { from = item.parent, to = item.id }
      end
    end
  end
  -- Where does a controlled-test marker first appear along the lineage?
  local first_marker
  for _, obj in ipairs(objects) do
    for _, st in ipairs(obj.states) do
      if st.markers and #st.markers > 0 and not first_marker then
        first_marker = { object = obj.id, state = st.label, markers = st.markers }
      end
    end
  end
  return {
    schema = "sentinel.provenance.v1", version = VERSION,
    generated_utc = now(),
    note = "surface_class is entropy/printability only; not the VM-structure classifier.",
    object_count = #objects,
    objects = objects,
    edges = edges,
    controlled_test_marker_first_seen = first_marker,
    limits = { sample_bytes = MAX_FINGERPRINT_BYTES, entropy_units = "bits_per_byte_0_8" },
  }
end

    if S.path then append(S.path .. "/extracts/remotes.txt", line) end
  end
end

end

local _old_loadstring, _old_load, _old_print, _old_warn
local _old_request
local _old_game_methods = {}
local _old_schedulers = {}
local _wrapped_loadstring
local _hooks_on = false
local _native_hook_target, _native_hook_original
local _native_hooked = false
local _loader_active = false

-- Re-arming -- Attach, Run, SelfTest or Analyze after an earlier session --
-- must never adopt one of our own wrappers as the "original".  print is the
-- sharp edge: wrap_print resolves _old_print at call time out of a shared
-- local, so a second attach that captured the first wrapper leaves it
-- tail-calling itself.  Being a tail call it never grows the stack and never
-- errors; it just spins forever on the first print, pinning the client.  The
-- other hooks stack one redundant layer per attach, permanently taxing calls
-- the whole game makes.  keep() marks everything we install and
-- capture_original() refuses to adopt anything so marked.
local _installed = setmetatable({}, { __mode = "k" })
local function keep(fn)
  if type(fn) == "function" then _installed[fn] = true end
  return fn
end
local function capture_original(current, previous)
  if type(current) ~= "function" or _installed[current] then return previous end
  return current
end

-- Teardown for the full build's secondary hooks.  Restoration only ever put
-- loadstring back, so print, warn, GetService, __namecall, the genv proxy and
-- request all stayed live after Stop() -- exactly the overhead and
-- anti-tamper surface the mode gate exists to avoid.  Each thunk re-checks
-- that the live value is still ours, so a hook layered over ours by something
-- else is left alone.
local _full_restores = {}
local function restore_full_hooks()
  for i = #_full_restores, 1, -1 do
    pcall(_full_restores[i])
    _full_restores[i] = nil
  end
end
local function on_restore(fn) _full_restores[#_full_restores + 1] = fn end

local function loader_envs()
  return cached_envs
end

local function set_loader_binding(fn)
  for _, env in ipairs(loader_envs()) do
    pcall(function() rawset(env, "loadstring", fn) end)
  end
end

local function restore_loader_hook()
  restore_full_hooks()
  if not _wrapped_loadstring or not _old_loadstring then return end
  local restored_binding = _old_loadstring
  if _native_hooked and _native_hook_target then
    local restored = false
    local function attempt(fn, ...)
      local ok, result = pcall(fn, ...)
      return ok and result ~= false
    end
    if type(restorefunction) == "function" then
      restored = attempt(restorefunction, _native_hook_target)
    end
    if not restored and type(restoreclosure) == "function" then
      restored = attempt(restoreclosure, _native_hook_target)
    end
    if not restored and type(hookfunction) == "function" and _native_hook_original then
      restored = attempt(hookfunction, _native_hook_target, _native_hook_original)
    end
    if restored then restored_binding = _native_hook_target end
    _native_hooked = not restored
    S.restore_status = restored and "api_reported_success_unverified" or "failed"
  elseif S.restore_status == "not_attempted" then
    S.restore_status = "binding_only"
  end
  for _, env in ipairs(loader_envs()) do
    pcall(function()
      if rawget(env, "loadstring") == _wrapped_loadstring then
        rawset(env, "loadstring", restored_binding)
      end
    end)
  end
  _hooks_on = _native_hooked
  S.attached = _native_hooked
end

local function wrap_loader(original, label)
  local compiling = setmetatable({}, { __mode = "k" })
  return function(src, chunkname, ...)
    local thread = current_thread()
    if compiling[thread] then
      return nil, "Sentinel prevented recursive loadstring hook"
    end
    if not original then return nil, "no loader" end
    local context = target_threads[thread]
    local generation = S.generation
    local active = S.running and not S.force_stopped and not S._finalizing and S.runtime_capture
    local scoped = context and context.generation == generation
    local item
    if active and type(src) == "string" and (not S.runtime_capture.selected or scoped) then
      item = reserve_runtime_load(src, type(chunkname) == "string" and chunkname:sub(1, 120) or label,
        scoped and context.id or nil)
    end
    compiling[thread] = true
    local result = table.pack(pcall(original, src, chunkname, ...))
    compiling[thread] = nil
    if not result[1] then
      if item then item.execution = "compile_error" end
      error(result[2], 0)
    end
    local fn = result[2]
    if not item then return table.unpack(result, 2, result.n) end
    if type(fn) ~= "function" then item.execution = "compile_error" end
    local queued = enqueue_work(function()
      if generation ~= S.generation or S.force_stopped then return end
      item.processing = "processing"
      if #item.source > #S.largest_captured_stage then S.largest_captured_stage = item.source end
      S.graph[#S.graph + 1] = { id = item.id, parent = item.parent, name = item.name, bytes = item.bytes }
      record_runtime("loadstring", { load_id = item.id, parent_id = item.parent, bytes = item.bytes,
        fingerprint = fingerprint(item.source), source_omitted = true })
      set_phase("extracting", "Captured stage " .. item.id .. " · " .. item.bytes .. " bytes")
      if type(fn) == "function" then
        local constants = extract_constants(fn, item.id)
        if generation == S.generation and not S.force_stopped then
          S.runtime_capture.constants[#S.runtime_capture.constants + 1] = constants
          S.counters.const = S.counters.const + constants.count
        end
        -- Deep VM extraction: upvalues + nested prototypes + instruction-stream
        -- detection.  OFF by default (CAPTURE.vm_extraction) because it is heavy
        -- live work; always pcall-guarded so it can never break capture.
        local vm_ok, vm = false, nil
        if CAPTURE.vm_extraction then vm_ok, vm = pcall(extract_vm, fn, item.id) end
        if vm_ok and vm and generation == S.generation and not S.force_stopped and vm.status ~= "unsupported" then
          S.runtime_capture.vm[#S.runtime_capture.vm + 1] = vm
          if (vm.instruction_stream_count or 0) > 0 then
            local best = vm.instruction_streams[1]
            add_evidence("vm.instruction_stream", "high", "RuntimeObservation",
              string.format("stage %d: %d VM instruction stream(s) recovered — %s of %d entries, %d distinct opcodes (from %s)",
                item.id, vm.instruction_stream_count, best.shape, best.length,
                best.distinct_opcodes or 0, best.source), best.confidence or 0.8)
          end
        end
      end
      if generation == S.generation then item.processing = "complete" end
    end)
    item.processing = queued and "queued" or "queue_full"
    if type(fn) ~= "function" then return table.unpack(result, 2, result.n) end
    result[2] = runtime_function(fn, item, generation)
    return table.unpack(result, 2, result.n)
  end
end

local function install_hooks()
  if _hooks_on then
    if S.restore_status == "failed" then error("Previous hook restoration failed; restart the client before rearming", 0) end
    return
  end
  _native_hook_original = nil
  _hooks_on = true
  -- Reset here rather than in init_session, which runs after this: the map
  -- describes what this arming installed, and an early return above leaves
  -- the previous session's map in place because those hooks are still live.
  S.hooks = {}
  local loader_target
  pcall(function()
    loader_target = capture_original(loadstring, _old_loadstring)
    _old_loadstring = loader_target
  end)
  pcall(function() _old_load = capture_original(load, _old_load) end)
  pcall(function() _old_print = capture_original(print, _old_print) end)

  local cloned_loader
  if type(clonefunction) == "function" and type(loader_target) == "function" then
    pcall(function() cloned_loader = clonefunction(loader_target) end)
  end
  local original_for_this_hook = cloned_loader or loader_target
  local function call_original_loader(...)
    local fn = original_for_this_hook
    if type(fn) ~= "function" then return nil, "no original loader" end
    return fn(...)
  end

  local wrapped_ls = keep(wrap_loader(call_original_loader, "loadstring"))
  -- Resolved here, before the native hook can rebind _old_loadstring, but only
  -- wrapped in the full build below.
  local load_original = _old_load or _old_loadstring
  if type(newcclosure) == "function" then
    local ok, wrapped = pcall(newcclosure, wrapped_ls)
    if ok and type(wrapped) == "function" then wrapped_ls = keep(wrapped) end
  end
  _wrapped_loadstring = wrapped_ls

  S.hook_mode = "binding"
  if type(hookfunction) == "function" and type(loader_target) == "function" then
    local ok, prior = pcall(hookfunction, loader_target, wrapped_ls)
    if ok and type(prior) == "function" then
      _native_hook_target = loader_target
      _native_hook_original = cloned_loader or prior
      original_for_this_hook = _native_hook_original
      _old_loadstring = _native_hook_original
      _native_hooked = true
      S.hook_mode = cloned_loader and "native-scoped-clone" or "native-scoped"
    end
  end

  set_loader_binding(wrapped_ls)
  S.hooks.loadstring = S.hook_mode

  -- ── full build only ───────────────────────────────────────────────
  -- One gate for every secondary hook.  Each replaces a function the whole
  -- game calls, so each costs live-game time and hands the loader under test
  -- another identity change to fingerprint; lite therefore stops here with the
  -- loadstring wrapper armed and nothing else.  Returning rather than testing
  -- the mode inside each hook also means lite never allocates the load()
  -- wrapper, its weak compile table, or these pcall thunks.
  if LITE_MODE then return end

  -- Replacing a method on `game` by plain assignment is refused in most Roblox
  -- contexts, so prefer the executor's hookfunction and only fall back to
  -- assignment where it is actually permitted.  Records how each method was
  -- intercepted -- or that it could not be -- so the report never implies
  -- coverage the runtime did not give us.
  local function hook_method(host, name, label, make)
    local current
    if not pcall(function() current = host[name] end) then S.hooks[label] = "absent"; return false end
    local target = capture_original(current, _old_game_methods[label])
    if type(target) ~= "function" then S.hooks[label] = "absent"; return false end
    _old_game_methods[label] = target
    local call_original = target
    local replacement = keep(make(function(...) return call_original(...) end))
    -- hookfunction patches in place, so the method keeps its identity: nothing
    -- comparing it against a stored reference can tell it was replaced.
    if type(hookfunction) == "function" then
      local ok, prior = pcall(hookfunction, target, replacement)
      if ok and type(prior) == "function" then
        call_original = prior
        on_restore(function()
          if type(restorefunction) == "function" and pcall(restorefunction, target) then return end
          pcall(hookfunction, target, prior)
        end)
        S.hooks[label] = "hookfunction"
        return true
      end
    end
    -- Assignment only sticks on a plain table; a method on an Instance or an
    -- RBXScriptSignal will refuse it, which is why the result is checked.
    local assigned = pcall(function() host[name] = replacement end)
    local landed = false
    pcall(function() landed = host[name] == replacement end)
    if assigned and landed then
      on_restore(function()
        local still = false
        pcall(function() still = host[name] == replacement end)
        if still then pcall(function() host[name] = target end) end
      end)
      S.hooks[label] = "assignment"
      return true
    end
    S.hooks[label] = "unavailable"
    return false
  end
  local function hook_game_method(name, make) return hook_method(game, name, name, make) end

  if CAPTURE.loads then
    local wrapped_load = keep(wrap_loader(load_original, "load"))

    local function set_all(name, fn)
      for _, env in ipairs(cached_envs) do pcall(rawset, env, name, fn) end
      pcall(function() _G[name] = fn end)
      pcall(function() if getgenv then getgenv()[name] = fn end end)
      pcall(function()
        if getrenv then
          local r = getrenv()
          if r then r[name] = fn end
        end
      end)
    end
    set_all("load", wrapped_load)
    S.hooks.load = "binding"
    if _old_load then
      on_restore(function()
        if rawget(_G, "load") == wrapped_load then set_all("load", _old_load) end
      end)
    end
  end

  if CAPTURE.tasks then
    -- A loader that fires its key check from task.spawn used to fall out of
    -- the capture scope: its prints, remotes and nested loads went
    -- unattributed and Sentinel.Bind was the manual fix.  Wrapping the thread
    -- schedulers carries the scope across for it.  bind_if_target decides at
    -- call time, so a task the game schedules is passed straight through with
    -- one in_scope() and no wrapper allocated.
    local function bind_first(original)
      return function(fn, ...)
        if type(fn) ~= "function" then return original(fn, ...) end
        return original(bind_if_target(fn), ...)
      end
    end
    local function bind_second(original)
      return function(first, fn, ...)
        if type(fn) ~= "function" then return original(first, fn, ...) end
        return original(first, bind_if_target(fn), ...)
      end
    end

    -- Replace a field on a library table, unlocking it only if the executor
    -- says it is read-only and can say so.
    local function hook_table_field(host, name, make, label)
      if type(host) ~= "table" then S.hooks[label] = "absent"; return false end
      local original = capture_original(host[name], _old_schedulers[label])
      if type(original) ~= "function" then S.hooks[label] = "absent"; return false end
      _old_schedulers[label] = original
      local replacement = keep(make(original))
      local locked = type(isreadonly) == "function" and isreadonly(host) or false
      local function unlock() if locked and type(setreadonly) == "function" then pcall(setreadonly, host, false) end end
      local function relock() if locked and type(setreadonly) == "function" then pcall(setreadonly, host, true) end end
      unlock()
      local assigned = pcall(function() host[name] = replacement end)
      relock()
      if assigned and host[name] == replacement then
        S.hooks[label] = locked and "table_unlocked" or "table"
        on_restore(function()
          if host[name] ~= replacement then return end
          unlock()
          pcall(function() host[name] = original end)
          relock()
        end)
        return true
      end
      S.hooks[label] = "unavailable"
      return false
    end

    -- Deprecated globals live in the script environments, not a library table.
    local function hook_global(name, make, label)
      local original
      for _, env in ipairs(cached_envs) do
        local candidate = rawget(env, name)
        if type(candidate) == "function" then original = candidate break end
      end
      original = capture_original(original, _old_schedulers[label])
      if type(original) ~= "function" then S.hooks[label] = "absent"; return false end
      _old_schedulers[label] = original
      local replacement = keep(make(original))
      for _, env in ipairs(cached_envs) do pcall(rawset, env, name, replacement) end
      pcall(function() if getgenv then getgenv()[name] = replacement end end)
      S.hooks[label] = "binding"
      on_restore(function()
        for _, env in ipairs(cached_envs) do
          if rawget(env, name) == replacement then pcall(rawset, env, name, original) end
        end
        pcall(function()
          if getgenv and getgenv()[name] == replacement then getgenv()[name] = original end
        end)
      end)
      return true
    end

    -- task.spawn/defer also accept a thread as their first argument; bind_first
    -- only touches the function form and passes a thread through untouched.
    if type(task) == "table" then
      hook_table_field(task, "spawn", bind_first, "task.spawn")
      hook_table_field(task, "defer", bind_first, "task.defer")
      hook_table_field(task, "delay", bind_second, "task.delay")
    end
    if type(coroutine) == "table" then
      hook_table_field(coroutine, "wrap", bind_first, "coroutine.wrap")
      hook_table_field(coroutine, "create", bind_first, "coroutine.create")
    end
    hook_global("spawn", bind_first, "spawn")
    hook_global("delay", bind_second, "delay")
  end
  if CAPTURE.connections then
    -- Every RBXScriptSignal shares one set of methods, so intercepting Connect
    -- once reaches every signal in the place.  The decisive detail is when the
    -- binding happens: at connect time, not fire time.  When the callback
    -- finally runs the engine is the caller and the target is nowhere on the
    -- stack, so the only moment it can be attributed is the moment the target
    -- registers it -- which is also the moment in_scope() can still tell.
    local sample = RunService.Heartbeat
    local function bind_callback(call_original)
      return function(self, callback, ...)
        if type(callback) ~= "function" then return call_original(self, callback, ...) end
        return call_original(self, bind_if_target(callback), ...)
      end
    end
    local reached = false
    for _, name in ipairs({ "Connect", "Once", "ConnectParallel", "connect" }) do
      local present = false
      pcall(function() present = type(sample[name]) == "function" end)
      if present and hook_method(sample, name, "signal." .. name, bind_callback) then
        reached = true
      end
    end
    if not reached then
      add_evidence("capture.connect_unavailable", "info", "RuntimeObservation",
        "signal Connect could not be intercepted; this executor has no working hookfunction, "
          .. "so callbacks the target connects need Sentinel.Bind", 1)
    end
  end

  if CAPTURE.prints then
    pcall(function()
      local function wrap_print(...)
        if in_scope() then push_print(...) end
        if _old_print then return _old_print(...) end
      end
      local installed = keep(newcclosure and newcclosure(wrap_print) or wrap_print)
      print = installed
      S.hooks.print = "binding"
      on_restore(function() if print == installed then print = _old_print end end)
    end)

    pcall(function()
      _old_warn = capture_original(warn, _old_warn)
      local function wrap_warn(...)
        if in_scope() then push_print("[WARN]", ...) end
        if _old_warn then return _old_warn(...) end
      end
      local installed = keep(newcclosure and newcclosure(wrap_warn) or wrap_warn)
      warn = installed
      S.hooks.warn = "binding"
      on_restore(function() if warn == installed then warn = _old_warn end end)
    end)
  end

  if CAPTURE.services then
    pcall(hook_game_method, "GetService", function(call_original)
      return function(self, name)
        if in_scope() then
          S.counters.svc = S.counters.svc + 1
          if S.path then append(S.path .. "/extracts/services.txt", string.format("[%s] GetService %s\n", now(), tostring(name))) end
          if S.phase == "extracting" then set_phase("tools", "Service: " .. tostring(name)) end
        end
        return call_original(self, name)
      end
    end)
  end

  -- proxy env writes
  if CAPTURE.globals then
    pcall(function()
      local genv = getgenv and getgenv()
      if not genv then return end
      local mt = getrawmetatable(genv)
      if not mt then return end
      local old_new = mt.__newindex
      if _installed[old_new] then return end
      local setro = setreadonly or function() end
      pcall(setro, mt, false)
      local installed = keep(function(t, k, v)
        if in_scope() then
          append(S.path and (S.path .. "/extracts/constants.txt") or "constants.txt",
            string.format("[%s] genv.%s = %s\n", now(), tostring(k), clip(tostring(v), 200)))
        end
        if old_new then return old_new(t, k, v) end
        rawset(t, k, v)
      end)
      mt.__newindex = installed
      pcall(setro, mt, true)
      S.hooks.global_writes = "metatable"
      on_restore(function()
        if rawget(mt, "__newindex") ~= installed then return end
        pcall(setro, mt, false)
        mt.__newindex = old_new
        pcall(setro, mt, true)
      end)
    end)
  end

  if CAPTURE.http then
    -- Where a whitelist loader actually fetches.  Hooking the two methods
    -- directly costs nothing between calls; reading them out of __namecall
    -- instead would mean resolving a method name on every method call in the
    -- place, which is the bill CAPTURE.remotes pays.  HTTP is deliberately
    -- not scope-filtered: a loader that fires its key check from a spawned
    -- task is exactly the case worth catching, and these fire rarely enough
    -- that recording all of them is cheap.
    local reached = false
    for _, method_name in ipairs({ "HttpGet", "HttpGetAsync" }) do
      local ok = hook_game_method(method_name, function(call_original)
        return function(self, url, ...)
          if S.running and not S.force_stopped then
            push_http(tostring(url or ""), method_name, "", 0)
          end
          return call_original(self, url, ...)
        end
      end)
      reached = reached or ok
    end
    if not reached then
      add_evidence("capture.httpget_unavailable", "info", "RuntimeObservation",
        "game:HttpGet could not be intercepted on this executor; enable CAPTURE.remotes "
          .. "for __namecall coverage or rely on the static endpoint inventory", 1)
    end

    pcall(function()
      -- Replacing executor request functions changes function identity and can
      -- trip loader anti-tamper checks.  It earns that risk because the URL
      -- and status of the key check are the whole point of a whitelist test.
      local raw = capture_original(get_request(), _old_request)
      if not raw then S.hooks.request = "absent"; return end
      _old_request = raw
      local genv = getgenv and getgenv()
      local previous = genv and rawget(genv, "request") or nil
      local function wrap_req(opts)
        opts = opts or {}
        local url = tostring(opts.Url or opts.url or "")
        if RELAY_HTTP ~= "" and url:find(RELAY_HTTP, 1, true) then return raw(opts) end
        local res = raw(opts)
        local body, status = "", 0
        if type(res) == "table" then
          body = tostring(res.Body or res.body or "")
          status = tonumber(res.StatusCode or res.status_code or 0) or 0
        end
        push_http(url, opts.Method or opts.method or "GET", body, status)
        return res
      end
      local installed = keep(wrap_req)
      if genv then
        pcall(function() genv.request = installed end)
        S.hooks.request = "genv_binding"
        on_restore(function()
          if rawget(genv, "request") == installed then genv.request = previous end
        end)
      end
    end)
  end

  if CAPTURE.remotes then
    pcall(function()
      -- The only hook that runs on every method call in the place.  in_scope()
      -- must be the first thing it does: resolving the method name is a C call,
      -- and paying that game-wide is exactly what turns a whitelist test into a
      -- whole-game trace.
      if not getrawmetatable or not getnamecallmethod then return end
      local mt = getrawmetatable(game)
      if not mt or type(mt.__namecall) ~= "function" then return end
      local old = mt.__namecall
      if _installed[old] then return end
      local setro = setreadonly or function() end
      pcall(setro, mt, false)
      -- HttpGet is recorded here only when the direct method hook could not be
      -- installed.  With both live the same call would be counted twice.
      local http_needs_namecall = not (S.hooks.HttpGet == "hookfunction" or S.hooks.HttpGet == "assignment")
      local function hooked(self, ...)
        if not in_scope() then return old(self, ...) end
        local resolved, raw_method = pcall(getnamecallmethod)
        local method = resolved and tostring(raw_method) or ""

        -- Snapshot what we need, but record nothing yet.  The engine's real
        -- __namecall reads L->namecall to decide which method to dispatch, and
        -- the VM sets that once, at the call site.  Every namecall Sentinel
        -- makes in the meantime overwrites it -- and the recording path makes
        -- plenty, because set_phase touches the GUI.  Recording before
        -- delegating therefore hands `old` the name of whatever Sentinel
        -- happened to call last, and the engine runs the wrong method.
        -- Reads below are __index and __tostring only, never namecalls.
        local remote_brief, http_url
        if method == "FireServer" or method == "InvokeServer" then
          local n = select("#", ...)
          local brief = {}
          for i = 1, math.min(n, 6) do
            local a = select(i, ...)
            local t = typeof(a)
            if t == "string" then brief[i] = "str:" .. clip(a, 80)
            elseif t == "Instance" then brief[i] = "Inst:" .. a.ClassName .. ":" .. a.Name
            else brief[i] = t .. ":" .. clip(tostring(a), 60) end
          end
          remote_brief = { self and self.Name or "?", fullPath(self), table.concat(brief, ", ") }
        elseif http_needs_namecall and (method == "HttpGet" or method == "HttpGetAsync") then
          http_url = tostring(select(1, ...) or "")
        end

        if not (remote_brief or http_url) then return old(self, ...) end

        local results = table.pack(old(self, ...))
        if remote_brief then
          push_remote(method, remote_brief[1], remote_brief[2], remote_brief[3])
        else
          push_http(http_url, method, "", 0)
        end
        return table.unpack(results, 1, results.n)
      end
      local installed = keep(newcclosure and newcclosure(hooked) or hooked)
      mt.__namecall = installed
      pcall(setro, mt, true)
      S.hooks.namecall = "metatable"
      on_restore(function()
        if rawget(mt, "__namecall") ~= installed then return end
        pcall(setro, mt, false)
        mt.__namecall = old
        pcall(setro, mt, true)
      end)
    end)
  end
end

-- ── watchdog + snapshot ─────────────────────────────────────────────
local finalize -- forward

local function start_watchdog()
  if S._watch then S._watch:Disconnect() end
  local watch_acc = 0
  S._watch = RunService.Heartbeat:Connect(function(dt)
    if not S.running then return end
    watch_acc = watch_acc + (tonumber(dt) or 0.016)
    if watch_acc < 0.25 then return end
    watch_acc = 0
    local elapsed = S.capture_started and (os.clock() - S.capture_started) or 0
    local window_done = S.capture_started and COLLECTION_SECS > 0 and elapsed >= COLLECTION_SECS
    -- Idle finish: a target was captured and nothing new has come in for
    -- IDLE_FINISH_SECS.  This is what stops the "finished running but still
    -- sitting on the same phase" hang -- Sentinel wraps up shortly after the
    -- target goes quiet instead of waiting out the full window.
    local idle_done = S.target_captured and IDLE_FINISH_SECS > 0
      and S.last_activity and (os.clock() - S.last_activity) >= IDLE_FINISH_SECS
    if window_done or idle_done then
      if S._watch then S._watch:Disconnect(); S._watch = nil end
      task.spawn(function()
        set_phase("tools", idle_done
          and ("idle " .. IDLE_FINISH_SECS .. "s after last capture — finalizing…")
          or "collection window complete — finalizing…")
        finalize()
      end)
    end
  end)
end

local function start_snapshot()
  if S._snap then S._snap:Disconnect() end
  -- One connection does both jobs.  The environment sampling used to be
  -- lite-only, so the full build silently lost it; the live-status write used
  -- to re-serialise final_blob.txt every four seconds whether or not it had
  -- changed, which on a megabyte payload is a megabyte of writefile every four
  -- seconds for the length of the session.
  local sampled, pending, cursor = 0, false, 0
  local status_acc, written_blob, last_status_sig = 0, -1, nil
  S._snap = RunService.Heartbeat:Connect(function(dt)
    if not S.running or S.force_stopped or S._finalizing then return end
    dt = tonumber(dt) or 0.016
    sampled = sampled + dt
    status_acc = status_acc + dt

    if sampled >= 1 and not pending and S.runtime_envs and #S.runtime_envs > 0 then
      sampled = 0
      cursor = cursor % #S.runtime_envs + 1
      local entry, generation = S.runtime_envs[cursor], S.generation
      pending = enqueue_work(function()
        local ok = pcall(record_environment, entry.id, "periodic", entry.env, generation)
        pending = false
        if not ok then S.queue_dropped = S.queue_dropped + 1 end
      end)
    end

    if status_acc >= 4 and S.path then
      status_acc = 0
      if S.largest_captured_stage ~= "" and #S.largest_captured_stage ~= written_blob then
        written_blob = #S.largest_captured_stage
        write(S.path .. "/largest_captured_stage.txt", S.largest_captured_stage)
      end
      -- Only rewrite live_status when something actually moved.  Rewriting it
      -- every 4s regardless kept inflating write_bytes (the KB stat) and made
      -- an idle session look like it was still doing work when it was not.
      local c = S.counters
      local sig = string.format("%s|%d|%d|%d|%d|%d|%d",
        S.phase, #S.largest_captured_stage, c.load, c.http, c.remote, c.print, #S.evidence)
      if sig ~= last_status_sig then
        last_status_sig = sig
        write(S.path .. "/extracts/live_status.txt", safe_json({
          t = now(),
          phase = S.phase,
          counters = S.counters,
          blob = #S.largest_captured_stage,
          time_left = S.capture_started and math.max(0, COLLECTION_SECS - (os.clock() - S.capture_started)) or COLLECTION_SECS,
        }))
      end
    end
  end)
end

-- ── session ─────────────────────────────────────────────────────────
local function init_session()
  reset_runtime_capture()
  table.clear(work_queue)
  S.queue_dropped = 0
  S.capture_pending = false
  S.timings = {}
  if not has_fs() then warn("[Sentinel] no writefile/makefolder — memory-only") end
  S.fs_error = nil
  S.fs_warned = false
  local id, path = next_script_id()
  S.session = id
  S.path = path
  S.t0 = os.clock()
  S.last_activity = os.clock()
  S.capture_started = nil
  S.target_captured = false
  S.force_stopped = false
  S.snapshot_count = 0
  analysis_ticks = 0
  S.write_bytes = 0
  S.write_limit_hit = false
  S._finalizing = false
  S.running = true
  S.largest_captured_stage = ""
  S.deob_out = ""
  S.full_deob = ""
  S.detect = nil
  S.hexvm = nil
  S.provenance = nil
  S.layers = nil
  S.layered_output = nil
  S.static = {
    strings = {}, urls = {}, keys = {}, tables = 0, concats = {}, cff = 0, density = 0,
    constant_inventory = { strings = {}, numbers = {}, summary = {} }, source_structure = {},
  }
  S.evidence = {}
  S.evidence_seen = {}
  S.runtime_events = {}
  S.analysis_status = {
    deobfuscation = "partial_surface_only",
    runtime_validation = "unverified",
    behavioral_equivalence = "not_tested",
  }
  for k in pairs(S.counters) do S.counters[k] = 0 end
  table.clear(S.graph)
  table.clear(S.https)

  ensure_dir(ROOT)
  ensure_dir(path)
  ensure_dir(path .. "/extracts")

  write("Sentinel_LATEST.txt", table.concat({
    "Sentinel Local " .. VERSION,
    "status=collecting",
    "session=" .. tostring(id),
    "output=" .. tostring(path) .. "/",
    "hook_mode=" .. tostring(S.hook_mode),
    "Open this path in your executor's file workspace after pressing SAVE.",
  }, "\n"))

  local env = snapshot_env()
  write(path .. "/info.txt", table.concat({
    "Sentinel Local " .. VERSION,
    "Mode: " .. (LITE_MODE and "whitelist-lite" or "full")
      .. (SCOPED_CAPTURE and " · target-scoped" or " · global")
      .. (CAPTURE.remotes and " · +remotes" or ""),
    "Capture scope: selected function, nested loads, threads it starts and callbacks it connects",
    "Session: " .. id,
    "Started: " .. now() .. " UTC",
    "PlaceId: " .. tostring(game.PlaceId),
    "Player: " .. tostring(env.player),
    "Executor: " .. tostring(env.executor),
    "Hook mode: " .. tostring(S.hook_mode),
    "Collection window: " .. tostring(COLLECTION_SECS) .. "s",
    "Write limit: " .. tostring(MAX_LOCAL_WRITE_BYTES) .. " bytes",
  }, "\n"))
  write(path .. "/env_log.txt", safe_json(env))
  write(path .. "/environment_surface.txt", safe_json({
    executor = env.executor,
    runtime = env.runtime,
    touch_enabled = env.touch_enabled,
    keyboard_enabled = env.keyboard_enabled,
    viewport = env.viewport,
    api_surface = env.api_surface,
    key_context = env.key_context,
    privacy = env.privacy,
  }))
  -- No placeholder pre-creation.  Every report file is now written only when
  -- there is real content for it (at finalize, or by the sink/appender that
  -- captured it), so an empty session leaves an empty folder instead of a pile
  -- of "-- pending --" stubs.  tool_diagnostics is the one exception: it
  -- describes the environment and active hooks, which is real from the start.
  write(path .. "/tool_diagnostics.txt", safe_json(tool_diagnostics()))

  GUI.build()
  GUI.start_anim()
  if S.fs_error then
    set_phase("idle", "FILE ERROR: " .. tostring(S.fs_error):sub(1, 80))
  else
    set_phase("idle", "Ready — saves to " .. tostring(path) .. "/")
  end
  start_watchdog()
  start_snapshot()
  print(("[Sentinel] session %s → %s/  collection=%ds  cap=5MB"):format(id, path, COLLECTION_SECS))
  return id, path
end

finalize = function()
  if S._finalizing or (not S.running and S.phase == "done") then return S.path end
  S._finalizing = true
  restore_loader_hook()
  while (#work_queue > 0 or job_active) and not S.force_stopped do
    drain_work()
    if #work_queue > 0 or job_active then task.wait() end
  end
  if S.force_stopped then S._finalizing = false; return S.path end
  if S._watch then S._watch:Disconnect(); S._watch = nil end
  if S._snap then S._snap:Disconnect(); S._snap = nil end
  set_phase("deob", "Mining payload…")
  S.running = false
  for _, item in ipairs(S.runtime_capture.loads) do
    if #item.source > #S.largest_captured_stage then S.largest_captured_stage = item.source end
  end
  save_runtime_capture()
  S.runtime_envs = {}

  local capture_failed = not S.target_captured
  if capture_failed then
    S.analysis_status.runtime_validation = "capture_failed"
    add_evidence("capture.no_payload", "high", "RuntimeObservation",
      "no target payload reached Sentinel; hook=" .. tostring(S.hook_mode), 1)
    if S.path then
      write(S.path .. "/capture_failure.txt", table.concat({
        "Sentinel target capture failed",
        "hook_mode=" .. tostring(S.hook_mode),
        "filesystem_error=" .. tostring(S.fs_error or "none"),
        "Try a fresh Roblox join before loading this Sentinel version.",
      }, "\n"))
    end
  end
  if S.path then
    local saved_blob = S.largest_captured_stage
    if S.device and S.device.mobile and #saved_blob > MAX_MOBILE_BLOB_WRITE_BYTES then
      saved_blob = saved_blob:sub(1, MAX_MOBILE_BLOB_WRITE_BYTES)
      add_evidence("output.mobile_truncated", "info", "RuntimeObservation",
        "raw output capped at " .. tostring(MAX_MOBILE_BLOB_WRITE_BYTES) .. " bytes for Android stability", 1)
    end
    -- Only emit the captured-stage files when something was actually captured.
    if saved_blob ~= "" then
      write(S.path .. "/largest_captured_stage.txt", saved_blob)
      -- Compatibility alias for one release: tooling that reads the old name
      -- still finds the finalized artifact.  "largest_captured_stage" is the
      -- accurate name -- the biggest stage Sentinel captured, not a proven
      -- final protected program.
      write(S.path .. "/final_blob.txt", saved_blob)
    end
    -- Load graph: only when wrap_loader actually recorded edges.
    if #S.graph > 0 then
      local gl = { "-- load graph --" }
      for _, e in ipairs(S.graph) do
        table.insert(gl, string.format("#%d <- %s | %s | %dB", e.id, tostring(e.parent or "-"), e.name, e.bytes))
      end
      write(S.path .. "/extracts/load_graph.txt", table.concat(gl, "\n"))
    end
    if LITE_MODE and #S.https > 0 then
      -- The incremental HTTP sinks are full-build only; in lite this is the
      -- only chance to emit whatever was recorded directly.
      write(S.path .. "/extracts/http.txt", "-- lite HTTP metadata --\n" .. table.concat(S.https, ""))
    end
  end
  local analysis_ok, analysis_err = pcall(static_analyze, S.largest_captured_stage)
  if S.force_stopped then
    S._finalizing = false
    return S.path
  end
  if not analysis_ok then
    add_evidence("analysis.error", "high", "StaticAnalysis",
      "bounded analyzer stopped: " .. tostring(analysis_err):sub(1, 220), 1)
  end
  local detect, deob_body = run_deob(S.largest_captured_stage)
  -- Build the lineage graph before the security report so the report can carry
  -- its summary; the full graph goes to provenance.txt.
  local prov_ok, prov = pcall(build_provenance)
  S.provenance = prov_ok and prov or nil
  local security_report = build_security_report(detect)
  -- prints/http/remotes/services are fed only by hooks lite does not install,
  -- so in lite they are structurally zero.  Say "off" rather than report a
  -- zero a reader would take as "looked and found none".
  local sink_off = LITE_MODE and "off(lite)" or nil
  if S.path then
    -- Environment/diagnostic reports: always meaningful (they describe the
    -- runtime and what was and wasn't observed), so they are written even when
    -- nothing was captured.
    write(S.path .. "/tool_diagnostics.txt", safe_json(tool_diagnostics()))
    write(S.path .. "/capability_matrix.txt", safe_json(capability_matrix()))
    write(S.path .. "/observations.txt", safe_json(observation_report()))
    -- Everything below is written only when it actually has captured content.
    if prov_ok and prov and (prov.object_count or 0) > 0 then
      write(S.path .. "/provenance.txt", safe_json(prov))
    end
    local have_deob = type(deob_body) == "string" and #deob_body > 0 and #S.largest_captured_stage > 0
    if have_deob then
      write(S.path .. "/deobfuscated_payload.txt", deob_body)
      write(S.path .. "/analysis_surface.txt", deob_body)
    end
    if type(S.full_deob) == "string" and #S.full_deob > 0 then
      write(S.path .. "/full_deob.txt", S.full_deob)
    end
    if S.target_captured or #S.evidence > 0 then
      write(S.path .. "/security_findings.txt", safe_json(security_report))
    end
    if #S.runtime_events > 0 then
      write(S.path .. "/runtime_events.txt", safe_json(S.runtime_events))
    end
    -- VM instruction extraction: emit the recovered program only when the deep
    -- walk actually found nested prototypes or an instruction stream.
    do
      local vm = S.runtime_capture.vm or {}
      local total_streams, walked_protos, total_strings = 0, 0, 0
      for _, e in ipairs(vm) do
        total_streams = total_streams + (e.instruction_stream_count or 0)
        walked_protos = walked_protos + ((e.proto_count or 1) - 1)
        total_strings = total_strings + (e.string_count or 0)
      end
      -- Recover the VM's constant-pool strings (URLs / keys / names / messages).
      local rec_ok, recovered = pcall(recover_vm_strings, vm, os.clock() + 10)
      if not rec_ok or type(recovered) ~= "table" then recovered = {} end
      if total_streams > 0 or walked_protos > 0 or #recovered > 0 then
        write(S.path .. "/vm_instructions.txt", safe_json({
          schema = "sentinel.vm-instructions.v1", version = VERSION,
          generated_utc = now(),
          note = "opcodes/records are a bounded sample of the decoded VM program; not recovered source.",
          stage_count = #vm, total_instruction_streams = total_streams,
          total_pool_strings = total_strings, recovered_strings = recovered,
          stages = vm,
        }))
        -- URL / secret hits in the recovered pool are worth flagging.
        local urls, secrets = 0, 0
        for _, r in ipairs(recovered) do
          if r.class == "url" or r.decoded_class == "url" then urls = urls + 1 end
          if r.class == "secret_like" or r.decoded_class == "secret_like" then secrets = secrets + 1 end
        end
        if urls > 0 or secrets > 0 then
          add_evidence("vm.recovered_strings", "high", "RuntimeObservation",
            string.format("VM constant pool: %d strings recovered (%d URLs, %d secret-like)",
              #recovered, urls, secrets), 0.85)
        end
        -- Human-readable recovered strings.
        if #recovered > 0 then
          local sl = { "===== SENTINEL VM RECOVERED STRINGS =====",
            "pool_strings=" .. total_strings .. "  listed=" .. #recovered, "" }
          for _, r in ipairs(recovered) do
            sl[#sl + 1] = string.format("[%s] stage %s · %s · %dB", r.class,
              tostring(r.stage), r.source, r.bytes or 0)
            sl[#sl + 1] = "  " .. tostring(r.preview)
            if r.decoded_preview then
              sl[#sl + 1] = "  -> (" .. table.concat(r.decode_chain or {}, ">") .. " => "
                .. tostring(r.decoded_class) .. ") " .. tostring(r.decoded_preview)
            end
          end
          write(S.path .. "/extracts/vm_strings.txt", table.concat(sl, "\n"))
        end
        -- Human-readable view of the decode instructions.
        local lines = { "===== SENTINEL VM INSTRUCTIONS =====",
          "streams=" .. total_streams .. "  nested_protos=" .. walked_protos, "" }
        for _, e in ipairs(vm) do
          for _, s in ipairs(e.instruction_streams or {}) do
            lines[#lines + 1] = string.format(
              "stage %s · %s · proto %s · %s · length=%d sampled=%d distinct_opcodes=%d conf=%.2f",
              tostring(e.load_id), s.source, tostring(s.proto_id), s.shape,
              s.length, s.sampled, s.distinct_opcodes or 0, s.confidence or 0)
            local head = {}
            for k = 1, math.min(48, #s.sample) do
              local v = s.sample[k]
              head[#head + 1] = type(v) == "table" and ("{" .. table.concat(v, ",") .. "}") or tostring(v)
            end
            lines[#lines + 1] = "  opcodes: " .. table.concat(head, " ")
            local top = {}
            for op, c in pairs(s.opcode_histogram or {}) do top[#top + 1] = { op = op, c = c } end
            table.sort(top, function(a, b) return a.c > b.c end)
            local th = {}
            for k = 1, math.min(12, #top) do th[#th + 1] = string.format("%s:%d", tostring(top[k].op), top[k].c) end
            lines[#lines + 1] = "  top_opcodes: " .. table.concat(th, "  ")
            lines[#lines + 1] = ""
          end
        end
        write(S.path .. "/extracts/vm_instructions.txt", table.concat(lines, "\n"))
      end
    end
    -- Layered deobfuscation chain: only when at least one layer was decoded.
    do
      local layers = S.layers or {}
      local decoded = 0
      for _, l in ipairs(layers) do if l.layer and not l.deferred then decoded = decoded + 1 end end
      if decoded > 0 then
        write(S.path .. "/layers.txt", safe_json({
          schema = "sentinel.layers.v1", version = VERSION, generated_utc = now(),
          decoded_layers = decoded, reached_source = looks_like_lua(S.layered_output or ""),
          chain = layers,
        }))
        local ll = { "===== SENTINEL DEOBFUSCATION LAYERS =====" }
        for _, l in ipairs(layers) do
          if l.layer then
            ll[#ll + 1] = string.format("layer %d: %-14s %-28s %d -> %dB  ent %.2f->%.2f%s",
              l.layer, l.name, tostring(l.method), l.input_bytes or 0, l.output_bytes or 0,
              l.input_entropy or 0, l.output_entropy or 0, l.deferred and "  [deferred]" or "")
          else
            ll[#ll + 1] = "-- " .. tostring(l.name) .. (l.reason and (": " .. l.reason) or "")
          end
        end
        write(S.path .. "/extracts/layers.txt", table.concat(ll, "\n"))
        -- The peeled buffer, only if it actually became readable source.
        if looks_like_lua(S.layered_output or "") then
          write(S.path .. "/layered_output.txt", S.layered_output)
        end
      end
    end
    local evidence_lines = {
      "===== SENTINEL EVIDENCE REPORT =====",
      "Verdict: " .. tostring(security_report.verdict),
      "Family: " .. tostring(detect.family) .. " confidence=" .. tostring(detect.confidence),
      "Deobfuscation: " .. tostring(S.analysis_status.deobfuscation),
      "Runtime validation: " .. tostring(S.analysis_status.runtime_validation),
      "Behavioral equivalence: " .. tostring(S.analysis_status.behavioral_equivalence),
      "",
    }
    if S.hexvm then
      local hv = S.hexvm
      table.insert(evidence_lines, "----- BLOB CONTAINER (HexBlob-VM) -----")
      table.insert(evidence_lines, string.format(
        "blob_bytes=%d  declared_len=%d  decompressed=%d  length_match=%s  decode_ok=%s",
        hv.blob_bytes or 0, hv.declared_len or 0, hv.decompressed or 0,
        tostring(hv.length_match), tostring(hv.decode_ok)))
      table.insert(evidence_lines, string.format(
        "xor_key=%s  keylen=%d  program_djb2=%s%s",
        tostring(hv.xor_key), hv.keylen or 0, tostring(hv.program_djb2),
        (hv.derived_keys and #hv.derived_keys > 0)
          and ("  derived_keys=" .. table.concat(hv.derived_keys, "/")) or ""))
      if hv.string_consts then
        table.insert(evidence_lines, string.format(
          "translation: protos=%d  instructions=%d  strings=%d  numbers=%d  status=%s",
          hv.proto_count or 0, hv.instr_count or 0, hv.string_consts or 0,
          hv.number_consts or 0, tostring(hv.translation_status)))
        table.insert(evidence_lines, "  full constant pool -> deobfuscated_payload.txt")
        if (hv.full_deob_bytes or 0) > 0 then
          table.insert(evidence_lines, string.format(
            "  decoded VM IR -> full_deob.txt (%d bytes, instruction_key=%s)",
            hv.full_deob_bytes, tostring(hv.instruction_key)))
        end
      end
      table.insert(evidence_lines, "")
    end
    for i, item in ipairs(S.evidence) do
      table.insert(evidence_lines, string.format(
        "%03d [%s] [%s] %s confidence=%.2f — %s",
        i, item.severity, item.origin, item.id, item.confidence, item.detail
      ))
    end
    if #S.evidence > 0 or S.hexvm then
      write(S.path .. "/extracts/evidence_report.txt", table.concat(evidence_lines, "\n"))
    end
    if S.target_captured then
    write(S.path .. "/detect.txt", table.concat({
      "family: " .. tostring(detect.family),
      "confidence: " .. tostring(detect.confidence),
      "features: " .. table.concat(detect.features or {}, ", "),
      "families: " .. table.concat(detect.families or {}, ", "),
      "blob_bytes: " .. tostring(#S.largest_captured_stage),
      "loads: " .. tostring(S.counters.load),
      "threads_bound: " .. tostring(S.counters.bound),
      "hook_mode: " .. tostring(S.hook_mode),
      "mode: " .. (LITE_MODE and "lite" or "full"),
      "capture_scope: " .. (SCOPED_CAPTURE and "target_threads_only" or "global"),
      "target_captured: " .. tostring(S.target_captured),
      "prints: " .. (sink_off or tostring(S.counters.print)),
      "http: " .. (sink_off or tostring(S.counters.http)),
      "remotes: " .. (sink_off or tostring(S.counters.remote)),
      "services: " .. (sink_off or tostring(S.counters.svc)),
      "cff: " .. tostring(S.static and S.static.cff or 0),
      "density: " .. tostring(S.static and S.static.density or 0),
      "risk_score: " .. tostring(S.static and S.static.risk_score or 0),
      "verdict: " .. tostring(security_report.verdict),
      "deobfuscation_status: " .. tostring(S.analysis_status.deobfuscation),
      "runtime_validation: " .. tostring(S.analysis_status.runtime_validation),
      "behavioral_equivalence: " .. tostring(S.analysis_status.behavioral_equivalence),
      "evidence_items: " .. tostring(#S.evidence),
      "runtime_events: " .. tostring(#S.runtime_events),
      "duration_s: " .. string.format("%.2f", os.clock() - (S.capture_started or S.t0)),
      "write_bytes: " .. tostring(S.write_bytes),
      "write_limit_hit: " .. tostring(S.write_limit_hit),
    }, "\n"))
    write(S.path .. "/organized_summary.txt", table.concat({
      "===== SENTINEL ORGANIZED SUMMARY =====",
      "Session: " .. tostring(S.session),
      "Path: " .. tostring(S.path),
      "Time: " .. now() .. " UTC",
      "Detect: " .. tostring(detect.family) .. " (" .. tostring(detect.confidence) .. ")",
      "Final blob: " .. tostring(#S.largest_captured_stage) .. " bytes",
      "Loadstrings: " .. tostring(S.counters.load),
      "Threads auto-bound: " .. tostring(S.counters.bound),
      "Hook mode: " .. tostring(S.hook_mode),
      "Mode: " .. (LITE_MODE and "lite" or "full"),
      "Capture scope: " .. (SCOPED_CAPTURE and "target threads only" or "global"),
      "Target captured: " .. tostring(S.target_captured),
      "Load-graph edges: " .. tostring(#S.graph),
      "Prints: " .. (sink_off or tostring(S.counters.print)),
      "HTTP: " .. (sink_off or tostring(S.counters.http)),
      "Remotes: " .. (sink_off or tostring(S.counters.remote)),
      "Services: " .. (sink_off or tostring(S.counters.svc)),
      "Static strings: " .. tostring(S.static and #S.static.strings or 0),
      "CFF states: " .. tostring(S.static and S.static.cff or 0),
      "VM density: " .. tostring(S.static and S.static.density or 0),
      "Risk score: " .. tostring(S.static and S.static.risk_score or 0),
      "Verdict: " .. tostring(security_report.verdict),
      "Evidence items: " .. tostring(#S.evidence),
      "Runtime metadata events: " .. tostring(#S.runtime_events),
      "Deobfuscation: " .. tostring(S.analysis_status.deobfuscation),
      "Full deob IR: " .. ((type(S.full_deob) == "string" and #S.full_deob > 0)
        and ("full_deob.txt (" .. tostring(#S.full_deob) .. " bytes)") or "not produced"),
      "Written: " .. tostring(S.write_bytes) .. " / " .. tostring(MAX_LOCAL_WRITE_BYTES) .. " bytes",
      "Write limit reached: " .. tostring(S.write_limit_hit),
      "=====================================",
    }, "\n"))
    end -- if S.target_captured (detect.txt + organized_summary.txt)
    write(S.path .. "/environment_surface.txt", safe_json({
      executor = S.env and S.env.executor or "unknown",
      runtime = S.env and S.env.runtime or "unknown",
      touch_enabled = S.env and S.env.touch_enabled or false,
      keyboard_enabled = S.env and S.env.keyboard_enabled or false,
      viewport = S.env and S.env.viewport or {},
      api_surface = S.env and S.env.api_surface or {},
      key_context = S.env and S.env.key_context or {},
      privacy = S.env and S.env.privacy or {},
    }))
    local artifact_names = {
      "info.txt", "env_log.txt", "environment_surface.txt",
      "largest_captured_stage.txt", "final_blob.txt",
      "analysis_surface.txt", "deobfuscated_payload.txt", "full_deob.txt", "detect.txt",
      "security_findings.txt", "runtime_events.txt", "constant_inventory.txt",
      "endpoint_inventory.txt", "source_structure.txt",
      "organized_summary.txt", "extracts/evidence_report.txt",
      "extracts/static_report.txt", "extracts/strings.txt", "extracts/load_graph.txt",
      "extracts/cff_states.txt",
      "capture_failure.txt",
      "tool_diagnostics.txt", "capability_matrix.txt", "observations.txt",
      "provenance.txt", "vm_instructions.txt", "extracts/vm_instructions.txt",
      "extracts/vm_strings.txt",
      "layers.txt", "extracts/layers.txt", "layered_output.txt",
      "runtime_loads.txt", "runtime_constants.txt", "environment_changes.txt",
      "unknown.txt", "extracts/stage_hops.txt",
    }
    if not LITE_MODE then
      -- Extracts produced only by the full build's secondary hooks.  Listing
      -- them in lite would fill the manifest with present=false rows for files
      -- that were never going to exist.
      for _, name in ipairs({
        "extracts/prints.txt", "extracts/loadstrings.txt", "extracts/http.txt",
        "extracts/http_chain.txt", "extracts/remotes.txt", "extracts/constants.txt",
        "extracts/dumps.txt", "extracts/services.txt", "extracts/live_status.txt",
      }) do
        artifact_names[#artifact_names + 1] = name
      end
    end
    local artifacts = {}
    for _, item in ipairs(S.runtime_capture.loads) do
      artifact_names[#artifact_names + 1] = "extracts/stage_" .. string.format("%02d", item.id) .. ".txt"
    end
    for i = 1, S.snapshot_count do
      artifact_names[#artifact_names + 1] = "extracts/snapshot_" .. string.format("%02d", i) .. ".txt"
    end
    for _, name in ipairs(artifact_names) do
      table.insert(artifacts, { name = name, present = isfile_safe(S.path .. "/" .. name) })
    end
    write(S.path .. "/artifact_manifest.txt", safe_json({
      schema = "sentinel.artifact-manifest.v1",
      sentinel_version = VERSION,
      generated_utc = now(),
      session = S.session,
      force_stopped = S.force_stopped,
      artifacts = artifacts,
    }))
  end
  local pointer_ok = write("Sentinel_LATEST.txt", table.concat({
    "Sentinel Local " .. VERSION,
    "status=" .. (S.fs_error and "write_error" or (capture_failed and "capture_failed" or "saved")),
    "session=" .. tostring(S.session),
    "output=" .. tostring(S.path) .. "/",
    "family=" .. tostring(detect.family),
    "blob_bytes=" .. tostring(#S.largest_captured_stage),
    "hook_mode=" .. tostring(S.hook_mode),
    "written_bytes=" .. tostring(S.write_bytes),
    "error=" .. tostring(S.fs_error or "none"),
  }, "\n"))
  relay_post("finalize", { session = S.session, detect = detect, blob_bytes = #S.largest_captured_stage, counters = S.counters })
  if S.path then write(S.path .. "/tool_diagnostics.txt", safe_json(tool_diagnostics())) end
  if pointer_ok and not S.fs_error and not capture_failed then
    set_phase("done", "SAVED → " .. tostring(S.path) .. "/")
  elseif pointer_ok and not S.fs_error then
    set_phase("done", "NO TARGET → see capture_failure.txt")
  else
    set_phase("done", "SAVE ERROR → check console: " .. tostring(S.fs_error or "unknown"):sub(1, 60))
  end
  S._finalizing = false
  print(("[Sentinel] Stopped %s | family=%s blob=%dB → %s"):format(
    tostring(S.session), tostring(detect.family), #S.largest_captured_stage, tostring(S.path)
  ))
  return S.path
end

local function force_stop(reason)
  if S.force_stopped then return S.path end
  S.force_stopped = true
  S.queue_dropped = S.queue_dropped + #work_queue
  table.clear(work_queue)
  S.capture_pending = false
  restore_loader_hook()
  S.runtime_envs = {}
  if S._watch then S._watch:Disconnect(); S._watch = nil end
  if S._snap then S._snap:Disconnect(); S._snap = nil end
  record_runtime("force_stop", { reason = tostring(reason or "api"):sub(1, 40) })
  S.analysis_status.runtime_validation = "force_stopped"
  S.running = false
  S._finalizing = false
  set_phase("done", "Force stopped — hooks and collection disabled")
  if S._anim then S._anim:Disconnect(); S._anim = nil end
  if S.path then
    write(S.path .. "/force_stop.txt", table.concat({
      "Sentinel force-stopped",
      "time=" .. now() .. " UTC",
      "reason=" .. tostring(reason or "api"),
      "target_captured=" .. tostring(S.target_captured),
      "hook_restore=" .. tostring(S.restore_status),
      "discarded_jobs=" .. tostring(S.queue_dropped),
      "loads=" .. tostring(S.counters.load),
      "full analysis intentionally skipped",
    }, "\n"))
    write(S.path .. "/artifact_manifest.txt", safe_json({
      schema = "sentinel.artifact-manifest.v1",
      sentinel_version = VERSION,
      generated_utc = now(),
      session = S.session,
      force_stopped = true,
      analysis_skipped = true,
      artifacts = { { name = "force_stop.txt", present = true } },
    }))
    write("Sentinel_LATEST.txt", table.concat({
      "Sentinel Local " .. VERSION,
      "status=force_stopped",
      "session=" .. tostring(S.session),
      "output=" .. tostring(S.path) .. "/",
      "analysis=skipped",
      "error=" .. tostring(S.fs_error or "none"),
    }, "\n"))
  end
  return S.path
end
S._force_stop = force_stop

local function save_stop(reason)
  record_runtime("save_stop", { reason = tostring(reason or "api"):sub(1, 40) })
  return finalize()
end
S._save_stop = save_stop

-- ── API ─────────────────────────────────────────────────────────────
local Sentinel = {}

function Sentinel.Attach()
  if S.running and not S.target_captured then
    print("[Sentinel] already attached:", S.path)
    return S.path
  end
  if S.running and S.target_captured then finalize() end
  install_hooks()
  init_session()
  S.attached = true
  set_phase("idle", "Armed via " .. tostring(S.hook_mode) .. " — run target loadstring")
  print("[Sentinel] armed via " .. tostring(S.hook_mode) .. " — selected execution and nested loads")
  return S.path
end

function Sentinel.Run(src, opts)
  opts = opts or {}
  local requested = tonumber(opts.duration or opts.wait or opts.idle)
  if requested then COLLECTION_SECS = math.max(1, math.min(300, requested)) end
  if S.running then finalize() end
  install_hooks()
  init_session()
  S.attached = true
  src = tostring(src or "")
  set_phase("extracting", "Running target…")
  local fn, err = _wrapped_loadstring(src, "sentinel_target")
  if not fn then
    push_print("LOAD_ERROR", tostring(err))
    if S.path then write(S.path .. "/load_error.txt", tostring(err)) end
    return finalize()
  end
  local ok, res = pcall(fn)
  if not ok then
    push_print("RUNTIME_ERROR", tostring(res))
    if S.path then write(S.path .. "/runtime_error.txt", tostring(res)) end
  end
  while S.running do
    if S.counters.http > 0 or S.counters.remote > 0 then set_phase("tools", "Collecting side channels…") end
    task.wait(0.2)
  end
  return S.path
end

-- Direct loader entry bypasses executor global-binding discovery failures.
function Sentinel.Loadstring(src, name)
  if not S.running then Sentinel.Attach() end
  return _wrapped_loadstring(src, name or "Sentinel.Loadstring")
end

-- Explicit attribution for a callback Sentinel cannot see being scheduled.
-- CAPTURE.tasks covers threads the target starts and CAPTURE.connections
-- covers what it connects to signals; this is the escape hatch for the rest --
-- a callback handed to an executor API, or anything stored now and invoked
-- later through a path Sentinel does not wrap.
function Sentinel.Bind(callback)
  if type(callback) ~= "function" then error("Sentinel.Bind expects a function", 2) end
  if not S.running or not S.runtime_capture.selected then error("No selected runtime target", 2) end
  local context = target_threads[current_thread()]
  local record = { id = context and context.id or 1, execution = "not_executed" }
  return runtime_function(callback, record, S.generation)
end

function Sentinel.Analyze(src)
  src = tostring(src or "")
  if src == "" then return nil, "Provide non-empty source for static analysis" end
  if S.running then finalize() end
  init_session()
  S.target_captured = true
  if not S.capture_started then S.capture_started = os.clock() end
  if #src > MAX_SOURCE_BYTES then
    add_evidence("input.truncated", "info", "StaticAnalysis",
      "input limited to " .. tostring(MAX_SOURCE_BYTES) .. " bytes", 1)
    src = src:sub(1, MAX_SOURCE_BYTES)
  end
  S.counters.load = math.min(MAX_LOADS, S.counters.load + 1)
  S.largest_captured_stage = src
  table.insert(S.graph, {
    id = S.counters.load, parent = nil, name = "Sentinel.Analyze/input", bytes = #src,
  })
  local src_fingerprint = fingerprint(src)
  record_runtime("static_input", {
    load_id = S.counters.load,
    chunk = "Sentinel.Analyze/input",
    bytes = #src,
    fingerprint = src_fingerprint,
    executed = false,
    source_omitted = true,
  })
  add_evidence("analysis.static_input", "info", "StaticAnalysis",
    "source accepted without execution; bytes=" .. tostring(#src) .. " " .. src_fingerprint, 1)
  S.analysis_status.runtime_validation = "not_run_static_only"
  restore_loader_hook()
  set_phase("tools", "Static-only analysis — target will not execute")
  return finalize()
end

function Sentinel.FullDeob(src)
  local output = Sentinel.Analyze(src)
  if type(S.full_deob) == "string" and #S.full_deob > 0 then
    return output .. "/full_deob.txt", S.analysis_status.deobfuscation
  end
  return nil, "LuaSnapper VM IR was not recovered; inspect deobfuscated_payload.txt"
end

function Sentinel.SelfTest()
  if S.running then finalize() end
  install_hooks()
  init_session()
  local fixture = 'local compile = ...; return compile(\'return 42, nil, "sentinel-child"\', "sentinel-selftest-child")()'
  local fn, compile_error = Sentinel.Loadstring(fixture, "sentinel-selftest-root")
  local result = type(fn) == "function" and table.pack(pcall(fn, Sentinel.Loadstring))
    or { false, compile_error, n = 2 }
  local output = finalize()
  local r = S.runtime_capture
  local checks = {
    nested_stages = #r.loads == 2,
    parent_link = #r.loads == 2 and r.loads[2].parent == r.loads[1].id,
    return_values = result[1] == true and result.n == 4 and result[2] == 42
      and result[3] == nil and result[4] == "sentinel-child",
    constants_attempted = #r.constants == 2,
    environment_observations = #r.environments >= 4,
  }
  local passed = true
  for _, value in pairs(checks) do if not value then passed = false end end
  local report = { passed = passed, checks = checks, constants = {},
    automatic_hook_coverage = "not_tested_uses_explicit_loader", version = VERSION }
  for _, value in ipairs(r.constants) do
    report.constants[#report.constants + 1] = { status = value.status, count = value.count, api = value.api }
  end
  report.saved = write(output .. "/self_test.txt", safe_json(report))
  print("[Sentinel] SelfTest pipeline " .. (passed and "PASS" or "FAIL")
    .. " | report saved=" .. tostring(report.saved) .. " -> " .. output .. "/self_test.txt")
  for _, item in ipairs(report.constants) do
    print("[Sentinel] constants: " .. tostring(item.status) .. " count=" .. tostring(item.count))
  end
  return report, output
end

function Sentinel.Snapshot(label, data)
  if not S.running or S.force_stopped then return nil, "Start a session before taking a snapshot" end
  if S.snapshot_count >= 16 then return nil, "Session snapshot limit reached (16)" end
  if type(label) ~= "string" then return nil, "Snapshot label must be a string" end
  S.snapshot_count = S.snapshot_count + 1
  local traversed, report = measured("traversal", snapshot_value, data)
  if not traversed then return nil, "Snapshot traversal failed" end
  report.label = label:sub(1, 64):gsub("[^%w _%-]", "_")
  local path = S.path .. "/extracts/snapshot_" .. string.format("%02d", S.snapshot_count) .. ".txt"
  local session = S.session
  task.wait()
  if S.force_stopped or not S.running or S.session ~= session then return report, nil, "Snapshot cancelled" end
  local ok, encoded = measured("encoding", function() return HttpService:JSONEncode(report) end)
  if not ok then return report, nil, "Snapshot JSON encoding failed" end
  if #encoded > 128000 then return report, nil, "Snapshot output exceeds 128 KB" end
  task.wait()
  if S.force_stopped or not S.running or S.session ~= session then return report, nil, "Snapshot cancelled" end
  local saved, err = write(path, encoded)
  if not saved then return report, nil, err or "Write budget exhausted" end
  S.counters.dump = S.counters.dump + 1
  return report, path
end
function Sentinel.Dump(label, data) return Sentinel.Snapshot(label, data) end
